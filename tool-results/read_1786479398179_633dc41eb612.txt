   186→   186→// ============== CUSTOM FIELDS ==============
   187→   187→// Two-tier custom field system:
   188→   188→//   1. GlobalCustomField — a global library of field definitions, managed by
   189→   189→//      the Developer. These are the "master" definitions available to all orgs.
   190→   190→//   2. OrgCustomField — per-org selections: which global fields are enabled
   191→   191→//      for this org, plus an optional org-specific label override.
   192→   192→//
   193→   193→// This allows the Developer to build a library of fields (Blood Type, Chest
   194→   194→// Measurement, Fabric Preference, etc.) once, and each org enables the ones
   195→   195→// relevant to their business type. Orgs can also rename fields locally
   196→   196→// (e.g. "Chest" → "Dada" in Malay) without affecting the global definition.
   197→   197→//
   198→   198→// The existing CustomField model is kept for backward compatibility — the
   199→   199→// /api/custom-fields GET endpoint merges both sources so existing UI code
   200→   200→// continues to work.
   201→   201→
   202→   202→model GlobalCustomField {
   203→   203→  id              String   @id @default(cuid())
   204→   204→  label           String                    // Global label, e.g. "Blood Type"
   205→   205→  key             String   @unique          // Auto-derived: "blood_type"
   206→   206→  type            String   @default("TEXT") // TEXT, NUMBER, DATE, SELECT, TEXTAREA, REFERENCE
   207→   207→  options         String?                   // JSON array for SELECT, e.g. ["A+","O+"]
   208→   208→  unit            String?                   // "cm", "kg"
   209→   209→  required        Boolean  @default(false)
   210→   210→  targetEntity    String   @default("resident") // resident, invoice, product, staff
   211→   211→  referenceEntity String?                   // for REFERENCE type: product, staff, resident, invoice
   212→   212→  description     String?                   // Developer notes about this field
   213→   213→  // Which business type(s) this field belongs to. Null = all business types.
   214→   214→  // Stored as JSON array: ["nursing_home","clinic"] or null for all.
   215→   215→  businessTypes   String?
   216→   216→  sortOrder       Int      @default(0)
   217→   217→  active          Boolean  @default(true)
   218→   218→  createdAt       DateTime @default(now())
   219→   219→  updatedAt       DateTime @updatedAt
   220→   220→
   221→   221→  orgSelections   OrgCustomField[]
   222→   222→}
   223→   223→
   224→   224→model OrgCustomField {
   225→   225→  id              String   @id @default(cuid())
   226→   226→  orgId           String                    // links to Organization
   227→   227→  globalFieldId   String                    // links to GlobalCustomField
   228→   228→  globalField     GlobalCustomField @relation(fields: [globalFieldId], references: [id], onDelete: Cascade)
   229→   229→  // Optional org-specific label override. If null, uses the global label.
   230→   230→  labelOverride   String?
   231→   231→  // Optional org-specific sort order (overrides global sortOrder)
   232→   232→  sortOrderOverride Int?
   233→   233→  // Whether the field is enabled for this org (default true — selecting a
   234→   234→  // field enables it; deselecting sets this to false)
   235→   235→  enabled         Boolean  @default(true)
   236→   236→  createdAt       DateTime @default(now())
   237→   237→  updatedAt       DateTime @updatedAt
   238→   238→
   239→   239→  @@unique([orgId, globalFieldId])

   242→   242→// ============== CUSTOM TABS ==============
   243→   243→// Global custom tab definitions — a collection of fields (built-in + custom)
   244→   244→// that appears as a tab in the customer detail view. Managed by the Developer
   245→   245→// and applied globally. Each org can show/hide tabs and override labels.
   246→   246→//
   247→   247→// Example: A "Measurements" tab for tailors containing Chest, Waist, Hip
   248→   248→// custom fields. A "Medical Info" tab for clinics containing Blood Type,
   249→   249→// Allergies, etc.
   250→   250→
   251→   251→model GlobalCustomTab {
   252→   252→  id          String   @id @default(cuid())
   253→   253→  label       String                    // Tab name, e.g. "Measurements"
   254→   254→  key         String   @unique          // Auto-derived: "measurements"
   255→   255→  description String?                   // Developer notes
   256→   256→  // JSON array of field IDs included in this tab. Can reference:
   257→   257→  //   - GlobalCustomField IDs (custom fields)
   258→   258→  //   - Built-in field keys (e.g. "firstName", "allergies", "conditions")
   259→   259→  fields      String   @default("[]")  // JSON: ["fieldId1","fieldId2","allergies"]

   267→   267→  businessTypes String?
   268→   268→  sortOrder   Int      @default(0)
   269→   269→  active      Boolean  @default(true)
   270→   270→  createdAt   DateTime @default(now())

   290→   290→// ============== CUSTOM FIELD VALUE VERSIONS ==============
   291→   291→// Stores snapshots of custom field values at a point in time — like a
   292→   292→// "measurement history". Each time a user records new measurements (e.g.
   293→   293→// body measurements for a tailor), a new version is created instead of
   294→   294→// overwriting the old values. The latest version's values are also kept
   295→   295→// in CustomFieldValue (the "current" values) for quick display in forms.
   296→   296→//
   297→   297→// Example: A tailor takes measurements on Jan 1 (version 1), then again
   298→   298→// on Mar 1 (version 2). The customer detail shows the latest (Mar 1),
   299→   299→// but the user can view the history to see how measurements changed.
   300→   300→
   301→   301→model CustomFieldValueVersion {
   302→   302→  id          String   @id @default(cuid())
   303→   303→  entityId    String                    // the entity (e.g. resident ID)
   304→   304→  entityType  String   @default("resident")
   305→   305→  // JSON snapshot of all field values at this point: { "fieldId1": "value1", "fieldId2": "value2" }
   306→   306→  values      String   @default("{}")  // JSON object
   307→   307→  // Optional label for this version, e.g. "Initial", "After diet", "3-month checkup"
   308→   308→  label       String?
   309→   309→  recordedAt  DateTime @default(now())  // when the measurements were taken
   310→   310→  recordedBy  String?                   // user ID who recorded this version
   311→   311→  recordedByName String?                // user name (denormalized for display)
   312→   312→  createdAt   DateTime @default(now())
   313→   313→
   314→   314→  @@index([entityId, entityType, recordedAt])
   315→   315→}

   418→   418→//   MATERNITY — Maternity leave (60 consecutive days, Employment Act s37)
   419→   419→//   PATERNITY — Paternity leave (3 consecutive days, 2023 amendment)
   420→   420→//   MARRIAGE — Marriage leave (3 days, 2023 amendment)
   421→   421→//   PATERNITY_Misc — Compassionate/ Bereavement leave (1-3 days)
   422→   422→//   EMERGENCY — Emergency leave (unpaid)

   432→   432→  halfDay       Boolean  @default(false) // half-day leave
   433→   433→  status        String   @default("PENDING") // PENDING, APPROVED, REJECTED, CANCELLED
   434→   434→  reason        String?
   435→   435→  daysCount     Float?    // computed: number of leave days (excluding rest days/public holidays)

  1267→  1267→  createdById     String?
