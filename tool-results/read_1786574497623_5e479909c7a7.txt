     1→'use client'
     2→
     3→import { useState } from 'react'
     4→import { useFetch, apiPatch } from './api'
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     6→import { Skeleton } from '@/components/ui/skeleton'
     7→import { Button } from '@/components/ui/button'
     8→import { Badge } from '@/components/ui/badge'
     9→import { Input } from '@/components/ui/input'
    10→import { StatusBadge } from './Badges'
    11→import { fmtDateTime, fmtDate } from '@/lib/types'
    12→import { Calendar, Stethoscope, Edit, X, Activity } from 'lucide-react'
    13→import { toast } from 'sonner'
    14→import { StandardSearchBar } from './StandardSearchBar'
    15→
    16→export function Visits({ facilityId }: { facilityId?: string }) {
    17→  const [tab, setTab] = useState<'upcoming' | 'past'>('upcoming')
    18→  const [search, setSearch] = useState('')
    19→  const [noteVisitId, setNoteVisitId] = useState<string | null>(null)
    20→  const [cancellingId, setCancellingId] = useState<string | null>(null)
    21→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
    22→  const { data, loading, refetch } = useFetch<any[]>(`/api/data?type=visits${tab === 'upcoming' ? '&upcoming=true' : ''}${facilityParam}`)
    23→
    24→  const all = data || []
    25→  const filtered = all.filter(v => {
    26→    if (!search) return true
    27→    const s = search.toLowerCase()
    28→    return (
    29→      `${v.resident?.firstName} ${v.resident?.lastName}`.toLowerCase().includes(s) ||
    30→      v.resident?.code?.toLowerCase().includes(s) ||
    31→      v.resident?.room?.roomNumber?.toLowerCase().includes(s) ||
    32→      v.visitType?.toLowerCase().includes(s) ||
    33→      v.staff && `${v.staff.firstName} ${v.staff.lastName}`.toLowerCase().includes(s)
    34→    )
    35→  })
    36→
    37→  // Group by date
    38→  const grouped: Record<string, any[]> = {}
    39→  for (const v of filtered) {
    40→    const k = fmtDate(v.scheduledAt)
    41→    if (!grouped[k]) grouped[k] = []
    42→    grouped[k].push(v)
    43→  }
    44→
    45→  // Find the visit being noted
    46→  const noteVisit = noteVisitId ? all.find(v => v.id === noteVisitId) : null
    47→
    48→  return (
    49→    <div className="space-y-4">
    50→      <div className="flex gap-2 border-b">
    51→        {(['upcoming', 'past'] as const).map(t => (
    52→          <button key={t} onClick={() => { setTab(t); setSearch('') }} className={`px-4 py-2 text-sm border-b-2 capitalize ${tab === t ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground'}`}>
    53→            {t === 'upcoming' ? 'Upcoming' : 'Past Visits'}
    54→          </button>
    55→        ))}
    56→      </div>
    57→
    58→      <StandardSearchBar
    59→        value={search}
    60→        onChange={setSearch}
    61→        placeholder="Search by resident, code, room, visit type, staff..."
    62→        totalCount={all.length}
    63→        filteredCount={filtered.length}
    64→      />
    65→
    66→      {Object.keys(grouped).length === 0 && <p className="text-sm text-muted-foreground text-center py-8">No {tab} visits</p>}
    67→
    68→      {Object.entries(grouped).map(([date, visits]) => (
    69→        <div key={date}>
    70→          <h3 className="text-sm font-semibold text-muted-foreground mb-2">{date}</h3>
    71→          <div className="grid gap-2">
    72→            {visits.map(v => (
    73→              <Card key={v.id}>
    74→                <CardContent className="p-3">
    75→                  <div className="flex flex-col sm:flex-row sm:items-start justify-between gap-2">
    76→                    <div className="flex-1 min-w-0">
    77→                      <div className="flex items-center gap-2 flex-wrap">
    78→                        <Badge variant="outline">{v.visitType.replace(/_/g, ' ')}</Badge>
    79→                        <StatusBadge status={v.status} />
    80→                        <span className="text-xs text-muted-foreground whitespace-nowrap">{fmtDateTime(v.scheduledAt)}</span>
    81→                      </div>
    82→                      <div className="font-medium mt-1">
    83→                        {v.resident?.code && <span className="text-xs font-mono text-primary mr-1">{v.resident.code}</span>}
    84→                        {v.resident?.firstName} {v.resident?.lastName} <span className="text-xs text-muted-foreground">Room {v.resident?.room?.roomNumber || '—'}</span>
    85→                      </div>
    86→                      {v.staff && <div className="text-xs text-muted-foreground mt-0.5 whitespace-nowrap">With: {v.staff.firstName} {v.staff.lastName} ({v.staff.role.replace(/_/g, ' ')})</div>}
    87→
    88→                      {/* Clinical note fields — shown if filled */}
    89→                      {v.chiefComplaint && <div className="text-sm mt-2"><span className="font-medium">Chief Complaint:</span> {v.chiefComplaint}</div>}
    90→                      {v.vitalsNote && <div className="text-sm mt-1"><span className="font-medium">Vitals:</span> {v.vitalsNote}</div>}
    91→                      {v.findings && <div className="text-sm mt-1"><span className="font-medium">Findings:</span> {v.findings}</div>}
    92→                      {v.diagnosis && <div className="text-sm mt-1"><span className="font-medium">Diagnosis:</span> {v.diagnosis}</div>}
    93→                      {v.treatmentPlan && <div className="text-sm mt-1"><span className="font-medium">Treatment Plan:</span> {v.treatmentPlan}</div>}
    94→                      {v.prescription && <div className="text-sm mt-1"><span className="font-medium">Prescription:</span> {v.prescription}</div>}
    95→                      {v.followUpNote && <div className="text-sm mt-1"><span className="font-medium">Follow-up:</span> {v.followUpNote}</div>}
    96→                      {v.recommendations && <div className="text-sm mt-1"><span className="font-medium">Recommendations:</span> {v.recommendations}</div>}
    97→                      {v.completedAt && <div className="text-xs text-muted-foreground mt-1 whitespace-nowrap">Completed: {fmtDate(v.completedAt)}{v.duration ? ` • ${v.duration} min` : ''}</div>}
    98→                    </div>
    99→                    <div className="flex gap-1 flex-shrink-0">
   100→                      {v.status === 'SCHEDULED' && (
   101→                        <>
   102→                          <Button size="sm" variant="outline" onClick={() => setNoteVisitId(v.id)}>
   103→                            <Edit className="h-3 w-3 mr-1" /> Fill Form
   104→                          </Button>
   105→                          <Button size="sm" variant="ghost" className="text-red-600" disabled={cancellingId === v.id} onClick={async () => {
   106→                            if (!confirm('Cancel this visit? This cannot be undone.')) return
   107→                            setCancellingId(v.id)
   108→                            try {
   109→                              await apiPatch(`/api/data?type=visits&id=${v.id}`, { status: 'CANCELLED' })
   110→                              toast.success('Visit cancelled')
   111→                              refetch()
   112→                            } catch (e: any) {
   113→                              toast.error(e.message || 'Failed to cancel visit')
   114→                            }
   115→                            setCancellingId(null)
   116→                          }}>
   117→                            <X className="h-3 w-3" />
   118→                          </Button>
   119→                        </>
   120→                      )}
   121→                      {v.status === 'COMPLETED' && (
   122→                        <Button size="sm" variant="ghost" onClick={() => setNoteVisitId(v.id)} title="View visit notes">
   123→                          <Stethoscope className="h-3 w-3" /> View
   124→                        </Button>
   125→                      )}
   126→                    </div>
   127→                  </div>
   128→                </CardContent>
   129→              </Card>
   130→            ))}
   131→          </div>
   132→        </div>
   133→      ))}
   134→
   135→      {/* Visit note dialog */}
   136→      {noteVisit && (
   137→        <VisitFormDialog visit={noteVisit} onClose={() => setNoteVisitId(null)} onSaved={() => { setNoteVisitId(null); refetch() }} />
   138→      )}
   139→    </div>
   140→  )
   141→}
   142→
   143→/**
   144→ * VisitFormDialog — clinical notes form for completing a visit.
   145→ * Includes chief complaint, vitals (auto-saved to Vital Signs), findings,
   146→ * diagnosis, treatment plan, prescription, and follow-up.
   147→ */
   148→function VisitFormDialog({ visit, onClose, onSaved }: { visit: any; onClose: () => void; onSaved: () => void }) {
   149→  const { data: currentUser } = useFetch<any>('/api/auth/me')
   150→
   151→  // No access denial — if the user can see the button, they can fill/edit.
   152→  // Buttons are only shown to users with module access (controlled by role/level).
   153→  const [editMode, setEditMode] = useState(false)
   154→  const isReadOnly = visit.status === 'COMPLETED' && !editMode
   155→
   156→  // Prefill start/end with current date/time (formatted for datetime-local input)
   157→  const nowLocal = () => {
   158→    const d = new Date()
   159→    const off = d.getTimezoneOffset()
   160→    return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16)
   161→  }
   162→  const addMinutes = (dtStr: string, mins: number) => {
   163→    const d = new Date(dtStr)
   164→    d.setMinutes(d.getMinutes() + mins)
   165→    const off = d.getTimezoneOffset()
   166→    return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16)
   167→  }
   168→
   169→  const [form, setForm] = useState<any>({
   170→    visitStart: visit.completedAt ? new Date(visit.completedAt).toISOString().slice(0, 16) : nowLocal(),
   171→    visitEnd: visit.completedAt && visit.duration ? addMinutes(new Date(visit.completedAt).toISOString().slice(0, 16), visit.duration) : addMinutes(nowLocal(), 30),
   172→    chiefComplaint: visit.chiefComplaint || '',
   173→    // Structured vitals fields
   174→    bpSys: visit.vitalsNote ? (visit.vitalsNote.match(/BP\s*(\d+)/i)?.[1] || '') : '',
   175→    bpDia: visit.vitalsNote ? (visit.vitalsNote.match(/BP\s*\d+\/(\d+)/i)?.[1] || '') : '',
   176→    hr: visit.vitalsNote ? (visit.vitalsNote.match(/HR\s*(\d+)/i)?.[1] || '') : '',
   177→    temp: visit.vitalsNote ? (visit.vitalsNote.match(/Temp\s*([\d.]+)/i)?.[1] || '') : '',
   178→    rr: visit.vitalsNote ? (visit.vitalsNote.match(/RR\s*(\d+)/i)?.[1] || '') : '',
   179→    o2: visit.vitalsNote ? (visit.vitalsNote.match(/SpO2?\s*(\d+)/i)?.[1] || '') : '',
   180→    bs: visit.vitalsNote ? (visit.vitalsNote.match(/BS\s*([\d.]+)/i)?.[1] || '') : '',
   181→    wt: visit.vitalsNote ? (visit.vitalsNote.match(/Wt\s*([\d.]+)/i)?.[1] || '') : '',
   182→    vitalsNote: visit.vitalsNote || '',
   183→    findings: visit.findings || '',
   184→    diagnosis: visit.diagnosis || '',
   185→    treatmentPlan: visit.treatmentPlan || '',
   186→    prescription: visit.prescription || '',
   187→    followUpNote: visit.followUpNote || '',
   188→    recommendations: visit.recommendations || '',
   189→    duration: visit.duration || '',
   190→  })
   191→  const [saving, setSaving] = useState(false)
   192→
   193→  // Auto-calculate duration when start/end changes
   194→  const calcDuration = () => {
   195→    if (form.visitStart && form.visitEnd) {
   196→      const start = new Date(form.visitStart)
   197→      const end = new Date(form.visitEnd)
   198→      const diffMin = Math.round((end.getTime() - start.getTime()) / 60000)
   199→      if (diffMin > 0) return diffMin
   200→    }
   201→    return form.duration ? parseInt(form.duration) : null
   202→  }
   203→
   204→  const submit = async () => {
   205→    setSaving(true)
   206→    try {
   207→      const userName = currentUser?.user?.name || 'Unknown'
   208→      const duration = calcDuration()
   209→
   210→      // Build vitalsNote string from structured fields (for display on the visit record)
   211→      const vitalsParts: string[] = []
   212→      if (form.bpSys && form.bpDia) vitalsParts.push(`BP ${form.bpSys}/${form.bpDia}`)
   213→      if (form.hr) vitalsParts.push(`HR ${form.hr}`)
   214→      if (form.temp) vitalsParts.push(`Temp ${form.temp}`)
   215→      if (form.rr) vitalsParts.push(`RR ${form.rr}`)
   216→      if (form.o2) vitalsParts.push(`SpO2 ${form.o2}`)
   217→      if (form.bs) vitalsParts.push(`BS ${form.bs}`)
   218→      if (form.wt) vitalsParts.push(`Wt ${form.wt}`)
   219→      const vitalsNoteStr = vitalsParts.join(', ')
   220→
   221→      const payload: any = {
   222→        chiefComplaint: form.chiefComplaint || null,
   223→        vitalsNote: vitalsNoteStr || null,
   224→        findings: form.findings || null,
   225→        diagnosis: form.diagnosis || null,
   226→        treatmentPlan: form.treatmentPlan || null,
   227→        prescription: form.prescription || null,
   228→        followUpNote: form.followUpNote || null,
   229→        recommendations: form.recommendations || null,
   230→        duration: duration,
   231→        scheduledAt: form.visitStart ? new Date(form.visitStart).toISOString() : undefined,
   232→      }
   233→      if (visit.status === 'SCHEDULED') {
   234→        payload.status = 'COMPLETED'
   235→        payload.completedAt = form.visitEnd ? new Date(form.visitEnd).toISOString() : new Date().toISOString()
   236→        payload.completedById = currentUser?.user?.id || null
   237→        payload.completedByName = userName
   238→      } else {
   239→        payload.completedAt = form.visitEnd ? new Date(form.visitEnd).toISOString() : undefined
   240→      }
   241→      await apiPatch(`/api/data?type=visits&id=${visit.id}`, payload)
   242→
   243→      // Auto-create a VitalSign record from the structured vitals fields
   244→      const vitalsData: any = {}
   245→      if (form.bpSys) vitalsData.bloodPressureSystolic = parseInt(form.bpSys)
   246→      if (form.bpDia) vitalsData.bloodPressureDiastolic = parseInt(form.bpDia)
   247→      if (form.hr) vitalsData.heartRate = parseInt(form.hr)
   248→      if (form.temp) vitalsData.temperature = parseFloat(form.temp)
   249→      if (form.rr) vitalsData.respiratoryRate = parseInt(form.rr)
   250→      if (form.o2) vitalsData.oxygenSaturation = parseInt(form.o2)
   251→      if (form.bs) vitalsData.bloodSugar = parseFloat(form.bs)
   252→      if (form.wt) vitalsData.weight = parseFloat(form.wt)
   253→
   254→      if (Object.keys(vitalsData).length > 0) {
   255→        try {
   256→          const { apiPost } = await import('./api')
   257→          await apiPost('/api/data?type=vitals', {
   258→            residentId: visit.residentId,
   259→            ...vitalsData,
   260→            notes: `Recorded during ${visit.visitType.replace(/_/g, ' ')} visit by ${userName}`,
   261→            recordedById: currentUser?.user?.id || null,
   262→          })
   263→          toast.success('Vitals auto-saved to Vital Signs module')
   264→        } catch (e: any) {
   265→          console.log('Vitals auto-save failed (non-critical):', e.message)
   266→        }
   267→      }
   268→
   269→      toast.success(visit.status === 'SCHEDULED' ? 'Visit form submitted — moved to Past Visits' : 'Visit notes updated')
   270→      onSaved()
   271→    } catch (e: any) { toast.error(e.message) }
   272→    setSaving(false)
   273→  }
   274→
   275→  return (
   276→    <div className="fixed inset-0 bg-black/50 z-50 flex items-start sm:items-center justify-center p-4 overflow-y-auto">
   277→      <div className="bg-background rounded-lg shadow-xl w-full max-w-lg my-8 max-h-[calc(100vh-4rem)] flex flex-col">
   278→        <div className="flex justify-between items-center border-b p-4 flex-shrink-0">
   279→          <h3 className="font-semibold">
   280→            {isReadOnly ? 'Visit Notes (Read-Only)' : visit.status === 'SCHEDULED' ? 'Visit Form — Clinical Notes' : 'Edit Visit Notes'}
   281→          </h3>
   282→          <Button variant="ghost" size="sm" onClick={onClose}>×</Button>
   283→        </div>
   284→        <div className="p-4 overflow-y-auto flex-1 space-y-3">
   285→          {/* Visit info */}
   286→          <div className="text-xs text-muted-foreground bg-muted/30 rounded p-2 flex flex-wrap items-center gap-x-2 gap-y-1">
   287→            <Badge variant="outline">{visit.visitType.replace(/_/g, ' ')}</Badge>
   288→            <span className="whitespace-nowrap">Scheduled: {fmtDateTime(visit.scheduledAt)}</span>
   289→            {visit.staff && <span className="whitespace-nowrap">• Assigned: {visit.staff.firstName} {visit.staff.lastName}</span>}
   290→            {visit.resident && <span className="whitespace-nowrap">• {visit.resident.firstName} {visit.resident.lastName}</span>}
   291→            {visit.completedByName && <span className="whitespace-nowrap">• Filled by: {visit.completedByName}</span>}
   292→            {isReadOnly && <Badge variant="outline" className="text-[9px] text-amber-700 border-amber-300">Read-Only</Badge>}
   293→          </div>
   294→
   295→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 text-sm">
   296→            {/* Visit start & end date/time — prefilled with current time */}
   297→            <div>
   298→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Visit Start *</label>
   299→              <Input type="datetime-local" value={form.visitStart} onChange={e => setForm({ ...form, visitStart: e.target.value })} disabled={isReadOnly} className="text-sm" />
   300→            </div>
   301→            <div>
   302→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Visit End</label>
   303→              <Input type="datetime-local" value={form.visitEnd} onChange={e => setForm({ ...form, visitEnd: e.target.value })} disabled={isReadOnly} className="text-sm" />
   304→              {calcDuration() && <p className="text-[9px] text-muted-foreground mt-0.5">Duration: {calcDuration()} min</p>}
   305→            </div>
   306→            <div className="sm:col-span-2">
   307→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Chief Complaint</label>
   308→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.chiefComplaint} onChange={e => setForm({ ...form, chiefComplaint: e.target.value })} disabled={isReadOnly} placeholder="Patient's main complaint, e.g. lower back pain for 2 weeks" />
   309→            </div>
   310→            <div className="sm:col-span-2 border rounded-md p-2 bg-emerald-50/30">
   311→              <div className="text-xs font-semibold text-emerald-700 mb-2 flex items-center gap-1">
   312→                <Activity className="h-3.5 w-3.5" /> Vital Signs <span className="text-[9px] text-muted-foreground font-normal">(auto-saves to Vital Signs module)</span>
   313→              </div>
   314→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   315→                <div>
   316→                  <label className="text-[10px] text-muted-foreground block mb-0.5">BP Systolic (mmHg)</label>
   317→                  <Input type="number" value={form.bpSys || ''} onChange={e => setForm({ ...form, bpSys: e.target.value })} disabled={isReadOnly} placeholder="140" className="text-sm h-8" />
   318→                </div>
   319→                <div>
   320→                  <label className="text-[10px] text-muted-foreground block mb-0.5">BP Diastolic (mmHg)</label>
   321→                  <Input type="number" value={form.bpDia || ''} onChange={e => setForm({ ...form, bpDia: e.target.value })} disabled={isReadOnly} placeholder="90" className="text-sm h-8" />
   322→                </div>
   323→                <div>
   324→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Heart Rate (bpm)</label>
   325→                  <Input type="number" value={form.hr || ''} onChange={e => setForm({ ...form, hr: e.target.value })} disabled={isReadOnly} placeholder="72" className="text-sm h-8" />
   326→                </div>
   327→                <div>
   328→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Temperature (°C)</label>
   329→                  <Input type="number" step="0.1" value={form.temp || ''} onChange={e => setForm({ ...form, temp: e.target.value })} disabled={isReadOnly} placeholder="37.2" className="text-sm h-8" />
   330→                </div>
   331→                <div>
   332→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Resp Rate (/min)</label>
   333→                  <Input type="number" value={form.rr || ''} onChange={e => setForm({ ...form, rr: e.target.value })} disabled={isReadOnly} placeholder="18" className="text-sm h-8" />
   334→                </div>
   335→                <div>
   336→                  <label className="text-[10px] text-muted-foreground block mb-0.5">SpO₂ (%)</label>
   337→                  <Input type="number" value={form.o2 || ''} onChange={e => setForm({ ...form, o2: e.target.value })} disabled={isReadOnly} placeholder="98" className="text-sm h-8" />
   338→                </div>
   339→                <div>
   340→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Blood Sugar (mmol/L)</label>
   341→                  <Input type="number" step="0.1" value={form.bs || ''} onChange={e => setForm({ ...form, bs: e.target.value })} disabled={isReadOnly} placeholder="5.5" className="text-sm h-8" />
   342→                </div>
   343→                <div>
   344→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Weight (kg)</label>
   345→                  <Input type="number" step="0.1" value={form.wt || ''} onChange={e => setForm({ ...form, wt: e.target.value })} disabled={isReadOnly} placeholder="65" className="text-sm h-8" />
   346→                </div>
   347→              </div>
   348→            </div>
   349→            <div className="sm:col-span-2">
   350→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Findings (examination)</label>
   351→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.findings} onChange={e => setForm({ ...form, findings: e.target.value })} disabled={isReadOnly} placeholder="Physical examination findings, observations" />
   352→            </div>
   353→            <div className="sm:col-span-2">
   354→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Diagnosis</label>
   355→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.diagnosis} onChange={e => setForm({ ...form, diagnosis: e.target.value })} disabled={isReadOnly} placeholder="Working diagnosis / assessment" />
   356→            </div>
   357→            <div className="sm:col-span-2">
   358→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Treatment Plan</label>
   359→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.treatmentPlan} onChange={e => setForm({ ...form, treatmentPlan: e.target.value })} disabled={isReadOnly} placeholder="Management plan, procedures, therapy" />
   360→            </div>
   361→            <div className="sm:col-span-2">
   362→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Prescription</label>
   363→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.prescription} onChange={e => setForm({ ...form, prescription: e.target.value })} disabled={isReadOnly} placeholder="Medications prescribed, dosage, frequency" />
   364→            </div>
   365→            <div className="sm:col-span-2">
   366→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Follow-up Instructions</label>
   367→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.followUpNote} onChange={e => setForm({ ...form, followUpNote: e.target.value })} disabled={isReadOnly} placeholder="Follow-up advice, next appointment, referral" />
   368→            </div>
   369→            <div className="sm:col-span-2">
   370→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Recommendations</label>
   371→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.recommendations} onChange={e => setForm({ ...form, recommendations: e.target.value })} disabled={isReadOnly} placeholder="Additional recommendations for care staff" />
   372→            </div>
   373→          </div>
   374→        </div>
   375→        {/* Footer: edit mode shows Cancel + Submit; read-only shows Edit + Close */}
   376→        {(!isReadOnly || editMode) ? (
   377→        <div className="flex justify-end gap-2 p-4 border-t flex-shrink-0">
   378→          {editMode && (
   379→            <Button variant="outline" onClick={() => setEditMode(false)}>Cancel</Button>
   380→          )}
   381→          {!editMode && (
   382→            <Button variant="outline" onClick={onClose}>Cancel</Button>
   383→          )}
   384→          <Button onClick={submit} disabled={saving}>
   385→            {saving ? 'Submitting...' : visit.status === 'SCHEDULED' ? 'Submit Form' : 'Update Notes'}
   386→          </Button>
   387→        </div>
   388→        ) : (
   389→        <div className="flex justify-end gap-2 p-4 border-t flex-shrink-0">
   390→          <Button onClick={() => setEditMode(true)}>
   391→            <Edit className="h-3.5 w-3.5 mr-1" /> Edit
   392→          </Button>
   393→          <Button variant="outline" onClick={onClose}>Close</Button>
   394→        </div>
   395→        )}
   396→      </div>
   397→    </div>
   398→  )
   399→}

   401→/**
   402→ * Parses vital signs from free-text input.
   403→ * Supports: "BP 140/90, HR 72, Temp 37.2, RR 18, SpO2 98, BS 5.5, Wt 65"
   404→ */
   405→function parseVitalsFromText(text: string): any | null {
   406→  const result: any = {}
   407→  const lower = text.toLowerCase()
   408→
   409→  const bpMatch = lower.match(/(?:bp|blood\s*pressure)?\s*(\d{2,3})\s*\/\s*(\d{2,3})/)
   410→  if (bpMatch) { result.bloodPressureSystolic = parseInt(bpMatch[1]); result.bloodPressureDiastolic = parseInt(bpMatch[2]) }
   411→
   412→  const hrMatch = lower.match(/(?:hr|heart\s*rate|pulse)\s*:?\s*(\d{2,3})/)
   413→  if (hrMatch) result.heartRate = parseInt(hrMatch[1])
   414→
   415→  const tempMatch = lower.match(/(?:temp|temperature)?\s*:?\s*(\d{2,3}(?:\.\d)?)\s*(?:°?c|°?f)?/i)
   416→  if (tempMatch) {
   417→    const temp = parseFloat(tempMatch[1])
   418→    if (temp >= 30 && temp <= 45) result.temperature = temp
   419→    else if (temp >= 90 && temp <= 115) result.temperature = parseFloat(((temp - 32) * 5/9).toFixed(1))
   420→  }
   421→
   422→  const rrMatch = lower.match(/(?:rr|resp(?:iratory)?\s*rate)\s*:?\s*(\d{2,3})/)
   423→  if (rrMatch) { const rr = parseInt(rrMatch[1]); if (rr >= 8 && rr <= 60) result.respiratoryRate = rr }
   424→
   425→  const o2Match = lower.match(/(?:spo2|o2|oxygen|sat)\s*:?\s*(\d{2,3})\s*%?/)
   426→  if (o2Match) { const o2 = parseInt(o2Match[1]); if (o2 >= 50 && o2 <= 100) result.oxygenSaturation = o2 }
   427→
   428→  const bsMatch = lower.match(/(?:bs|blood\s*sugar|glucose|bgl)\s*:?\s*(\d{1,3}(?:\.\d)?)\s*(?:mmol\/l|mg\/dl)?/)
   429→  if (bsMatch) {
   430→    const bs = parseFloat(bsMatch[1])
   431→    if (bs >= 3 && bs <= 30) result.bloodSugar = bs
   432→    else if (bs >= 50 && bs <= 500) result.bloodSugar = parseFloat((bs / 18.0182).toFixed(1))
   433→  }
   434→
   435→  const wtMatch = lower.match(/(?:wt|weight)\s*:?\s*(\d{2,3}(?:\.\d)?)\s*kg?/)
   436→  if (wtMatch) result.weight = parseFloat(wtMatch[1])
   437→
   438→  return Object.keys(result).length > 0 ? result : null
   439→}
