    16→    16→async function apiPost(url: string, body: any) {
    17→    17→  const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
    18→    18→  if (!r.ok) throw new Error((await r.json().catch(() => ({}))).error || `HTTP ${r.status}`)
    19→    19→  return r.json()
    20→    20→}

    22→    22→export function SettingsModule() {
    23→    23→  const [activeSection, setActiveSection] = useState<string>('medications')
    24→    24→  const [settings, setSettings] = useState<any>(null)
    25→    25→
    26→    26→  const loadSettings = useCallback(async () => {
    27→    27→    try {
    28→    28→      const r = await fetch('/api/settings')
    29→    29→      const data = await r.json()
    30→    30→      setSettings(data)
    31→    31→    } catch (e: any) {
    32→    32→      toast.error('Failed to load settings')
    33→    33→    }
    34→    34→  }, [])
    35→    35→
    36→    36→  useEffect(() => { loadSettings() }, [loadSettings])
    37→    37→
    38→    38→  const saveSetting = async (key: string, value: any) => {
    39→    39→    try {
    40→    40→      await apiPost('/api/settings', { key, value })
    41→    41→      toast.success('Setting saved')
    42→    42→      setSettings({ ...settings, [key]: value })
    43→    43→    } catch (e: any) {
    44→    44→      toast.error(e.message)
    45→    45→    }
    46→    46→  }
    47→    47→

   229→   229→function UserLevelSettings({ settings, saveSetting }: { settings: any; saveSetting: (k: string, v: any) => void }) {
   230→   230→  const levels: any[] = settings.userLevels || []
   231→   231→  const [editing, setEditing] = useState<number | null>(null)
   232→   232→  const [editLabel, setEditLabel] = useState('')
   233→   233→  const [editDesc, setEditDesc] = useState('')
   234→   234→  const [editRoles, setEditRoles] = useState('')
   235→   235→  const [newLabel, setNewLabel] = useState('')
   236→   236→  const [newDesc, setNewDesc] = useState('')
   237→   237→  const [newRoles, setNewRoles] = useState('')
   238→   238→
   239→   239→  const levelColors: Record<number, string> = {
   240→   240→    1: 'bg-rose-100 text-rose-700',
   241→   241→    2: 'bg-orange-100 text-orange-700',
   242→   242→    3: 'bg-sky-100 text-sky-700',
   243→   243→    4: 'bg-emerald-100 text-emerald-700',
   244→   244→    5: 'bg-violet-100 text-violet-700',
   245→   245→    6: 'bg-amber-100 text-amber-700',
   246→   246→    7: 'bg-teal-100 text-teal-700',
   247→   247→    8: 'bg-pink-100 text-pink-700',
   248→   248→  }
   249→   249→
   250→   250→  const updateLevel = (index: number, label: string, desc: string, rolesStr: string) => {
   251→   251→    const updated = [...levels]
   252→   252→    const roles = rolesStr.split(',').map((r: string) => r.trim().toUpperCase()).filter(Boolean)
   253→   253→    updated[index] = { ...updated[index], label, desc, roles }
   254→   254→    saveSetting('userLevels', updated)
   255→   255→    setEditing(null)
   256→   256→  }
   257→   257→
   258→   258→  const addLevel = () => {
   259→   259→    if (!newLabel.trim()) { toast.error('Label required'); return }
   260→   260→    const nextLevel = Math.max(...levels.map((l: any) => l.level), 0) + 1
   261→   261→    const roles = newRoles.split(',').map(r => r.trim().toUpperCase()).filter(Boolean)
   262→   262→    const updated = [...levels, { level: nextLevel, label: newLabel.trim(), desc: newDesc.trim(), roles }]
   263→   263→    saveSetting('userLevels', updated)
   264→   264→    setNewLabel(''); setNewDesc(''); setNewRoles('')
   265→   265→    toast.success(`Level ${nextLevel} added`)
   266→   266→  }
   267→   267→
   268→   268→  const deleteLevel = (index: number) => {
   269→   269→    const level = levels[index]
   270→   270→    if (level.level <= 2) { toast.error('Cannot delete Level 1 (Owner) or Level 2 (Manager)'); return }
   271→   271→    if (!confirm(`Delete Level ${level.level} (${level.label})? Users at this level will need to be reassigned.`)) return
   272→   272→    const updated = levels.filter((_: any, i: number) => i !== index)
   273→   273→    saveSetting('userLevels', updated)
   274→   274→    toast.success('Level deleted')
   275→   275→  }
   276→   276→
   277→   277→  return (
   278→   278→    <div className="space-y-4">
   279→   279→      <Card>
   280→   280→        <CardHeader>
   281→   281→          <CardTitle className="text-sm flex items-center gap-2"><Users className="h-4 w-4" /> User Hierarchy Levels</CardTitle>
   282→   282→          <CardDescription>Edit, add, or remove user levels. Level 1 (Owner) is highest — can see all. Lower levels can only see users at their level or below.</CardDescription>
   283→   283→        </CardHeader>
   284→   284→        <CardContent>
   285→   285→          <div className="space-y-2">
   286→   286→            {levels.map((l: any, i: number) => (
   287→   287→              <div key={l.level} className="flex items-center gap-3 p-3 border rounded-md">
   288→   288→                {editing === i ? (
   289→   289→                  <>
   290→   290→                    <Badge variant="outline" className={`font-bold ${levelColors[l.level] || 'bg-muted'}`}>L{l.level}</Badge>
   291→   291→                    <div className="flex-1 grid grid-cols-1 sm:grid-cols-3 gap-2">
   292→   292→                      <Input value={editLabel} onChange={e => setEditLabel(e.target.value)} placeholder="Label" className="text-sm" />
   293→   293→                      <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} placeholder="Description" className="text-sm" />
   294→   294→                      <Input value={editRoles} onChange={e => setEditRoles(e.target.value)} placeholder="Roles (comma-sep)" className="text-sm" />
   295→   295→                    </div>
   296→   296→                    <Button size="sm" onClick={() => updateLevel(i, editLabel, editDesc, editRoles)}><Check className="h-3 w-3" /></Button>
   297→   297→                    <Button size="sm" variant="outline" onClick={() => setEditing(null)}>Cancel</Button>
   298→   298→                  </>
   299→   299→                ) : (
   300→   300→                  <>
   301→   301→                    <Badge variant="outline" className={`font-bold ${levelColors[l.level] || 'bg-muted'}`}>L{l.level}</Badge>
   302→   302→                    <div className="flex-1">
   303→   303→                      <div className="font-medium text-sm">{l.label}</div>
   304→   304→                      <div className="text-xs text-muted-foreground">{l.desc}</div>
   305→   305→                      {l.roles && l.roles.length > 0 && (
   306→   306→                        <div className="flex flex-wrap gap-1 mt-1">
   307→   307→                          {l.roles.map((r: string) => <Badge key={r} variant="outline" className="text-[10px] px-1 py-0">{r}</Badge>)}
   308→   308→                        </div>
   309→   309→                      )}
   310→   310→                    </div>
   311→   311→                    <Button size="sm" variant="ghost" className="h-7" onClick={() => { setEditing(i); setEditLabel(l.label); setEditDesc(l.desc); setEditRoles((l.roles || []).join(', ')) }}>
   312→   312→                      Edit
   313→   313→                    </Button>
   314→   314→                    {l.level > 2 && (
   315→   315→                      <Button size="sm" variant="ghost" className="h-7 text-red-600" onClick={() => deleteLevel(i)}>
   316→   316→                        <Trash2 className="h-3 w-3" />
   317→   317→                      </Button>
   318→   318→                    )}
   319→   319→                  </>
   320→   320→                )}
   321→   321→              </div>
   322→   322→            ))}
   323→   323→          </div>
   324→   324→
   325→   325→          {/* Add new level */}
   326→   326→          <div className="mt-4 pt-3 border-t">
   327→   327→            <div className="text-xs font-medium text-muted-foreground mb-2">Add New Level</div>
   328→   328→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 mb-2">
   329→   329→              <Input placeholder="Label (e.g., Volunteer)" value={newLabel} onChange={e => setNewLabel(e.target.value)} className="text-sm" />
   330→   330→              <Input placeholder="Description" value={newDesc} onChange={e => setNewDesc(e.target.value)} className="text-sm" />

   336→   336→      </Card>
   337→   337→
   338→   338→      <Card>
   339→   339→        <CardHeader>
   340→   340→          <CardTitle className="text-sm">How Levels Work</CardTitle>
   341→   341→        </CardHeader>
   342→   342→        <CardContent className="text-sm space-y-1 text-muted-foreground">
   343→   343→          <p>• Level 1 (Owner) can see and manage ALL users at any level.</p>
   344→   344→          <p>• Level 2 (Manager) can see users at Level 2 and below — cannot see Owner.</p>
   345→   345→          <p>• Level 3+ can only see users at their own level or below.</p>

   348→   348→        </CardContent>
   349→   349→      </Card>
   350→   350→
   351→   351→      <Card>
   352→   352→        <CardHeader>

   359→   359→        </CardContent>
   360→   360→      </Card>
   361→   361→    </div>

   366→   366→function ResidentStatusSettings({ settings, saveSetting }: { settings: any; saveSetting: (k: string, v: any) => void }) {
   367→   367→  const statuses = settings.residentStatuses || []
   368→   368→  const [editing, setEditing] = useState<number | null>(null)
   369→   369→  const [editLabel, setEditLabel] = useState('')
   370→   370→  const [editDesc, setEditDesc] = useState('')
   371→   371→  const [newLabel, setNewLabel] = useState('')
   372→   372→  const [newDesc, setNewDesc] = useState('')
   373→   373→
   374→   374→  const updateStatus = (index: number, label: string, desc: string) => {
   375→   375→    const updated = [...statuses]
   376→   376→    updated[index] = { ...updated[index], label, desc }
   377→   377→    saveSetting('residentStatuses', updated)
   378→   378→    setEditing(null)
   379→   379→  }
   380→   380→
   381→   381→  const addStatus = () => {
   382→   382→    if (!newLabel.trim()) { toast.error('Label required'); return }
   383→   383→    const id = newLabel.trim().toUpperCase().replace(/\s+/g, '_')
   384→   384→    if (statuses.some((s: any) => s.id === id)) { toast.error('Status already exists'); return }
   385→   385→    const updated = [...statuses, { id, label: newLabel.trim(), desc: newDesc.trim() }]
   386→   386→    saveSetting('residentStatuses', updated)
   387→   387→    setNewLabel('')
