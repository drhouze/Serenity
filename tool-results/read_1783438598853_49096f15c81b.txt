   386→// ============== PAYMENTS ==============
   387→// Standalone payment records that can be matched to invoices.
   388→// Each payment has its own code (PMT-000001) and may be applied to one
   389→// or more invoices (via PaymentApplication). Unapplied payments act as
   390→// resident/facility credit that can be allocated later.
   391→model Payment {
   392→  id              String   @id @default(cuid())
   393→  facilityId      String?
   394→  facility        Facility? @relation(fields: [facilityId], references: [id])
   395→  paymentCode     String   @unique  // PMT-000001 format
   396→  residentId      String?
   397→  resident        Resident? @relation(fields: [residentId], references: [id])
   398→  invoiceId       String?   // optional primary invoice this payment is for (legacy/quick-match)
   399→  invoice         Invoice?  @relation(fields: [invoiceId], references: [id])
   400→  payerName       String?   // who actually paid (resident, family, insurance co.)
   401→  paymentDate     DateTime @default(now())
   402→  amount          Float    // total amount of the payment received
   403→  appliedAmount   Float    @default(0)  // sum allocated to invoices (amount - appliedAmount = unapplied credit)
   404→  method          String   @default("CASH")  // CASH, BANK_TRANSFER, CHEQUE, CARD, INSURANCE, ONLINE, OTHER
   405→  reference       String?  // bank reference, cheque number, transaction id
   406→  bankAccount     String?  // which bank account received it
   407→  status          String   @default("CLEARED")  // PENDING, CLEARED, BOUNCED, REFUNDED
   408→  notes           String?
   409→  receivedBy      String?  // staff name who received the payment
   410→  createdAt       DateTime @default(now())
   411→  updatedAt       DateTime @updatedAt
   412→
   413→  applications    PaymentApplication[]
   414→  journalEntries  JournalEntry[]
   415→}
   416→
   417→// Many-to-many between Payment and Invoice (a payment can be split across
   418→// multiple invoices; an invoice can have multiple payments applied).
   419→model PaymentApplication {
   420→  id          String   @id @default(cuid())
   421→  paymentId   String
   422→  payment     Payment   @relation(fields: [paymentId], references: [id], onDelete: Cascade)
   423→  invoiceId   String
   424→  invoice     Invoice   @relation(fields: [invoiceId], references: [id], onDelete: Cascade)
   425→  amount      Float     // how much of this payment is applied to this invoice
   426→  appliedAt   DateTime @default(now())
   427→}

   555→// Each account has a code (e.g. "1000"), name, type, and normal balance side.
   556→// Accounts are grouped by type: ASSET, LIABILITY, EQUITY, REVENUE, EXPENSE.
   557→// Optional parentId allows hierarchical grouping (e.g. "Current Assets" → "Cash").
   558→model Account {
   559→  id            String   @id @default(cuid())
   560→  facilityId    String?
   561→  facility      Facility? @relation(fields: [facilityId], references: [id])
   562→  code          String                    // "1000", "4010", etc.
   563→  name          String                    // "Cash — Maybank", "Room & Board Revenue"
   564→  type          String                    // ASSET, LIABILITY, EQUITY, REVENUE, EXPENSE
   565→  subtype       String?                   // "CURRENT_ASSET", "FIXED_ASSET", "CURRENT_LIABILITY", etc. (optional)
   566→  normalBalance String   @default("DEBIT") // DEBIT or CREDIT — which side increases the account
   567→  parentId      String?                   // for hierarchy (optional)
   568→  parent        Account? @relation("AccountHierarchy", fields: [parentId], references: [id])
   569→  children      Account[] @relation("AccountHierarchy")
   570→  isGroup       Boolean  @default(false) // true = header account (no postings), false = posting account
   571→  active        Boolean  @default(true)
   572→  description   String?
   573→  createdAt     DateTime @default(now())
   574→  updatedAt     DateTime @updatedAt
   575→
   576→  journalLines  JournalLine[]
   577→  bankAccounts  BankAccount[]
   578→  budgets       Budget[]
   579→  productRevenue Product[] @relation("ProductRevenueAccount")
   580→  productExpense Product[] @relation("ProductExpenseAccount")
   581→
   582→  @@unique([facilityId, code])
   583→}

   585→// ============== JOURNAL ENTRY ==============
   586→// A journal entry is a collection of journal lines that must balance
   587→// (total debits = total credits). Each entry has a source:
   588→//   - MANUAL: created by the user via the Journal Entries UI
   589→//   - AUTO_INVOICE: auto-posted when an invoice is created
   590→//   - AUTO_EXPENSE: auto-posted when an expense is recorded
   591→//   - AUTO_PAYMENT: auto-posted when a payment is received
   592→//   - AUTO_RECURRING: posted by a recurring JE template
   593→model JournalEntry {
   594→  id            String   @id @default(cuid())
   595→  facilityId    String?
   596→  facility      Facility? @relation(fields: [facilityId], references: [id])
   597→  entryNumber   String   @unique          // JE-000001 format
   598→  entryDate     DateTime @default(now())
   599→  memo          String                    // description / narration
   600→  source        String   @default("MANUAL") // MANUAL, AUTO_INVOICE, AUTO_EXPENSE, AUTO_PAYMENT, AUTO_RECURRING
   601→  reference     String?                   // e.g. invoice number, expense description, payment code
   602→  posted        Boolean  @default(true)   // false = draft, true = posted to GL
   603→  reversedBy    String?                   // ID of reversing JE (if this entry was reversed)
   604→  reversedJEId  String?                   // ID of the JE that reversed this one
   605→  
   606→  // Links to source documents (optional — set when source is AUTO_*)
   607→  invoiceId     String?
   608→  invoice       Invoice? @relation(fields: [invoiceId], references: [id])
   609→  expenseId     String?
   610→  expense       Expense? @relation(fields: [expenseId], references: [id])
   611→  paymentId     String?
   612→  payment       Payment? @relation(fields: [paymentId], references: [id])
   613→
   614→  createdById   String?
   615→  createdByName String?
   616→  createdAt     DateTime @default(now())
   617→  updatedAt     DateTime @updatedAt
   618→
   619→  lines         JournalLine[]
   620→}

   622→// ============== JOURNAL LINE ==============
   623→// A single debit or credit line within a journal entry.
   624→// Either debit or credit will be non-zero (never both).
   625→model JournalLine {
   626→  id              String   @id @default(cuid())
   627→  journalEntryId  String
   628→  journalEntry    JournalEntry @relation(fields: [journalEntryId], references: [id], onDelete: Cascade)
   629→  accountId       String
   630→  account         Account     @relation(fields: [accountId], references: [id])
   631→  debit           Float    @default(0)   // amount debited (credit will be 0)
   632→  credit          Float    @default(0)   // amount credited (debit will be 0)
   633→  description     String?                  // line-level memo (optional)
   634→  
   635→  // Optional dimension links for drill-down
   636→  residentId      String?
   637→  resident        Resident? @relation(fields: [residentId], references: [id])
   638→  
   639→  createdAt       DateTime @default(now())
   640→}

   686→  updatedAt     DateTime @updatedAt
