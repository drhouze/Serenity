    16→async function apiPost(url: string, body: any) {
    17→  const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
    18→  if (!r.ok) throw new Error((await r.json().catch(() => ({}))).error || `HTTP ${r.status}`)
    19→  return r.json()
    20→}

    22→export function SettingsModule() {
    23→  const [activeSection, setActiveSection] = useState<string>('medications')
    24→  const [settings, setSettings] = useState<any>(null)
    25→
    26→  const loadSettings = useCallback(async () => {
    27→    try {
    28→      const r = await fetch('/api/settings')
    29→      const data = await r.json()
    30→      setSettings(data)
    31→    } catch (e: any) {
    32→      toast.error('Failed to load settings')
    33→    }
    34→  }, [])
    35→
    36→  useEffect(() => { loadSettings() }, [loadSettings])
    37→
    38→  const saveSetting = async (key: string, value: any) => {
    39→    try {
    40→      await apiPost('/api/settings', { key, value })
    41→      toast.success('Setting saved')
    42→      setSettings({ ...settings, [key]: value })
    43→    } catch (e: any) {
    44→      toast.error(e.message)
    45→    }
    46→  }
    47→
    48→  const sections = [
    49→    { id: 'medications', label: 'Medications', icon: Pill },
    50→    { id: 'users', label: 'Users & Levels', icon: Users },
    51→    { id: 'residents', label: 'Resident Statuses', icon: Stethoscope },
    52→    { id: 'facility', label: 'Facility Info', icon: Building2 },
    53→    { id: 'billing', label: 'Billing', icon: DollarSign },
    54→  ]
    55→
    56→  if (!settings) return <div className="space-y-2">{Array.from({ length: 3 }).map((_, i) => <div key={i} className="h-32 bg-muted animate-pulse rounded" />)}</div>
    57→
    58→  return (
    59→    <div className="space-y-4">
    60→      <div className="flex gap-1 overflow-x-auto border-b pb-px">
    61→        {sections.map(s => (
    62→          <button
    63→            key={s.id}
    64→            onClick={() => setActiveSection(s.id)}
    65→            className={`flex items-center gap-1.5 px-3 py-2 text-sm whitespace-nowrap border-b-2 transition-colors ${
    66→              activeSection === s.id ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground hover:text-foreground'
    67→            }`}
    68→          >
    69→            <s.icon className="h-3.5 w-3.5" /> {s.label}
    70→          </button>
    71→        ))}
    72→      </div>
    73→
    74→      {activeSection === 'medications' && <MedicationSettings settings={settings} saveSetting={saveSetting} />}
    75→      {activeSection === 'users' && <UserLevelSettings settings={settings} saveSetting={saveSetting} />}
    76→      {activeSection === 'residents' && <ResidentStatusSettings settings={settings} saveSetting={saveSetting} />}
    77→      {activeSection === 'facility' && <FacilitySettings settings={settings} saveSetting={saveSetting} />}
    78→      {activeSection === 'billing' && <BillingSettings settings={settings} saveSetting={saveSetting} />}
    79→    </div>
    80→  )
    81→}

   228→// ============ USER LEVEL SETTINGS ============
   229→function UserLevelSettings({ settings, saveSetting }: { settings: any; saveSetting: (k: string, v: any) => void }) {
   230→  const levels: any[] = settings.userLevels || []
   231→  const [editing, setEditing] = useState<number | null>(null)
   232→  const [editLabel, setEditLabel] = useState('')
   233→  const [editDesc, setEditDesc] = useState('')
   234→  const [editRoles, setEditRoles] = useState('')
   235→  const [newLabel, setNewLabel] = useState('')
   236→  const [newDesc, setNewDesc] = useState('')
   237→  const [newRoles, setNewRoles] = useState('')
   238→
   239→  const levelColors: Record<number, string> = {
   240→    1: 'bg-rose-100 text-rose-700',
   241→    2: 'bg-orange-100 text-orange-700',
   242→    3: 'bg-sky-100 text-sky-700',
   243→    4: 'bg-emerald-100 text-emerald-700',
   244→    5: 'bg-violet-100 text-violet-700',
   245→    6: 'bg-amber-100 text-amber-700',
   246→    7: 'bg-teal-100 text-teal-700',
   247→    8: 'bg-pink-100 text-pink-700',
   248→  }
   249→
   250→  const updateLevel = (index: number, label: string, desc: string, rolesStr: string) => {
   251→    const updated = [...levels]
   252→    const roles = rolesStr.split(',').map((r: string) => r.trim().toUpperCase()).filter(Boolean)
   253→    updated[index] = { ...updated[index], label, desc, roles }
   254→    saveSetting('userLevels', updated)
   255→    setEditing(null)
   256→  }
   257→
   258→  const addLevel = () => {
   259→    if (!newLabel.trim()) { toast.error('Label required'); return }
   260→    const nextLevel = Math.max(...levels.map((l: any) => l.level), 0) + 1
   261→    const roles = newRoles.split(',').map(r => r.trim().toUpperCase()).filter(Boolean)
   262→    const updated = [...levels, { level: nextLevel, label: newLabel.trim(), desc: newDesc.trim(), roles }]
   263→    saveSetting('userLevels', updated)
   264→    setNewLabel(''); setNewDesc(''); setNewRoles('')
   265→    toast.success(`Level ${nextLevel} added`)
   266→  }
   267→
   268→  const deleteLevel = (index: number) => {
   269→    const level = levels[index]
   270→    if (level.level <= 2) { toast.error('Cannot delete Level 1 (Owner) or Level 2 (Manager)'); return }
   271→    if (!confirm(`Delete Level ${level.level} (${level.label})? Users at this level will need to be reassigned.`)) return
   272→    const updated = levels.filter((_: any, i: number) => i !== index)
   273→    saveSetting('userLevels', updated)
   274→    toast.success('Level deleted')
   275→  }
   276→
   277→  return (
   278→    <div className="space-y-4">
   279→      <Card>
   280→        <CardHeader>
   281→          <CardTitle className="text-sm flex items-center gap-2"><Users className="h-4 w-4" /> User Hierarchy Levels</CardTitle>
   282→          <CardDescription>Edit, add, or remove user levels. Level 1 (Owner) is highest — can see all. Lower levels can only see users at their level or below.</CardDescription>
   283→        </CardHeader>
   284→        <CardContent>
   285→          <div className="space-y-2">
   286→            {levels.map((l: any, i: number) => (
   287→              <div key={l.level} className="flex items-center gap-3 p-3 border rounded-md">
   288→                {editing === i ? (
   289→                  <>
   290→                    <Badge variant="outline" className={`font-bold ${levelColors[l.level] || 'bg-muted'}`}>L{l.level}</Badge>
   291→                    <div className="flex-1 grid grid-cols-1 sm:grid-cols-3 gap-2">
   292→                      <Input value={editLabel} onChange={e => setEditLabel(e.target.value)} placeholder="Label" className="text-sm" />
   293→                      <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} placeholder="Description" className="text-sm" />
   294→                      <Input value={editRoles} onChange={e => setEditRoles(e.target.value)} placeholder="Roles (comma-sep)" className="text-sm" />
   295→                    </div>
   296→                    <Button size="sm" onClick={() => updateLevel(i, editLabel, editDesc, editRoles)}><Check className="h-3 w-3" /></Button>
   297→                    <Button size="sm" variant="outline" onClick={() => setEditing(null)}>Cancel</Button>
   298→                  </>
   299→                ) : (
   300→                  <>
   301→                    <Badge variant="outline" className={`font-bold ${levelColors[l.level] || 'bg-muted'}`}>L{l.level}</Badge>
   302→                    <div className="flex-1">
   303→                      <div className="font-medium text-sm">{l.label}</div>
   304→                      <div className="text-xs text-muted-foreground">{l.desc}</div>
   305→                      {l.roles && l.roles.length > 0 && (
   306→                        <div className="flex flex-wrap gap-1 mt-1">
   307→                          {l.roles.map((r: string) => <Badge key={r} variant="outline" className="text-[10px] px-1 py-0">{r}</Badge>)}
   308→                        </div>
   309→                      )}
   310→                    </div>
   311→                    <Button size="sm" variant="ghost" className="h-7" onClick={() => { setEditing(i); setEditLabel(l.label); setEditDesc(l.desc); setEditRoles((l.roles || []).join(', ')) }}>
   312→                      Edit
   313→                    </Button>
   314→                    {l.level > 2 && (
   315→                      <Button size="sm" variant="ghost" className="h-7 text-red-600" onClick={() => deleteLevel(i)}>
   316→                        <Trash2 className="h-3 w-3" />
   317→                      </Button>
   318→                    )}
   319→                  </>
   320→                )}
   321→              </div>
   322→            ))}
   323→          </div>
   324→
   325→          {/* Add new level */}
   326→          <div className="mt-4 pt-3 border-t">
   327→            <div className="text-xs font-medium text-muted-foreground mb-2">Add New Level</div>
   328→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 mb-2">
   329→              <Input placeholder="Label (e.g., Volunteer)" value={newLabel} onChange={e => setNewLabel(e.target.value)} className="text-sm" />
   330→              <Input placeholder="Description" value={newDesc} onChange={e => setNewDesc(e.target.value)} className="text-sm" />
   331→              <Input placeholder="Roles (e.g., VOLUNTEER, INTERN)" value={newRoles} onChange={e => setNewRoles(e.target.value)} className="text-sm" />
   332→            </div>
   333→            <Button size="sm" onClick={addLevel}><Plus className="h-3 w-3 mr-1" /> Add Level</Button>
   334→          </div>
   335→        </CardContent>
   336→      </Card>
   337→
   338→      <Card>
   339→        <CardHeader>
   340→          <CardTitle className="text-sm">How Levels Work</CardTitle>
   341→        </CardHeader>
   342→        <CardContent className="text-sm space-y-1 text-muted-foreground">
   343→          <p>• Level 1 (Owner) can see and manage ALL users at any level.</p>
   344→          <p>• Level 2 (Manager) can see users at Level 2 and below — cannot see Owner.</p>
   345→          <p>• Level 3+ can only see users at their own level or below.</p>
   346→          <p>• When creating a user, you cannot assign a level higher than your own.</p>
   347→          <p className="mt-2 text-amber-600">⚠ Levels 1 and 2 cannot be deleted. They are essential for system operation.</p>
   348→        </CardContent>
   349→      </Card>
   350→
   351→      <Card>
   352→        <CardHeader>
   353→          <CardTitle className="text-sm">Module Access</CardTitle>
   354→          <CardDescription>Each user's module access can be customized individually via User Accounts → Modules button</CardDescription>
   355→        </CardHeader>
   356→        <CardContent className="text-sm text-muted-foreground">
   357→          <p>The Owner can override any user's module access at any time, regardless of their role or level.</p>
   358→          <p className="mt-1">Default module access is based on role, but can be fully customized per user.</p>
   359→        </CardContent>
   360→      </Card>
   361→    </div>
   362→  )
   363→}

   365→// ============ RESIDENT STATUS SETTINGS ============
   366→function ResidentStatusSettings({ settings, saveSetting }: { settings: any; saveSetting: (k: string, v: any) => void }) {
   367→  const statuses = settings.residentStatuses || []
   368→  const [editing, setEditing] = useState<number | null>(null)
   369→  const [editLabel, setEditLabel] = useState('')
   370→  const [editDesc, setEditDesc] = useState('')
   371→  const [newLabel, setNewLabel] = useState('')
   372→  const [newDesc, setNewDesc] = useState('')
   373→
   374→  const updateStatus = (index: number, label: string, desc: string) => {
   375→    const updated = [...statuses]
   376→    updated[index] = { ...updated[index], label, desc }
   377→    saveSetting('residentStatuses', updated)
   378→    setEditing(null)
   379→  }
   380→
   381→  const addStatus = () => {
   382→    if (!newLabel.trim()) { toast.error('Label required'); return }
   383→    const id = newLabel.trim().toUpperCase().replace(/\s+/g, '_')
   384→    if (statuses.some((s: any) => s.id === id)) { toast.error('Status already exists'); return }
   385→    const updated = [...statuses, { id, label: newLabel.trim(), desc: newDesc.trim() }]
   386→    saveSetting('residentStatuses', updated)
   387→    setNewLabel('')
   388→    setNewDesc('')
   389→    toast.success('Status added')
   390→  }
   391→
   392→  const deleteStatus = (index: number) => {
   393→    if (!confirm('Delete this status? Existing records with this status will not be affected.')) return
   394→    const updated = statuses.filter((_: any, i: number) => i !== index)
   395→    saveSetting('residentStatuses', updated)
   396→    toast.success('Status deleted')
   397→  }
   398→
   399→  return (
   400→    <div className="space-y-4">
   401→      <Card>
   402→        <CardHeader>
   403→          <CardTitle className="text-sm flex items-center gap-2"><Stethoscope className="h-4 w-4" /> Resident Statuses</CardTitle>
   404→          <CardDescription>Edit, add, or remove resident statuses. Changing status affects medication scheduling.</CardDescription>
   405→        </CardHeader>
   406→        <CardContent>
   407→          <div className="space-y-2">
   408→            {statuses.map((s: any, i: number) => (
   409→              <div key={s.id} className="flex items-center gap-3 p-3 border rounded-md">
   410→                {editing === i ? (
   411→                  <>
   412→                    <Input value={editLabel} onChange={e => setEditLabel(e.target.value)} className="flex-1 text-sm" />
   413→                    <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} className="flex-[2] text-sm" placeholder="Description" />
   414→                    <Button size="sm" onClick={() => updateStatus(i, editLabel, editDesc)}><Check className="h-3 w-3" /></Button>
   415→                    <Button size="sm" variant="outline" onClick={() => setEditing(null)}>Cancel</Button>
   416→                  </>
   417→                ) : (
   418→                  <>
   419→                    <Badge variant="outline" className="bg-primary/5">{s.label}</Badge>
   420→                    <span className="text-sm text-muted-foreground flex-1">{s.desc}</span>
   421→                    <Button size="sm" variant="ghost" className="h-7" onClick={() => { setEditing(i); setEditLabel(s.label); setEditDesc(s.desc) }}>
   422→                      Edit
   423→                    </Button>
   424→                    <Button size="sm" variant="ghost" className="h-7 text-red-600" onClick={() => deleteStatus(i)}>
   425→                      <Trash2 className="h-3 w-3" />
   426→                    </Button>
   427→                  </>
   428→                )}
   429→              </div>
   430→            ))}
   431→          </div>
   432→
   433→          {/* Add new status */}
   434→          <div className="mt-4 pt-3 border-t">
   435→            <div className="text-xs font-medium text-muted-foreground mb-2">Add New Status</div>
   436→            <div className="flex gap-2">
   437→              <Input placeholder="Status label (e.g., On Leave)" value={newLabel} onChange={e => setNewLabel(e.target.value)} className="flex-1 text-sm" />
   438→              <Input placeholder="Description" value={newDesc} onChange={e => setNewDesc(e.target.value)} className="flex-[2] text-sm" />
   439→              <Button size="sm" onClick={addStatus}><Plus className="h-3 w-3 mr-1" /> Add</Button>
   440→            </div>
   441→          </div>
   442→        </CardContent>
   443→      </Card>
   444→
   445→      <Card>
   446→        <CardHeader>
   447→          <CardTitle className="text-sm flex items-center gap-2"><AlertTriangle className="h-4 w-4" /> Status Change Effects</CardTitle>
   448→        </CardHeader>
   449→        <CardContent className="text-sm space-y-2">

   468→// ============ FACILITY SETTINGS ============
   469→function FacilitySettings({ settings, saveSetting }: { settings: any; saveSetting: (k: string, v: any) => void }) {
   470→  const { data: facilities, loading, refetch } = useFetch<any[]>('/api/facilities')
   471→  const [editing, setEditing] = useState<any | null>(null)
   472→  const [showAdd, setShowAdd] = useState(false)
   473→  const [refreshKey, setRefreshKey] = useState(0)
   474→  const triggerRefresh = () => { setRefreshKey(k => k + 1); refetch() }
   475→
   476→  if (loading && !facilities) return <Skeleton className="h-96" />
   477→
   478→  const handleSave = async (facility: any) => {
   479→    try {
   480→      if (facility.id) {
   481→        await fetch(`/api/facilities?id=${facility.id}`, {
   482→          method: 'PATCH', headers: { 'Content-Type': 'application/json' },
   483→          body: JSON.stringify(facility),
   484→        })
   485→        toast.success('Facility updated')
   486→      } else {
   487→        await fetch('/api/facilities', {
   488→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   489→          body: JSON.stringify(facility),
   490→        })
   491→        toast.success('Facility added')
   492→      }
   493→      setEditing(null)
   494→      setShowAdd(false)
   495→      triggerRefresh()
   496→    } catch (e: any) { toast.error(e.message) }
   497→  }
   498→
   499→  const handleDelete = async (id: string) => {
   500→    if (!confirm('Delete this facility? All data must be reassigned first.')) return
   501→    try {
   502→      const r = await fetch(`/api/facilities?id=${id}`, { method: 'DELETE' })
   503→      const d = await r.json()
   504→      if (!r.ok) { toast.error(d.error); return }
   505→      toast.success('Facility deleted')
   506→      triggerRefresh()
   507→    } catch (e: any) { toast.error(e.message) }
   508→  }
   509→
   510→  return (
   511→    <div className="space-y-4">
   512→      {/* Facility List */}
   513→      <Card>
   514→        <CardHeader>
   515→          <CardTitle className="text-sm flex items-center gap-2"><Building2 className="h-4 w-4" /> Facilities</CardTitle>
   516→          <CardDescription>Manage your nursing home locations. Each facility has its own residents, rooms, staff, and financial data.</CardDescription>
   517→        </CardHeader>
   518→        <CardContent className="p-0">
   519→          <div className="overflow-x-auto">
   520→            <table className="w-full text-sm">
   521→              <thead className="bg-muted/50">
   522→                <tr>
   523→                  <th className="text-left p-2 font-medium">Name</th>
   524→                  <th className="text-left p-2 font-medium hidden md:table-cell">Address</th>
   525→                  <th className="text-left p-2 font-medium hidden lg:table-cell">Director</th>
   526→                  <th className="text-center p-2 font-medium">Residents</th>
   527→                  <th className="text-center p-2 font-medium">Staff</th>
   528→                  <th className="text-center p-2 font-medium">Rooms</th>
   529→                  <th className="text-left p-2 font-medium">Status</th>
   530→                  <th className="p-2"></th>
   531→                </tr>
   532→              </thead>
   533→              <tbody>
   534→                {(facilities || []).map(f => (
   535→                  <tr key={f.id} className="border-t hover:bg-muted/30">
   536→                    <td className="p-2 font-medium">{f.name}</td>
   537→                    <td className="p-2 text-xs hidden md:table-cell">{f.address || '—'}</td>
   538→                    <td className="p-2 text-xs hidden lg:table-cell">{f.director || '—'}</td>
   539→                    <td className="p-2 text-center">{f._count?.residents || 0}</td>
   540→                    <td className="p-2 text-center">{f._count?.staff || 0}</td>
   541→                    <td className="p-2 text-center">{f._count?.rooms || 0}</td>
   542→                    <td className="p-2">
   543→                      {f.active ? <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 text-xs">Active</Badge> : <Badge variant="secondary" className="text-xs">Inactive</Badge>}
   544→                    </td>
   545→                    <td className="p-2 whitespace-nowrap">
   546→                      <Button size="sm" variant="ghost" className="h-7" onClick={() => setEditing(f)}><Edit className="h-3 w-3" /></Button>
   547→                      <Button size="sm" variant="ghost" className="h-7 text-red-600" onClick={() => handleDelete(f.id)}><Trash2 className="h-3 w-3" /></Button>
   548→                    </td>
   549→                  </tr>
   550→                ))}
   551→              </tbody>
   552→            </table>
   553→          </div>
   554→          {(facilities || []).length === 0 && <p className="text-sm text-muted-foreground text-center py-8">No facilities yet. Click "Add Facility" to create one.</p>}
   555→        </CardContent>
   556→      </Card>
   557→
   558→      <Button size="sm" onClick={() => setShowAdd(true)}><Plus className="h-4 w-4 mr-1" /> Add Facility</Button>
   559→
   560→      {/* Multi-facility info */}
   561→      <Card>
   562→        <CardHeader>
   563→          <CardTitle className="text-sm">How Multi-Facility Works</CardTitle>
   564→        </CardHeader>
   565→        <CardContent className="text-sm space-y-1 text-muted-foreground">
   566→          <p>• Each facility has its own residents, rooms, staff, inventory, and financial data.</p>
   567→          <p>• Owner (Level 1) can access ALL facilities.</p>
   568→          <p>• Staff and managers can be assigned to specific facilities via User Accounts.</p>
   569→          <p>• When adding residents, rooms, or staff, they are assigned to the currently selected facility.</p>
   570→          <p className="mt-2 text-amber-600">⚠ Facility switching UI (dropdown in header) is coming soon. The database schema fully supports multi-facility.</p>
   571→        </CardContent>
   572→      </Card>
   573→
   574→      {/* Edit/Add dialog */}
   575→      {(editing || showAdd) && (
   576→        <FacilityDialog
   577→          facility={editing}
   578→          onClose={() => { setEditing(null); setShowAdd(false) }}
   579→          onSave={handleSave}
   580→        />
   581→      )}
   582→    </div>

   586→function FacilityDialog({ facility, onClose, onSave }: { facility: any | null; onClose: () => void; onSave: (f: any) => void }) {
