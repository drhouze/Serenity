     8→// Helper: marks a stock transfer as RECEIVED. For each line:
     9→//   1. Find (or auto-create) a matching destination InventoryItem by name + unit
    10→//      within the destination facility.
    11→//   2. Create a TRANSFER_IN InventoryTransaction on the destination item.
    12→//   3. Increment destination item's currentStock + update unitCost (to source's cost).
    13→//   4. Stamp the line's receivedQty + destinationItemId.
    14→//   5. Update transfer.status = 'RECEIVED', receivedDate = now.
    15→//
    16→// Source stock was already decremented when the transfer moved to IN_TRANSIT
    17→// (or when RECEIVED was set at creation — see POST handler). This helper only
    18→// handles the "receive" half of the move.
    19→async function receiveStockTransfer(transfer: any, currentUser: any) {
    20→  const toFacility = await db.facility.findUnique({ where: { id: transfer.toFacilityId }, select: { name: true } })
    21→  const fromFacility = await db.facility.findUnique({ where: { id: transfer.fromFacilityId }, select: { name: true } })
    22→  for (const line of transfer.lines) {
    23→    // Find matching destination item by name + unit (case-insensitive) within the dest facility
    24→    let destItem = await db.inventoryItem.findFirst({
    25→      where: {
    26→        facilityId: transfer.toFacilityId,
    27→        name: { equals: line.itemName },
    28→        // match by unit too — same name but different unit is treated as different item
    29→      },
    30→    })
    31→    // If not found, auto-create it
    32→    if (!destItem) {
    33→      const destCode = await generateInventoryCode(transfer.toFacilityId)
    34→      destItem = await db.inventoryItem.create({
    35→        data: {
    36→          code: destCode,
    37→          facilityId: transfer.toFacilityId,
    38→          name: line.itemName,
    39→          sku: line.itemSku || null,
    40→          category: line.itemCategory || 'OTHER',
    41→          unit: line.itemUnit || 'each',
    42→          currentStock: 0,  // will be incremented below
    43→          reorderLevel: 10,
    44→          reorderQty: 50,
    45→          unitCost: line.itemUnitCost || 0,
    46→          supplier: `Transferred from ${fromFacility?.name || 'another facility'}`,
    47→          active: true,
    48→          lastCountDate: new Date(),
    49→        },
    50→      })
    51→    }
    52→    // TRANSFER_IN transaction on destination item
    53→    await db.inventoryTransaction.create({
    54→      data: {
    55→        itemId: destItem.id,
    56→        type: 'TRANSFER_IN',
    57→        quantity: line.quantity,  // positive for in
    58→        reason: `Transfer ${transfer.transferNumber} ← ${fromFacility?.name || 'source facility'}`,
    59→        date: new Date(),
    60→        recordedBy: currentUser.name,
    61→        stockTransferId: transfer.id,
    62→      },
    63→    })
    64→    // Increment destination stock + update unitCost
    65→    await db.inventoryItem.update({
    66→      where: { id: destItem.id },
    67→      data: {
    68→        currentStock: { increment: line.quantity },
    69→        lastCountDate: new Date(),
    70→        unitCost: line.itemUnitCost > 0 ? line.itemUnitCost : undefined,
    71→      },
    72→    })
    73→    // Stamp the line
    74→    await db.stockTransferLine.update({
    75→      where: { id: line.id },
    76→      data: { receivedQty: line.quantity, destinationItemId: destItem.id },
    77→    })
    78→  }
    79→  await db.stockTransfer.update({
    80→    where: { id: transfer.id },
    81→    data: { status: 'RECEIVED', receivedDate: new Date() },
    82→  })
    83→}

   158→    return undefined

   175→export async function GET(req: NextRequest) {
   176→  const { searchParams } = new URL(req.url)
   177→  const type = searchParams.get('type') || ''
   178→  const id = searchParams.get('id')
   179→  const residentId = searchParams.get('residentId')
   180→  const upcoming = searchParams.get('upcoming') === 'true'
   181→  const unbilled = searchParams.get('unbilled') === 'true'
   182→  const date = searchParams.get('date')
   183→  const facilityId = searchParams.get('facilityId') // selected facility from frontend
   184→
   185→  // Auth check — all data access requires login
   186→  const currentUser = await getSessionUser(req)
   187→  if (!currentUser) {
   188→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   189→  }
   190→
   191→  // Determine which facility IDs the user can access.
   192→  // Hierarchy:
   193→  //   - APP_DEVELOPER (L0): sees ALL facilities across ALL organizations (no scope)
   194→  //   - OWNER (L1): sees only facilities in their OWN organization
   195→  //   - MANAGER (L2) and below: sees only their assigned facilities
   196→  let accessibleFacilityIds: string[] = []
   197→  if (currentUser.role === 'APP_DEVELOPER') {
   198→    // Developer: no scope — sees everything
   199→    if (facilityId) accessibleFacilityIds = [facilityId]
   200→  } else if (currentUser.level === 1) {
   201→    // Owner: scoped to their organization's facilities
   202→    if (!currentUser.organizationId) {
   203→      // Owner without org → sees nothing (data isolation safeguard)
   204→      accessibleFacilityIds = ['__NO_ORG__'] // impossible ID → returns no rows
   205→    } else {
   206→      const orgFacilities = await db.facility.findMany({
   207→        where: { organizationId: currentUser.organizationId },
   208→        select: { id: true },
   209→      })
   210→      const orgFidSet = orgFacilities.map(f => f.id)
   211→      if (facilityId && orgFidSet.includes(facilityId)) {
   212→        accessibleFacilityIds = [facilityId]
   213→      } else {
   214→        accessibleFacilityIds = orgFidSet
   215→      }
   216→    }
   217→  } else {
   218→    // Non-owner: only their assigned facilities
   219→    const userFacilityIds = (currentUser.facilityIds || '').split(',').map(s => s.trim()).filter(Boolean)
   220→    if (facilityId && userFacilityIds.includes(facilityId)) {
   221→      accessibleFacilityIds = [facilityId]
   222→    } else {
   223→      accessibleFacilityIds = userFacilityIds
   224→    }
   225→  }
   226→
   227→  // Helper: build facility filters for queries
   228→  // Different models need different filter strategies based on how they relate to Facility:
   229→  //   - Direct facilityId          → residents, rooms, staff, expenses, products, inventory, invoices
   230→  //   - Via resident.facilityId    → medications, medAdmins, vitals, visits, incidents, careLogs, messages, invoiceItems
   231→  //   - Via staff.facilityId       → shifts
   232→  //   - Via item.facilityId        → inventoryTransactions
   233→  //   - No facility link           → auditLogs (system-wide), users (filter via facilityIds string match)
   234→  //
   235→  const hasFacilityScope = accessibleFacilityIds.length > 0
   236→  const facilityFilter = hasFacilityScope ? { facilityId: { in: accessibleFacilityIds } } : {}
   237→  const residentFacilityFilter = hasFacilityScope ? { resident: { facilityId: { in: accessibleFacilityIds } } } : {}
   238→  const staffFacilityFilter = hasFacilityScope ? { staff: { facilityId: { in: accessibleFacilityIds } } } : {}
   239→  const itemFacilityFilter = hasFacilityScope ? { item: { facilityId: { in: accessibleFacilityIds } } } : {}
   240→  // For users: facilityIds is a comma-separated string. Filter where the comma-separated list contains any of the accessible facility IDs.
   241→  // SQLite doesn't have native array operations, so we use contains for each ID. This is approximate but works for Owner's single-facility view.
   242→  const userFacilityFilter = hasFacilityScope
   243→    ? { OR: accessibleFacilityIds.flatMap(fid => [
   244→        { facilityIds: { contains: fid } },
   245→        { level: 1, role: 'OWNER' as any }, // Owner always visible (Owner sees all facilities)
   246→      ]) }
   247→    : {}
   248→
   249→  // For FAMILY role, restrict to linked residents only
   250→  const linkedResidentIds: string[] = currentUser.role === 'FAMILY'
   251→    ? (currentUser.linkedResidentIds || '').split(',').map(s => s.trim()).filter(Boolean)
   252→    : []
   253→
   254→  try {
   255→    // FAMILY role: block access to sensitive data types that they should never see directly.
   256→    // NOTE: incidents, careLogs, and visits are ALLOWED for family users — they are scoped
   257→    // to their linked residents via the `linkedResidentIds` filter in each case handler.
   258→    if (currentUser.role === 'FAMILY') {
   259→      const blockedTypes = ['rooms', 'staff', 'medications', 'medAdmins', 'invoices', 'invoiceItems', 'expenses', 'payments', 'paymentApplications', 'deposits', 'accounts', 'journalEntries', 'vendors', 'bankAccounts', 'shifts', 'users', 'products', 'inventory', 'inventoryTransactions', 'finance']
   260→      if (blockedTypes.includes(type)) {
   261→        return NextResponse.json({ error: 'Access denied for your role' }, { status: 403 })
   262→      }
   263→      // incidents, careLogs, visits, messages are allowed — each handler restricts to linkedResidentIds
   264→    }
   265→
   266→    switch (type) {
   267→      case 'residents': {
   268→        // FAMILY users can only see their linked residents
   269→        if (currentUser.role === 'FAMILY') {
   270→          if (id) {
   271→            // Verify this resident is linked to this family user
   272→            if (!linkedResidentIds.includes(id)) {
   273→              return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   274→            }
   275→          }
   276→          // Include the same relations as the staff query so the family user sees
   277→          // incidents, care logs, visits, vitals, and med administrations in the resident detail view.
   278→          // (Billing/financial relations like invoiceItems are intentionally excluded for family.)
   279→          const list = await db.resident.findMany({
   280→            where: { id: { in: linkedResidentIds }, status: 'ACTIVE' },
   281→            include: {
   282→              room: true,
   283→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   284→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   285→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   286→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   287→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   288→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   289→            },
   290→            orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   291→          })
   292→          if (id) {
   293→            const r = list.find(x => x.id === id)
   294→            if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   295→            return NextResponse.json(r)
   296→          }
   297→          return NextResponse.json(list)
   298→        }
   299→
   300→        if (id) {
   301→          const r = await db.resident.findUnique({
   302→            where: { id },
   303→            include: {
   304→              room: true,
   305→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   306→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   307→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   308→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   309→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   310→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   311→              invoiceItems: { where: { billed: false }, orderBy: { createdAt: 'desc' } },
   312→            },
   313→          })
   314→          if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   315→          return NextResponse.json(r)
   316→        }
   317→        const includeArchived = searchParams.get('includeArchived') === 'true'
   318→        const statusFilter = searchParams.get('status') // ACTIVE, DISCHARGED, etc.
   319→        const where: any = { ...facilityFilter }
   320→        if (statusFilter) {
   321→          where.status = statusFilter
   322→        } else if (!includeArchived) {
   323→          where.status = 'ACTIVE'
   324→        }
   325→        const list = await db.resident.findMany({
   326→          where,
   327→          include: { room: true },
   328→          orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   329→        })
   330→        return NextResponse.json(list)
   331→      }
   332→
   333→      case 'rooms': {
   334→        const rooms = await db.room.findMany({
   335→          include: { residents: { where: { status: 'ACTIVE' } } },
   336→          orderBy: { roomNumber: 'asc' },
   337→        })
   338→        return NextResponse.json(rooms)
   339→      }
   340→
   341→      case 'staff': {
   342→        const includeArchived = searchParams.get('includeArchived') === 'true'
   343→        // Multi-facility match: staff may be assigned via facilityId (primary) OR via the
   344→        // comma-separated facilityIds field (multi-facility assignment).
   345→        const where: any = hasFacilityScope
   346→          ? { OR: [
   347→              { facilityId: { in: accessibleFacilityIds } },
   348→              ...accessibleFacilityIds.flatMap(fid => [{ facilityIds: { contains: fid } }]),
   349→            ] }
   350→          : {}
   351→        if (!includeArchived) where.active = true
   352→        const staff = await db.staff.findMany({
   353→          where,
   354→          include: { _count: { select: { shifts: true, visits: true } } },
   355→          orderBy: [{ role: 'asc' }, { lastName: 'asc' }],
   356→        })
   357→        return NextResponse.json(staff)
   358→      }
   359→
   360→      case 'medications': {
   361→        const where: any = { ...residentFacilityFilter }
   362→        if (residentId) where.residentId = residentId
   363→        if (searchParams.get('active') !== 'false') where.active = true
   364→        const meds = await db.medication.findMany({
   365→          where,
   366→          include: { resident: { include: { room: true } } },
   367→          orderBy: { name: 'asc' },
   368→        })
   369→        return NextResponse.json(meds)
   370→      }
   371→
   372→      case 'medAdmins': {
   373→        const where: any = { ...residentFacilityFilter }
   374→        if (residentId) where.residentId = residentId
   375→        if (searchParams.get('today') === 'true') {
   376→          const start = new Date(); start.setHours(0, 0, 0, 0)
   377→          const end = new Date(); end.setHours(23, 59, 59, 999)
   378→          where.scheduledAt = { gte: start, lte: end }
   379→        } else {
   380→          // Generic date-range filter (yyyy-MM-dd, inclusive on both ends).
   381→          // Server-side; reduces payload when users pick a preset like Last 7 / This month.
   382→          const startDate = searchParams.get('startDate')
   383→          const endDate = searchParams.get('endDate')
   384→          if (startDate || endDate) {
   385→            where.scheduledAt = {}
   386→            if (startDate) {
   387→              const s = new Date(startDate); s.setHours(0, 0, 0, 0)
   388→              where.scheduledAt.gte = s
   389→            }
   390→            if (endDate) {
   391→              const e = new Date(endDate); e.setHours(23, 59, 59, 999)
   392→              where.scheduledAt.lte = e
   393→            }
   394→          }
   395→        }
   396→        const status = searchParams.get('status')
   397→        if (status) where.status = status
   398→        const admins = await db.medAdministration.findMany({
   399→          where,
   400→          include: { medication: true, resident: { include: { room: true } }, staff: true },
   401→          orderBy: { scheduledAt: 'desc' },
   402→          take: 500,
   403→        })
   404→        return NextResponse.json(admins)
   405→      }
   406→
   407→      case 'vitals': {
   408→        const where: any = { ...residentFacilityFilter }
   409→        if (residentId) where.residentId = residentId
   410→        // FAMILY users: restrict to their linked residents only
   411→        if (currentUser.role === 'FAMILY') {
   412→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   413→          where.residentId = { in: linkedResidentIds }
   414→        }
   415→        // Date-range filter (yyyy-MM-dd, inclusive on both ends)
   416→        const startDate = searchParams.get('startDate')
   417→        const endDate = searchParams.get('endDate')
   418→        if (startDate || endDate) {
   419→          where.recordedAt = {}
   420→          if (startDate) {
   421→            const s = new Date(startDate); s.setHours(0, 0, 0, 0)
   422→            where.recordedAt.gte = s
   423→          }
   424→          if (endDate) {
   425→            const e = new Date(endDate); e.setHours(23, 59, 59, 999)
   426→            where.recordedAt.lte = e
   427→          }
   428→        }
   429→        const vitals = await db.vitalSign.findMany({
   430→          where,
   431→          include: { resident: { include: { room: true } } },
   432→          orderBy: { recordedAt: 'desc' },
   433→          take: residentId ? 200 : 500,
   434→        })
   435→        return NextResponse.json(vitals)
   436→      }
   437→
   438→      case 'visits': {
   439→        const where: any = { ...residentFacilityFilter }
   440→        if (residentId) where.residentId = residentId
   441→        // FAMILY users can only see visits for their linked residents
   442→        if (currentUser.role === 'FAMILY') {
   443→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   444→          where.residentId = { in: linkedResidentIds }
   445→        }
   446→        if (upcoming) {
   447→          where.scheduledAt = { gte: new Date() }
   448→          where.status = 'SCHEDULED'
   449→        } else {
   450→          // Past visits: exclude SCHEDULED (only show COMPLETED, CANCELLED, NO_SHOW)
   451→          where.status = { not: 'SCHEDULED' }
   452→        }
   453→        const visits = await db.visit.findMany({

   995→          const totalCredit = b.account?.journalLines?.reduce((s: number, l: any) => s + l.credit, 0) || 0
   996→          const computedBalance = (b.openingBalance || 0) + totalDebit - totalCredit
   997→          return {
   998→            ...b,
   999→            account: b.account ? { id: b.account.id, code: b.account.code, name: b.account.name } : null,
  1000→            totalDebit,
  1001→            totalCredit,
  1002→            currentBalance: Math.round(computedBalance * 100) / 100,
  1003→            transactionCount: b.account?.journalLines?.length || 0,
  1004→          }
  1005→        })
  1006→        return NextResponse.json(banksWithBalance)
  1007→      }
  1008→
  1009→      case 'deposits': {
  1010→        const where: any = { ...facilityFilter }
  1011→        const depResidentId = searchParams.get('residentId')
  1012→        if (depResidentId) where.residentId = depResidentId
  1013→        const depStatus = searchParams.get('status')
  1014→        if (depStatus) where.status = depStatus

  1017→          include: { resident: { select: { id: true, code: true, firstName: true, lastName: true } } },

  1019→          take: 500,
  1020→        })
  1021→        return NextResponse.json(deposits)
  1022→      }
  1023→
  1024→      default:

  1026→    }
  1027→  } catch (e: any) {
  1028→    console.error('API /data error:', e)
  1029→    return NextResponse.json({ error: e.message }, { status: 500 })
  1030→  }
  1031→}
