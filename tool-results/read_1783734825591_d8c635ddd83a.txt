    31→export function Finance({ facilityId }: { facilityId?: string }) {
    32→  const defaultTabs = [
    33→    { id: 'overview', label: 'Overview' },
    34→    { id: 'unbilled', label: 'Unbilled' },
    35→    { id: 'invoices', label: 'Invoices' },
    36→    { id: 'expenses', label: 'Expenses' },
    37→    { id: 'payments', label: 'Payments' },
    38→    { id: 'receipts', label: 'Receipts' },
    39→    { id: 'accounts', label: 'Chart of Accounts' },
    40→    { id: 'journal', label: 'Journal Entries' },
    41→    { id: 'vendors', label: 'Vendors' },
    42→    { id: 'vendorPayments', label: 'Vendor Payments' },
    43→    { id: 'banks', label: 'Bank Accounts' },
    44→    { id: 'reports', label: 'Reports' },
    45→  ] as const
    46→
    47→  const [tab, setTab] = useState<string>('overview')
    48→  const { data: currentUser } = useFetch<any>('/api/auth/me')
    49→  const { data: settings } = useFetch<any>(facilityId ? `/api/settings?facilityId=${facilityId}` : '/api/settings')
    50→  const billingSettings = settings ? { taxRate: settings.taxRate ?? 5, invoiceDueDays: settings.invoiceDueDays ?? 30, invoicePrefix: settings.invoicePrefix ?? 'INV-' } : { taxRate: 5, invoiceDueDays: 30, invoicePrefix: 'INV-' }
    51→
    52→  // Tab reordering (per-user, same pattern as Settings)
    53→  const [tabOrder, setTabOrder] = useState<string[] | null>(null)
    54→  const [editMode, setEditMode] = useState(false)
    55→  const [draggedId, setDraggedId] = useState<string | null>(null)
    56→  const [savingOrder, setSavingOrder] = useState(false)
    57→
    58→  // Load per-user tab order
    59→  // Version check: if the saved order doesn't include 'receipts' (added in v2),
    60→  // discard it and use the new default order.
    61→  useEffect(() => {
    62→    if (currentUser?.user?.id && settings) {
    63→      const key = `user:${currentUser.user.id}:financeTabOrder`
    64→      const order = settings[key]
    65→      if (Array.isArray(order) && order.includes('receipts')) {
    66→        setTabOrder(order)
    67→      } else if (Array.isArray(order)) {
    68→        // Stale order (missing 'receipts') — reset to new default
    69→        setTabOrder(null)
    70→      }
    71→    }
    72→  }, [currentUser?.user?.id, settings])
    73→
    74→  const allTabs = tabOrder
    75→    ? [
    76→        ...tabOrder.filter(id => defaultTabs.find(t => t.id === id)).map(id => defaultTabs.find(t => t.id === id)!),
    77→        ...defaultTabs.filter(t => !tabOrder.includes(t.id)),
    78→      ]
    79→    : defaultTabs
    80→
    81→  const moveTab = (id: string, direction: 'up' | 'down') => {
    82→    const ids = allTabs.map(t => t.id)
    83→    const idx = ids.indexOf(id)
    84→    if (idx === -1) return
    85→    const newIdx = direction === 'up' ? idx - 1 : idx + 1
    86→    if (newIdx < 0 || newIdx >= ids.length) return
    87→    ;[ids[idx], ids[newIdx]] = [ids[newIdx], ids[idx]]
    88→    setTabOrder(ids)
    89→  }
    90→
    91→  const handleDragStart = (id: string) => setDraggedId(id)
    92→  const handleDragOver = (e: React.DragEvent, overId: string) => {
    93→    e.preventDefault()
    94→    if (!draggedId || draggedId === overId) return
    95→    const ids = allTabs.map(t => t.id)
    96→    const fromIdx = ids.indexOf(draggedId)
    97→    const toIdx = ids.indexOf(overId)
    98→    if (fromIdx === -1 || toIdx === -1) return
    99→    ids.splice(fromIdx, 1)
   100→    ids.splice(toIdx, 0, draggedId)
   101→    setTabOrder(ids)
   102→  }
   103→  const handleDragEnd = () => setDraggedId(null)
   104→
   105→  const saveTabOrder = async () => {
   106→    setSavingOrder(true)
   107→    const order = allTabs.map(t => t.id)
   108→    const userId = currentUser?.user?.id
   109→    if (userId) {
   110→      const key = `user:${userId}:financeTabOrder`
   111→      try {
   112→        await fetch('/api/settings', {
   113→          method: 'POST',
   114→          headers: { 'Content-Type': 'application/json' },
   115→          body: JSON.stringify({ key, value: order }),
   116→        })
   117→        toast.success('Tab order saved')
   118→      } catch (e: any) { toast.error(e.message) }
   119→    }
   120→    setTabOrder(order)
   121→    setEditMode(false)
   122→    setSavingOrder(false)
   123→  }
   124→
   125→  const resetTabOrder = async () => {
   126→    setSavingOrder(true)
   127→    const userId = currentUser?.user?.id
   128→    if (userId) {
   129→      const key = `user:${userId}:financeTabOrder`
   130→      try {
   131→        await fetch(`/api/settings?key=${encodeURIComponent(key)}`, { method: 'DELETE' })
   132→        toast.success('Reset to default order')
   133→      } catch (e: any) { toast.error(e.message) }
   134→    }
   135→    setTabOrder(null)
   136→    setEditMode(false)
   137→    setSavingOrder(false)
   138→  }
   139→
   140→  const role = currentUser?.user?.role
   141→  const canRearrange = role === 'APP_DEVELOPER' || role === 'OWNER' || role === 'MANAGER'
   142→
   143→  return (
   144→    <div className="space-y-4">
   145→      {/* Edit mode toolbar */}
   146→      {editMode && canRearrange && (
   147→        <div className="flex items-center gap-2 p-2 rounded-md bg-primary/5 border border-primary/20">
   148→          <Pencil className="h-3.5 w-3.5 text-primary" />
   149→          <span className="text-xs text-primary font-medium">Rearrange tabs — drag or use arrows</span>
   150→          <div className="ml-auto flex gap-1">
   151→            <Button size="sm" className="h-7 text-xs" onClick={saveTabOrder} disabled={savingOrder}>
   152→              {savingOrder ? <Loader2 className="h-3 w-3 mr-1 animate-spin" /> : <Check className="h-3 w-3 mr-1" />}
   153→              Save
   154→            </Button>
   155→            <Button size="sm" variant="outline" className="h-7 text-xs" onClick={resetTabOrder} disabled={savingOrder}>
   156→              <RotateCcw className="h-3 w-3 mr-1" /> Reset
   157→            </Button>
   158→          </div>
   159→        </div>
   160→      )}
   161→
   162→      <div className="flex gap-1 overflow-x-auto border-b pb-px items-center scrollbar-thin">
   163→        {allTabs.map((t, idx) => (
   164→          <div
   165→            key={t.id}
   166→            draggable={editMode}
   167→            onDragStart={() => editMode && handleDragStart(t.id)}
   168→            onDragOver={(e) => editMode && handleDragOver(e, t.id)}
   169→            onDragEnd={handleDragEnd}
   170→            className={`flex items-center gap-1 flex-shrink-0 ${editMode ? 'cursor-grab active:cursor-grabbing' : ''} ${draggedId === t.id ? 'opacity-50' : ''}`}
   171→          >
   172→            {editMode && <GripVertical className="h-3 w-3 text-muted-foreground/60 flex-shrink-0" />}
   173→            <button
   174→              onClick={() => { if (!editMode) setTab(t.id) }}
   175→              className={`px-3 py-2 text-sm whitespace-nowrap border-b-2 transition-colors flex-shrink-0 ${
   176→                !editMode && tab === t.id
   177→                  ? 'border-primary font-medium text-primary'
   178→                  : 'border-transparent text-muted-foreground hover:text-foreground'
   179→              } ${editMode ? 'cursor-default' : ''}`}
   180→            >
   181→              {t.label}
   182→            </button>
   183→            {editMode && (
   184→              <div className="flex flex-col flex-shrink-0">
   185→                <button onClick={() => moveTab(t.id, 'up')} disabled={idx === 0} className="text-muted-foreground hover:text-foreground disabled:opacity-30 p-0.5">
   186→                  <ChevronUp className="h-3 w-3" />
   187→                </button>
   188→                <button onClick={() => moveTab(t.id, 'down')} disabled={idx === allTabs.length - 1} className="text-muted-foreground hover:text-foreground disabled:opacity-30 p-0.5">
   189→                  <ChevronDown className="h-3 w-3" />
   190→                </button>
   191→              </div>
   192→            )}
   193→          </div>
   194→        ))}
   195→        {!editMode && canRearrange && (
   196→          <button
   197→            onClick={() => setEditMode(true)}
   198→            className="ml-auto flex items-center gap-1 px-2 py-1.5 text-xs text-muted-foreground hover:text-foreground border border-dashed border-muted-foreground/30 rounded-md whitespace-nowrap"
   199→          >
   200→            <Pencil className="h-3 w-3" /> Edit
   201→          </button>
   202→        )}
   203→      </div>
   204→
   205→      {tab === 'overview' && <FinanceOverview facilityId={facilityId} />}
   206→      {tab === 'invoices' && <Invoices facilityId={facilityId} billingSettings={billingSettings} />}
   207→      {tab === 'expenses' && <Expenses facilityId={facilityId} />}
   208→      {tab === 'unbilled' && <UnbilledItems facilityId={facilityId} billingSettings={billingSettings} />}
   209→      {tab === 'payments' && <Payments facilityId={facilityId} />}
   210→      {tab === 'receipts' && <Receipts facilityId={facilityId} />}
   211→      {tab === 'accounts' && <ChartOfAccounts facilityId={facilityId} />}
   212→      {tab === 'journal' && <JournalEntries facilityId={facilityId} />}
   213→      {tab === 'vendors' && <Vendors facilityId={facilityId} />}
   214→      {tab === 'vendorPayments' && <VendorPayments facilityId={facilityId} />}
   215→      {tab === 'banks' && <BankAccounts facilityId={facilityId} />}
   216→      {tab === 'reports' && <AccountingReports facilityId={facilityId} />}
   217→    </div>
   218→  )
   219→}

   222→  const [range, setRange] = useState(90)
   223→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   224→  const { data, loading } = useFetch<any>(`/api/finance?range=${range}${facilityParam}`)
   225→  if (loading || !data) return <Skeleton className="h-96" />
   226→
   227→  const s = data.summary
   228→  const kpis = [
   229→    { label: 'Total Billed', value: fmtMoney(s.totalBilled), icon: FileText, color: 'text-sky-600', bg: 'bg-sky-50' },
   230→    { label: 'Collected', value: fmtMoney(s.totalCollected), icon: TrendingUp, color: 'text-emerald-600', bg: 'bg-emerald-50' },

   236→    { label: 'Net Income', value: fmtMoney(s.netIncome), icon: TrendingUp, color: s.netIncome >= 0 ? 'text-emerald-600' : 'text-red-600', bg: s.netIncome >= 0 ? 'bg-emerald-50' : 'bg-red-50' },
   237→  ]

   239→  const expensePieData = Object.entries(data.expenseByCategory).map(([name, value]) => ({ name: name.replace(/_/g, ' '), value: value as number }))

   242→    Revenue: Math.round(t.revenue),
   243→    Expenses: Math.round(t.expenses),
   244→  }))
   245→
   246→  return (
   247→    <div className="space-y-4">
   248→      <div className="flex items-center gap-2">
   249→        <span className="text-sm text-muted-foreground">Range:</span>
   250→        {[30, 90, 365].map(d => (
   251→          <Button key={d} size="sm" variant={range === d ? 'default' : 'outline'} onClick={() => setRange(d)}>
   252→            {d === 30 ? '30 days' : d === 90 ? '90 days' : '1 year'}
   253→          </Button>
   254→        ))}
   255→      </div>
   256→
   257→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3">
   258→        {kpis.map((k, i) => (
   259→          <Card key={i}>
   260→            <CardContent className="p-4">
   261→              <div className={`p-1.5 rounded-md inline-block ${k.bg} mb-2`}>
   262→                <k.icon className={`h-3.5 w-3.5 ${k.color}`} />
   263→              </div>
   264→              <div className="text-lg font-bold">{k.value}</div>
   265→              <div className="text-xs text-muted-foreground">{k.label}</div>
   266→              {k.sub && <div className="text-[10px] text-muted-foreground/80 mt-0.5">{k.sub}</div>}
   267→            </CardContent>
   268→          </Card>
   269→        ))}
   270→      </div>
   271→
   272→      <Card>
   273→        <CardHeader className="pb-3">
   274→          <CardTitle className="text-sm">Revenue vs Expenses</CardTitle>
   275→        </CardHeader>
   276→        <CardContent>
   277→          <div className="h-72">
   278→            <ResponsiveContainer width="100%" height="100%">
   279→              <BarChart data={timelineData}>
   280→                <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
   281→                <XAxis dataKey="month" fontSize={12} />
   282→                <YAxis fontSize={11} tickFormatter={v => `$${(v / 1000).toFixed(0)}k`} />
   283→                <Tooltip formatter={(v: number) => fmtMoney(v)} />

   285→                <Bar dataKey="Revenue" fill="#10b981" radius={[4, 4, 0, 0]} />
   286→                <Bar dataKey="Expenses" fill="#ef4444" radius={[4, 4, 0, 0]} />
   287→              </BarChart>
   288→            </ResponsiveContainer>
   289→          </div>
   290→        </CardContent>
   291→      </Card>
   292→
   293→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   294→        <Card>
   295→          <CardHeader className="pb-3">
   296→            <CardTitle className="text-sm">Expense Breakdown by Category</CardTitle>
   297→          </CardHeader>
   298→          <CardContent>
   299→            <div className="h-64">
   300→              <ResponsiveContainer width="100%" height="100%">
   301→                <PieChart>
   302→                  <Pie data={expensePieData} dataKey="value" nameKey="name" cx="50%" cy="50%" outerRadius={80} label={(e: any) => `${e.name}`}>
   303→                    {expensePieData.map((_, i) => <Cell key={i} fill={PIE_COLORS[i % PIE_COLORS.length]} />)}

   307→              </ResponsiveContainer>
   308→            </div>
   309→          </CardContent>
   310→        </Card>
   311→
   312→        <Card>

   314→            <CardTitle className="text-sm">Net Income Trend</CardTitle>
   315→          </CardHeader>
   316→          <CardContent>
   317→            <div className="h-64">
   318→              <ResponsiveContainer width="100%" height="100%">
   319→                <LineChart data={timelineData.map((t: any) => ({ ...t, Net: t.Revenue - t.Expenses }))}>
   320→                  <CartesianGrid strokeDasharray="3 3" className="stroke-muted" />
   321→                  <XAxis dataKey="month" fontSize={12} />
   322→                  <YAxis fontSize={11} tickFormatter={v => `$${(v / 1000).toFixed(0)}k`} />
   323→                  <Tooltip formatter={(v: number) => fmtMoney(v)} />
   324→                  <Line dataKey="Net" stroke="#8b5cf6" strokeWidth={2} dot={{ r: 4 }} />
   325→                </LineChart>
   326→              </ResponsiveContainer>
   327→            </div>
   328→          </CardContent>
   329→        </Card>
   330→      </div>
   331→    </div>

   335→function Invoices({ facilityId, billingSettings }: any) {
   336→  const [statusFilter, setStatusFilter] = useState<string>('')
   337→  const [search, setSearch] = useState('')
   338→  const [dateFrom, setDateFrom] = useState<string>('')
   339→  const [dateTo, setDateTo] = useState<string>('')
   340→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   341→  const { data, loading, refetch } = useFetch<any[]>(`/api/data?type=invoices${statusFilter ? `&status=${statusFilter}` : ''}${facilityParam}`)
   342→  const { data: settings } = useFetch<any>('/api/settings')
   343→  const [showCreate, setShowCreate] = useState(false)
   344→  const [printInvoice, setPrintInvoice] = useState<any | null>(null)
   345→  const [editInvoice, setEditInvoice] = useState<any | null>(null)
   346→  const [payInvoice, setPayInvoice] = useState<any | null>(null)
   347→  const [submittingLHDN, setSubmittingLHDN] = useState<string | null>(null) // invoiceId being submitted
   348→
   349→  const lhdnEnabled = settings?.lhdnEnabled === true
   350→
   351→  if (loading) return <Skeleton className="h-96" />
   352→
   353→  const handleLHDNSubmit = async (invoiceId: string) => {
   354→    setSubmittingLHDN(invoiceId)
   355→    try {
   356→      const res = await fetch('/api/e-invoice/submit', {
   357→        method: 'POST',
   358→        headers: { 'Content-Type': 'application/json' },
   359→        body: JSON.stringify({ invoiceId }),
   360→      })
   361→      const data = await res.json()
   362→      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
   363→      toast.success(data.message || 'Submitted to LHDN')
   364→      refetch()
   365→    } catch (e: any) {
   366→      toast.error(e.message)
   367→    }
   368→    setSubmittingLHDN(null)
   369→  }
   370→
   371→  const handleLHDNStatus = async (invoiceId: string) => {
   372→    try {
   373→      const res = await fetch('/api/e-invoice/status', {
   374→        method: 'POST',
   375→        headers: { 'Content-Type': 'application/json' },
   376→        body: JSON.stringify({ invoiceId }),
   377→      })
   378→      const data = await res.json()
   379→      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
   380→      toast.success(`LHDN status: ${data.status}`)
   381→      refetch()
   382→    } catch (e: any) {
   383→      toast.error(e.message)
   384→    }
   385→  }
   386→
   387→  const all = data || []
   388→  const list = all.filter(i => {
   389→    // Date range filter (by issue date)
   390→    if (dateFrom || dateTo) {
   391→      const iDate = new Date(i.issueDate)
   392→      if (dateFrom && iDate < new Date(dateFrom + 'T00:00:00')) return false
   393→      if (dateTo && iDate > new Date(dateTo + 'T23:59:59')) return false
   394→    }
   395→    if (!search) return true
   396→    const s = search.toLowerCase()
   397→    return (
   398→      i.invoiceNumber?.toLowerCase().includes(s) ||
   399→      i.recipient?.toLowerCase().includes(s) ||
   400→      i.status?.toLowerCase().includes(s) ||
   401→      i.notes?.toLowerCase().includes(s) ||
   402→      `${i.resident?.firstName} ${i.resident?.lastName}`.toLowerCase().includes(s) ||
   403→      i.resident?.code?.toLowerCase().includes(s)
   404→    )
   405→  })
   406→  const totalOutstanding = list.filter(i => i.status !== 'PAID' && i.status !== 'CANCELLED').reduce((s, i) => s + Math.round((i.total - i.amountPaid) * 100) / 100, 0)
   407→
   408→  return (
   409→    <div className="space-y-4">
   410→      <StandardSearchBar
   411→        value={search}
   412→        onChange={setSearch}
   413→        placeholder="Search invoices by number, resident, recipient, status..."
   414→        totalCount={all.length}
   415→        filteredCount={list.length}
   416→      />
   417→      <div className="flex flex-col sm:flex-row gap-2 items-stretch sm:items-center justify-between">
   418→        <div className="flex gap-2 flex-wrap items-center">
   419→          {['', 'UNPAID', 'PARTIAL', 'OVERDUE', 'PAID'].map(s => (
   420→            <Button key={s || 'all'} size="sm" variant={statusFilter === s ? 'default' : 'outline'} onClick={() => setStatusFilter(s)}>
   421→              {s || 'All'}
   422→            </Button>
   423→          ))}
   424→          <div className="flex items-center gap-1 text-xs ml-2">
   425→            <input type="date" value={dateFrom} onChange={e => setDateFrom(e.target.value)} className="border rounded px-1.5 py-1 text-xs" title="From issue date" />
   426→            <span className="text-muted-foreground">→</span>
   427→            <input type="date" value={dateTo} onChange={e => setDateTo(e.target.value)} className="border rounded px-1.5 py-1 text-xs" title="To issue date" />
   428→            {(dateFrom || dateTo) && (
   429→              <button onClick={() => { setDateFrom(''); setDateTo('') }} className="text-red-500 hover:text-red-700 ml-1" title="Clear dates">
   430→                <X className="h-3 w-3" />
   431→              </button>
   432→            )}

   672→function PayInvoiceDialog({ invoice, facilityId, onClose, onSaved }: { invoice: any; facilityId?: string; onClose: () => void; onSaved: () => void }) {
   673→  const balance = Math.round((invoice.total - invoice.amountPaid) * 100) / 100
   674→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   675→  const { data: bankAccounts } = useFetch<any[]>(`/api/data?type=bankAccounts${facilityParam}`)
   676→  const [form, setForm] = useState({
   677→    amount: String(balance),
   678→    paymentDate: new Date().toISOString().slice(0, 10),
   679→    method: 'BANK_TRANSFER',
   680→    payerName: invoice.recipient || (invoice.resident ? `${invoice.resident.firstName} ${invoice.resident.lastName}` : ''),
   681→    reference: '',
   682→    bankAccount: '',
   683→    status: 'CLEARED',
   684→    notes: '',
   685→  })
   686→  const [saving, setSaving] = useState(false)
   687→
   688→  const submit = async () => {
   689→    const amt = parseFloat(form.amount)
   690→    if (!amt || amt <= 0) { toast.error('Enter a valid amount'); return }
   691→    if (amt > balance + 0.01) { toast.error(`Amount exceeds invoice balance of ${fmtMoney(balance)}`); return }
   692→    setSaving(true)
   693→    try {
   694→      await apiPost(withFacility('/api/data?type=payments', facilityId), {
   695→        invoiceId: invoice.id,
   696→        residentId: invoice.residentId,
   697→        payerName: form.payerName || null,
   698→        amount: amt,
   699→        paymentDate: form.paymentDate ? new Date(form.paymentDate) : new Date(),
   700→        method: form.method,
   701→        reference: form.reference || null,
   702→        bankAccount: form.bankAccount || null,
   703→        status: form.status,
   704→        notes: form.notes || null,
   705→        applyToInvoice: true,
   706→        facilityId: facilityId || null,
   707→      })
   708→      toast.success(`Payment of ${fmtMoney(amt)} recorded and applied to ${invoice.invoiceNumber}`)
   709→      onSaved()
   710→    } catch (e: any) { toast.error(e.message) }
   711→    setSaving(false)
   712→  }
   713→
   714→  return (
   715→    <Modal title={`Record Payment — ${invoice.invoiceNumber}`} onClose={onClose}>
   716→      <div className="space-y-3 text-sm">
   717→        {/* Invoice summary */}

   720→          <div>Already Paid: <span className="font-medium text-emerald-600">{fmtMoney(invoice.amountPaid)}</span></div>
   721→          <div>Balance Due: <span className="font-medium text-red-600">{fmtMoney(balance)}</span></div>
   722→          <div>Status: <span className="font-medium text-foreground">{invoice.status}</span></div>
   723→          {invoice.resident && (
   724→            <div className="col-span-2">Customer: {invoice.resident.code && <span className="font-mono text-primary">{invoice.resident.code}</span>} {invoice.resident.firstName} {invoice.resident.lastName}</div>
   725→          )}
   726→        </div>
   727→
   728→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   729→          <Field label="Amount (RM) *">
