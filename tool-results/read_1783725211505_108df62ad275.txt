    22→
    23→interface LogEntry {
    24→  timestamp: string
    25→  step: string
    26→  status: 'pending' | 'success' | 'error' | 'info'
    27→  message: string
    28→  data?: any
    29→}
    30→
    31→export function Developer() {
    32→  const [downloading, setDownloading] = useState(false)
    33→  const [backupLoading, setBackupLoading] = useState(false)
    34→  const [restoring, setRestoring] = useState(false)
    35→  const [selectedFile, setSelectedFile] = useState<File | null>(null)
    36→  const [activityLog, setActivityLog] = useState<LogEntry[]>([])
    37→  const logContainerRef = useRef<HTMLDivElement>(null)
    38→  const logEndRef = useRef<HTMLDivElement>(null)
    39→  const { data: me } = useFetch<any>('/api/auth/me')
    40→  const [devTab, setDevTab] = useState<'tools' | 'customers' | 'settings'>('tools')
    41→  // Disk usage warning — fetched on mount, shows banner if > 80%
    42→  const [diskInfo, setDiskInfo] = useState<any>(null)
    43→
    44→  useEffect(() => {
    45→    // Fetch disk usage on mount (lightweight — just the GET endpoint)
    46→    fetch('/api/cleanup', { cache: 'no-store' })
    47→      .then(r => r.json())
    48→      .then(data => setDiskInfo(data))
    49→      .catch(() => {})
    50→  }, [])
    51→
    52→  // Auto-scroll the activity log box to the bottom when a new entry is added.
    53→  // IMPORTANT: We use the container's own scrollTop instead of scrollIntoView,
    54→  // because scrollIntoView also scrolls the nearest scrollable ANCESTOR
    55→  // (i.e. the whole page) — which causes the page to jump downward whenever
    56→  // any button that adds a log entry is clicked.
    57→  const logLengthRef = useRef(0)
    58→  useEffect(() => {
    59→    if (activityLog.length === logLengthRef.current) return // no new entries
    60→    logLengthRef.current = activityLog.length
    61→    const container = logContainerRef.current
    62→    if (container) {
    63→      // Defer until after the new entry renders
    64→      requestAnimationFrame(() => {
    65→        if (container) container.scrollTop = container.scrollHeight
    66→      })
    67→    }
    68→  }, [activityLog])
    69→
    70→  const addLog = (step: string, status: LogEntry['status'], message: string, data?: any) => {
    71→    setActivityLog(prev => [...prev, {
    72→      timestamp: new Date().toISOString(),
    73→      step,
    74→      status,
    75→      message,
    76→      data,
    77→    }])
    78→  }
    79→
    80→  const clearLog = () => setActivityLog([])
    81→
    82→  const handleDownloadProject = async () => {
    83→    setDownloading(true)
    84→    addLog('Download App', 'info', 'Starting project archive download...')
    85→    try {
    86→      addLog('Download App', 'pending', 'Fetching /api/download-project...')
    87→      const response = await fetch('/api/download-project')
    88→      if (!response.ok) {
    89→        const err = await response.json().catch(() => ({}))
    90→        throw new Error(err.error || `HTTP ${response.status}`)
    91→      }
    92→      addLog('Download App', 'success', `Response received: ${response.status} ${response.headers.get('content-type')}`)
    93→      const blob = await response.blob()
    94→      addLog('Download App', 'info', `Blob size: ${(blob.size / 1024 / 1024).toFixed(2)} MB`)
    95→      if (blob.size < 10000) {
    96→        throw new Error(`File too small (${blob.size} bytes)`)
    97→      }
    98→      const url = window.URL.createObjectURL(blob)
    99→      const a = document.createElement('a')
   100→      a.href = url
   101→      a.download = `nursing-home-app-${new Date().toISOString().slice(0, 10)}.tar.gz`
   102→      document.body.appendChild(a)
   103→      a.click()
   104→      document.body.removeChild(a)
   105→      setTimeout(() => window.URL.revokeObjectURL(url), 10000)
   106→      addLog('Download App', 'success', `Download complete: ${(blob.size / 1024 / 1024).toFixed(1)} MB`)
   107→      toast.success(`Project downloaded (${(blob.size / 1024 / 1024).toFixed(1)} MB)`)
   108→    } catch (e: any) {
   109→      addLog('Download App', 'error', e.message || 'Failed')
   110→      toast.error(e.message || 'Failed to download project')
   111→    }
   112→    setDownloading(false)
   113→  }
   114→
   115→  const handleDatabaseBackup = async () => {
   116→    setBackupLoading(true)
   117→    addLog('Backup DB', 'info', 'Starting database backup download...')
   118→    try {
   119→      const response = await fetch('/api/backup')
   120→      if (!response.ok) {
   121→        const err = await response.json().catch(() => ({}))
   122→        throw new Error(err.error || `HTTP ${response.status}`)
   123→      }
   124→      const blob = await response.blob()
   125→      addLog('Backup DB', 'success', `Backup downloaded: ${(blob.size / 1024 / 1024).toFixed(2)} MB`)
   126→      const url = window.URL.createObjectURL(blob)
   127→      const a = document.createElement('a')
   128→      a.href = url
   129→      const cd = response.headers.get('Content-Disposition') || ''
   130→      const match = cd.match(/filename="?([^"]+)"?/)
   131→      a.download = match ? match[1] : `backup-${new Date().toISOString().slice(0, 10)}.db`
   132→      document.body.appendChild(a)
   133→      a.click()
   134→      document.body.removeChild(a)
   135→      window.URL.revokeObjectURL(url)
   136→      toast.success('Database backup downloaded')
   137→    } catch (e: any) {
   138→      addLog('Backup DB', 'error', e.message || 'Failed')
   139→      toast.error(e.message || 'Failed to download backup')
   140→    }
   141→    setBackupLoading(false)
   142→  }
   143→
   144→  const handleDatabaseRestore = async () => {
   145→    if (!selectedFile) {
   146→      toast.error('Please select a database file first')
   147→      return
   148→    }
   149→
   150→    clearLog()
   151→    addLog('Restore DB', 'info', `Starting restore from: ${selectedFile.name} (${(selectedFile.size / 1024 / 1024).toFixed(2)} MB)`)
   152→
   153→    // Step 1: Validate file
   154→    addLog('Restore DB', 'pending', 'Step 1: Validating file...')
   155→    const fileName = selectedFile.name.toLowerCase()
   156→    if (!fileName.endsWith('.db') && !fileName.endsWith('.sqlite') && !fileName.endsWith('.sqlite3')) {
   157→      addLog('Restore DB', 'error', 'File must be .db, .sqlite, or .sqlite3')
   158→      toast.error('Invalid file type')
   159→      return
   160→    }
   161→    if (selectedFile.size < 1000) {
   162→      addLog('Restore DB', 'error', 'File too small (< 1KB)')
   163→      toast.error('File too small')
   164→      return
   165→    }
   166→    addLog('Restore DB', 'success', `File valid: ${selectedFile.name}, ${(selectedFile.size / 1024 / 1024).toFixed(2)} MB`)
   167→
   168→    // Step 2: Confirm
   169→    addLog('Restore DB', 'pending', 'Step 2: Waiting for confirmation...')
   170→    if (!confirm(
   171→      `Restore database from "${selectedFile.name}"?\n\nThis will REPLACE all current data.\nA backup will be saved automatically.`
   172→    )) {
   173→      addLog('Restore DB', 'info', 'User cancelled restore')
   174→      return
   175→    }
   176→    addLog('Restore DB', 'success', 'User confirmed')
   177→
   178→    // Step 3: Upload
   179→    setRestoring(true)
   180→    addLog('Restore DB', 'pending', 'Step 3: Uploading file to /api/restore-database...')
   181→    try {
   182→      const formData = new FormData()
   183→      formData.append('database', selectedFile)
   184→      const uploadStart = Date.now()
   185→      const response = await fetch('/api/restore-database', {
   186→        method: 'POST',
   187→        body: formData,
   188→      })
   189→      const uploadTime = ((Date.now() - uploadStart) / 1000).toFixed(2)
   190→      addLog('Restore DB', 'success', `Upload complete in ${uploadTime}s — HTTP ${response.status}`)
   191→
   192→      // Step 4: Parse response
   193→      addLog('Restore DB', 'pending', 'Step 4: Parsing server response...')
   194→      const result = await response.json()
   195→
   196→      if (!response.ok) {
   197→        throw new Error(result.error || `HTTP ${response.status}`)
   198→      }
   199→      addLog('Restore DB', 'success', `Server: ${result.message || 'OK'}`)
   200→      addLog('Restore DB', 'info', `Restore scope: ${result.scope || 'full'}`)
   201→
   202→      // Step 5: Show table counts (post-restore snapshot)
   203→      addLog('Restore DB', 'pending', 'Step 5: Verifying table counts in live DB...')
   204→
   205→      if (result.tableCounts) {
   206→        for (const [tableName, count] of Object.entries(result.tableCounts)) {
   207→          const c = count as number
   208→          addLog('Restore DB', c >= 0 ? 'success' : 'error',
   209→            `  ${tableName}: ${c >= 0 ? `${c} rows ✅` : 'count failed ❌'}`
   210→          )
   211→        }
   212→      }
   213→
   214→      if (result.success) {
   215→        addLog('Restore DB', 'success', '✅ RESTORE CONFIRMED SUCCESSFUL — server verified the data')
   216→        toast.success('Database restored and verified! Reloading...')
   217→        setSelectedFile(null)
   218→        const fileInput = document.getElementById('db-upload') as HTMLInputElement
   219→        if (fileInput) fileInput.value = ''
   220→        bumpDataVersion()
   221→        addLog('Restore DB', 'info', 'Performing full page reload in 2 seconds...')
   222→        setTimeout(() => {
   223→          window.location.href = '/?restored=' + Date.now()
   224→        }, 2000)
   225→      } else {
   226→        addLog('Restore DB', 'error', '❌ RESTORE VERIFICATION FAILED')
   227→        addLog('Restore DB', 'info', 'Try refreshing the page — the data may still have been restored.')
   228→        toast.error('Restore verification failed — check log')
   229→      }
   230→
   231→    } catch (e: any) {
   232→      addLog('Restore DB', 'error', `Failed: ${e.message || 'Unknown error'}`)
   233→      toast.error(e.message || 'Failed to restore database')
   234→    }
   235→    setRestoring(false)
   236→  }
   237→
   238→  return (
   239→    <div className="space-y-4">
   240→      {/* Disk usage warning banner — shows when disk > 80% */}
   241→      {diskInfo?.disk && (() => {
   242→        const pct = parseInt(diskInfo.disk.usePercent?.replace('%', '') || '0')
   243→        if (pct < 80) return null
   244→        const isCritical = pct >= 90
   245→        return (
   246→          <div className={`rounded-md border p-3 flex items-start gap-3 ${isCritical ? 'border-red-300 bg-red-50' : 'border-amber-300 bg-amber-50'}`}>
   247→            <AlertTriangle className={`h-5 w-5 flex-shrink-0 mt-0.5 ${isCritical ? 'text-red-600' : 'text-amber-600'}`} />
   248→            <div className="flex-1 text-sm">
   249→              <div className={`font-medium ${isCritical ? 'text-red-800' : 'text-amber-800'}`}>
   250→                {isCritical ? '⚠️ Critical: Disk almost full!' : '⚠️ Disk usage is high'}
   251→              </div>
   252→              <div className={`text-xs mt-0.5 ${isCritical ? 'text-red-700' : 'text-amber-700'}`}>
   253→                Disk is <strong>{diskInfo.disk.usePercent}</strong> used ({diskInfo.disk.used} of {diskInfo.disk.total}, only {diskInfo.disk.available} free).
   254→                {isCritical
   255→                  ? ' Deployments and backups may fail. Please clean up temporary files immediately.'
   256→                  : ' Consider cleaning up temporary files to prevent deployment failures.'}
   257→              </div>
   258→              {diskInfo.autoCleaned && (
   259→                <div className="text-xs mt-1 text-emerald-700 bg-emerald-50 border border-emerald-200 rounded p-1.5">
   260→                  ✅ Auto-cleaned {diskInfo.autoCleaned.count} old build directories ({diskInfo.autoCleaned.freedMB} MB freed). Refresh to see updated disk usage.
   261→                </div>
   262→              )}
   263→              <div className="mt-2 flex gap-2">
   264→                <Button
   265→                  size="sm"
   266→                  variant={isCritical ? 'destructive' : 'outline'}
   267→                  onClick={() => setDevTab('tools')}
   268→                >
   269→                  <HardDrive className="h-3 w-3 mr-1" /> Go to Cleanup
   270→                </Button>
   271→              </div>
   272→            </div>
   273→          </div>
   274→        )
   275→      })()}
   276→
   277→      {/* Tab bar */}
   278→      <div className="flex gap-1 overflow-x-auto border-b pb-px scrollbar-thin">
   279→        <button onClick={() => setDevTab('tools')} className={`px-3 py-2 text-sm whitespace-nowrap border-b-2 flex-shrink-0 ${devTab === 'tools' ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground'}`}>
   280→          Developer Tools
   281→        </button>
   282→        <button onClick={() => setDevTab('customers')} className={`px-3 py-2 text-sm whitespace-nowrap border-b-2 flex-shrink-0 ${devTab === 'customers' ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground'}`}>
   283→          App Customers
   284→        </button>
   285→        <button onClick={() => setDevTab('settings')} className={`px-3 py-2 text-sm whitespace-nowrap border-b-2 flex-shrink-0 ${devTab === 'settings' ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground'}`}>
   286→          App Settings
   287→        </button>
   288→        <button onClick={() => setDevTab('customization')} className={`px-3 py-2 text-sm whitespace-nowrap border-b-2 flex-shrink-0 ${devTab === 'customization' ? 'border-primary font-medium text-primary' : 'border-transparent text-muted-foreground'}`}>
   289→          Customization
   290→        </button>
   291→      </div>
   292→
   293→      {devTab === 'customers' && <AppCustomersTab />}
   294→
   295→      {devTab === 'settings' && <AppSettingsTab />}
   296→
   297→      {devTab === 'customization' && <CustomizationTab />}
   298→
   299→      {devTab === 'tools' && (
   300→      <>
   301→      {/* Warning banner */}
   302→      <Card className="border-amber-300 bg-amber-50">
   303→        <CardContent className="p-4 flex items-start gap-3">
   304→          <AlertTriangle className="h-5 w-5 text-amber-600 flex-shrink-0 mt-0.5" />
   305→          <div className="text-sm text-amber-900">
   306→            <div className="font-semibold mb-1">Developer Tools — Owner Access Only</div>
   307→            <p>This module contains tools for backing up, downloading, and restoring the application database. Only the Owner account can access this page.</p>
   308→          </div>
   309→        </CardContent>
   310→      </Card>
   311→
   312→      {/* Download full project */}
   313→      <Card>
   314→        <CardHeader>
   315→          <CardTitle className="text-sm flex items-center gap-2">
   316→            <FileArchive className="h-4 w-4" /> Download Full Application
   317→          </CardTitle>
   318→          <CardDescription>
   319→            Download the entire project as a .tar.gz archive — includes source code, database, and configuration.
   320→          </CardDescription>
   321→        </CardHeader>
   322→        <CardContent>
   323→          <Button onClick={handleDownloadProject} disabled={downloading} className="w-full sm:w-auto">
   324→            {downloading ? (
   325→              <><Loader2 className="h-4 w-4 mr-2 animate-spin" /> Preparing...</>
   326→            ) : (
   327→              <><Download className="h-4 w-4 mr-2" /> Download Full App (.tar.gz)</>
   328→            )}
   329→          </Button>
   330→        </CardContent>
   331→      </Card>
   332→
   333→      {/* Database backup & restore */}
   334→      <Card>
   335→        <CardHeader>
   336→          <CardTitle className="text-sm flex items-center gap-2">
   337→            <Database className="h-4 w-4" /> Database Backup & Restore
   338→          </CardTitle>
   339→          <CardDescription>Download a backup or restore from a previously downloaded .db file.</CardDescription>
   340→        </CardHeader>
   341→        <CardContent className="space-y-4">
   342→          {/* Download backup */}
   343→          <div className="flex items-center justify-between gap-3 flex-wrap">
   344→            <div className="text-sm">
   345→              <div className="font-medium">Download Backup</div>
   346→              <div className="text-xs text-muted-foreground">Download current database as .db file</div>
   347→            </div>
   348→            <Button onClick={handleDatabaseBackup} disabled={backupLoading} variant="outline">
   349→              {backupLoading ? (
   350→                <><Loader2 className="h-4 w-4 mr-2 animate-spin" /> Preparing...</>
   351→              ) : (
   352→                <><Download className="h-4 w-4 mr-2" /> Download (.db)</>
   353→              )}
   354→            </Button>
   355→          </div>
   356→
   357→          <div className="border-t pt-4 space-y-3">
   358→            {/* Restore */}
   359→            <div className="text-sm">
   360→              <div className="font-medium flex items-center gap-1">
   361→                <Upload className="h-3.5 w-3.5" /> Restore from Backup
   362→              </div>
   363→              <div className="text-xs text-muted-foreground">Upload a .db file to replace the current database</div>
   364→            </div>
   365→
   366→            <div className="flex flex-col sm:flex-row gap-2">
   367→              <input
   368→                id="db-upload"
   369→                type="file"
   370→                accept=".db,.sqlite,.sqlite3"
   371→                onChange={(e) => {
   372→                  const file = e.target.files?.[0]
   373→                  if (file) {
   374→                    const fn = file.name.toLowerCase()
   375→                    if (!fn.endsWith('.db') && !fn.endsWith('.sqlite') && !fn.endsWith('.sqlite3')) {
   376→                      toast.error('Please select a .db file')
   377→                      e.target.value = ''
   378→                      return
   379→                    }
   380→                    setSelectedFile(file)
   381→                    addLog('File Selected', 'info', `${file.name} (${(file.size / 1024 / 1024).toFixed(2)} MB)`)
   382→                  }
   383→                }}
   384→                className="text-sm border rounded px-3 py-1.5 flex-1 cursor-pointer file:mr-3 file:py-1 file:px-3 file:rounded file:border-0 file:text-sm file:font-medium file:bg-primary file:text-primary-foreground hover:file:bg-primary/90"
   385→              />
   386→              <Button onClick={handleDatabaseRestore} disabled={restoring || !selectedFile}>
   387→                {restoring ? (
   388→                  <><Loader2 className="h-4 w-4 mr-2 animate-spin" /> Restoring...</>
   389→                ) : (
   390→                  <><RefreshCw className="h-4 w-4 mr-2" /> Restore</>
   391→                )}
   392→              </Button>
   393→            </div>
   394→
   395→            {selectedFile && (
   396→              <div className="text-xs text-muted-foreground bg-muted/30 rounded-md p-2.5 flex items-center gap-2">
   397→                <Database className="h-3.5 w-3.5" />
   398→                <span className="font-medium">{selectedFile.name}</span>
   399→                <span>•</span>
   400→                <span>{(selectedFile.size / 1024 / 1024).toFixed(2)} MB</span>
   401→                <button
   402→                  onClick={() => { setSelectedFile(null); const fi = document.getElementById('db-upload') as HTMLInputElement; if (fi) fi.value = '' }}
   403→                  className="ml-auto text-red-500 hover:text-red-700"
   404→                >Remove</button>
   405→              </div>
   406→            )}
   407→
   408→            <div className="text-xs text-amber-700 bg-amber-50 border border-amber-200 rounded-md p-2.5 flex items-start gap-2">
   409→              <AlertTriangle className="h-3.5 w-3.5 flex-shrink-0 mt-0.5" />
   410→              <div><strong>Warning:</strong> Restoring replaces all current data. Current DB is backed up to /tmp. Only upload databases downloaded from this app.</div>
   411→            </div>
   412→          </div>
   413→        </CardContent>
   414→      </Card>
   415→
   416→      {/* Google Drive backup */}
   417→      <GoogleDriveCard addLog={addLog} />
   418→
   419→      {/* Activity Log */}
   420→      <Card>
   421→        <CardHeader>
   422→          <div className="flex items-center justify-between">
   423→            <CardTitle className="text-sm flex items-center gap-2">
   424→              <Terminal className="h-4 w-4" /> Activity Log
   425→              {activityLog.length > 0 && <Badge variant="outline" className="text-xs">{activityLog.length} entries</Badge>}
   426→            </CardTitle>
   427→            {activityLog.length > 0 && (
   428→              <Button size="sm" variant="ghost" onClick={clearLog}>Clear</Button>
   429→            )}
   430→          </div>
   431→          <CardDescription>Tracks every step of backup, download, and restore operations — use this to diagnose failures.</CardDescription>
   432→        </CardHeader>
   433→        <CardContent>
   434→          {activityLog.length === 0 ? (
   435→            <p className="text-sm text-muted-foreground text-center py-8">No activity yet. Perform a backup, download, or restore to see logs here.</p>
   436→          ) : (
   437→            <div ref={logContainerRef} className="bg-slate-900 text-slate-100 rounded-lg p-3 font-mono text-xs space-y-1 max-h-96 overflow-y-auto">
   438→              {activityLog.map((log, i) => (
   439→                <div key={i} className="flex items-start gap-2">
   440→                  <span className="text-slate-500 flex-shrink-0">{log.timestamp.slice(11, 23)}</span>
   441→                  <span className="flex-shrink-0">
   442→                    {log.status === 'success' && <CheckCircle className="h-3.5 w-3.5 text-emerald-400" />}
   443→                    {log.status === 'error' && <XCircle className="h-3.5 w-3.5 text-red-400" />}
   444→                    {log.status === 'pending' && <Loader2 className="h-3.5 w-3.5 text-amber-400" />}
   445→                    {log.status === 'info' && <Info className="h-3.5 w-3.5 text-sky-400" />}
   446→                  </span>
   447→                  <span className="text-slate-500 flex-shrink-0">[{log.step}]</span>
   448→                  <span className={
   449→                    log.status === 'success' ? 'text-emerald-300' :
   450→                    log.status === 'error' ? 'text-red-300' :
   451→                    log.status === 'pending' ? 'text-amber-300' :
   452→                    'text-slate-300'
   453→                  }>{log.message}</span>
   454→                </div>
   455→              ))}
   456→              <div ref={logEndRef} />
   457→            </div>
   458→          )}
   459→        </CardContent>
   460→      </Card>
   461→
   462→      {/* Disk Space & Cleanup */}
   463→      <DiskSpaceCard />
   464→
   465→      {/* System info */}
   466→      <Card>
   467→        <CardHeader>

   869→// ============ DEMO ACCOUNT ROW (editable credentials + org access) ============
   870→function DemoAccountRow({ user, facilities, organizations, onSaved }: { user: any; facilities: any[]; organizations: any[]; onSaved: () => void }) {
   871→  const [email, setEmail] = useState(user.email || '')
   872→  const [password, setPassword] = useState('')
   873→  const [showPassword, setShowPassword] = useState(false)
   874→  const [saving, setSaving] = useState(false)
   875→
   876→  const handleSaveCredentials = async () => {
   877→    setSaving(true)
   878→    try {
   879→      const body: any = {}
   880→      if (email.trim() && email.trim().toLowerCase() !== user.email) {
   881→        body.email = email.trim().toLowerCase()
   882→      }
   883→      if (password.trim()) {
   884→        if (password.length < 6) {
   885→          toast.error('Password must be at least 6 characters')
   886→          setSaving(false)
   887→          return
   888→        }
   889→        body.password = password
   890→      }
   891→      if (Object.keys(body).length === 0) {
   892→        toast.info('No changes to save')
   893→        setSaving(false)
   894→        return
   895→      }
   896→      const res = await fetch(`/api/users?id=${user.id}`, {
   897→        method: 'PATCH',
   898→        headers: { 'Content-Type': 'application/json' },
   899→        body: JSON.stringify(body),
   900→      })
   901→      const data = await res.json()
   902→      if (!res.ok) throw new Error(data.error || `HTTP ${res.status}`)
   903→      toast.success(`Credentials updated for ${user.name}`)
   904→      setPassword('')
   905→      onSaved()
   906→    } catch (e: any) {
   907→      toast.error(e.message)
   908→    }
   909→    setSaving(false)
   910→  }
   911→
   912→  // Toggle an entire organization's facilities for this demo user
   913→  const toggleOrganization = async (org: any) => {
   914→    const orgFacilityIds = (org.facilities || []).map((f: any) => f.id)
   915→    const currentFids = (user.facilityIds || '').split(',').map(s => s.trim()).filter(Boolean)
   916→    const allChecked = orgFacilityIds.length > 0 && orgFacilityIds.every((fid: string) => currentFids.includes(fid))
   917→    const next = allChecked
   918→      ? currentFids.filter(id => !orgFacilityIds.includes(id))
   919→      : [...new Set([...currentFids, ...orgFacilityIds])]
   920→    try {
   921→      const res = await fetch(`/api/users?id=${user.id}`, {
   922→        method: 'PATCH',
   923→        headers: { 'Content-Type': 'application/json' },
   924→        body: JSON.stringify({ facilityIds: next.join(',') }),
   925→      })
   926→      if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error || `HTTP ${res.status}`)
   927→      onSaved()
   928→    } catch (e: any) {
   929→      toast.error(e.message)
   930→    }
   931→  }
   932→
   933→  // Toggle a single facility
   934→  const toggleFacility = async (fid: string) => {
   935→    const fids = (user.facilityIds || '').split(',').map(s => s.trim()).filter(Boolean)
   936→    const next = fids.includes(fid) ? fids.filter(id => id !== fid) : [...fids, fid]
   937→    try {
   938→      const res = await fetch(`/api/users?id=${user.id}`, {
   939→        method: 'PATCH',
   940→        headers: { 'Content-Type': 'application/json' },

  1091→// ============ ORG MODULE PICKER ============
  1092→function OrgModulePicker({ orgId, orgName, settings, businessType, onSaved }: { orgId: string; orgName: string; settings: any; businessType?: string; onSaved: () => void }) {
  1093→  // Key: orgModules:<orgId> — if not set, org has access to ALL modules
  1094→  const settingKey = `orgModules:${orgId}`
  1095→  const serverModules: string[] | undefined = settings?.[settingKey]
  1096→
  1097→  // Local state for optimistic updates — checkboxes update instantly without waiting for refetch
  1098→  const [localModules, setLocalModules] = useState<string[] | undefined>(serverModules)
  1099→
  1100→  // Sync local state when server data changes (e.g. after onSaved refetch)
  1101→  useEffect(() => {
  1102→    setLocalModules(serverModules)
  1103→  }, [serverModules])
  1104→
  1105→  const currentModules = localModules
  1106→  const isEnabled = (moduleId: string) => {
  1107→    if (!Array.isArray(currentModules)) return true  // Not set = all modules enabled
  1108→    return currentModules.includes(moduleId)
  1109→  }
  1110→
  1111→  const toggleModule = async (moduleId: string) => {
  1112→    const current = Array.isArray(currentModules) ? currentModules : ALL_MODULES.map(m => m.id)
  1113→    const next = current.includes(moduleId)
  1114→      ? current.filter(id => id !== moduleId)
  1115→      : [...current, moduleId]
  1116→    // Optimistic update — update local state immediately so the checkbox reflects instantly
  1117→    setLocalModules(next)
  1118→    try {
  1119→      await fetch('/api/settings', {
  1120→        method: 'POST',
  1121→        headers: { 'Content-Type': 'application/json' },
  1122→        body: JSON.stringify({ key: settingKey, value: next }),
  1123→      })
  1124→      toast.success(`Module access updated for ${orgName}`)
  1125→      onSaved()
  1126→    } catch (e: any) {
  1127→      // Revert on error
  1128→      setLocalModules(current)
  1129→      toast.error(e.message)
  1130→    }
  1131→  }
  1132→
  1133→  const setAll = async (all: boolean) => {
  1134→    const value = all ? ALL_MODULES.map(m => m.id) : ['dashboard', 'profile']
  1135→    // Optimistic update
  1136→    setLocalModules(value)
  1137→    try {
  1138→      await fetch('/api/settings', {
  1139→        method: 'POST',
  1140→        headers: { 'Content-Type': 'application/json' },
  1141→        body: JSON.stringify({ key: settingKey, value }),
  1142→      })
  1143→      toast.success(all ? `All modules enabled for ${orgName}` : `Minimal modules set for ${orgName}`)
  1144→      onSaved()
  1145→    } catch (e: any) {
  1146→      setLocalModules(serverModules)
  1147→      toast.error(e.message)
  1148→    }
  1149→  }
  1150→
  1151→  const resetToDefault = async () => {
  1152→    // Reset to the business type preset modules.
  1153→    // Checks for Developer-customized module list (saved as businessTypeModules:<type>)
  1154→    // first — if the Developer has customized the preset via Org Type Management,
  1155→    // use that. Otherwise fall back to the static code preset.
  1156→    const overrideKey = `businessTypeModules:${businessType || 'nursing_home'}`
  1157→    const savedOverride = settings?.[overrideKey]
  1158→    const preset = getBusinessTypePreset(businessType)
  1159→    const presetModules = Array.isArray(savedOverride) ? savedOverride : preset.visibleModules
  1160→    // Optimistic update
  1161→    setLocalModules(presetModules)
  1162→    try {
  1163→      await fetch('/api/settings', {
  1164→        method: 'POST',
  1165→        headers: { 'Content-Type': 'application/json' },
  1166→        body: JSON.stringify({ key: settingKey, value: presetModules }),
  1167→      })
  1168→      const source = Array.isArray(savedOverride) ? `${preset.label} (customized)` : preset.label
  1169→      toast.success(`Reset to ${source} defaults (${presetModules.length} modules) for ${orgName}`)
  1170→      onSaved()
  1171→    } catch (e: any) {
  1172→      setLocalModules(serverModules)
  1173→      toast.error(e.message)
  1174→    }
  1175→  }
  1176→
  1177→  const enabledCount = Array.isArray(currentModules) ? currentModules.length : ALL_MODULES.length
  1178→
  1179→  // Compute the module count for the reset button — includes Developer overrides
  1180→  const resetOverrideKey = `businessTypeModules:${businessType || 'nursing_home'}`
  1181→  const resetSavedOverride = settings?.[resetOverrideKey]
  1182→  const resetPreset = getBusinessTypePreset(businessType)
  1183→  const resetModuleCount = Array.isArray(resetSavedOverride) ? resetSavedOverride.length : resetPreset.visibleModules.length
  1184→  const resetLabel = Array.isArray(resetSavedOverride) ? `${resetPreset.label} (customized)` : resetPreset.label
  1185→
  1186→  return (
  1187→    <div className="mt-2 ml-8 p-2 border rounded bg-muted/30 space-y-2">
  1188→      <div className="flex items-center justify-between">
  1189→        <div className="text-[10px] font-semibold text-muted-foreground">MODULE ACCESS FOR {orgName.toUpperCase()}</div>
  1190→        <div className="text-[10px] text-muted-foreground">{enabledCount} / {ALL_MODULES.length} modules</div>
  1191→      </div>
  1192→      <div className="text-[10px] text-muted-foreground">
  1193→        Select which modules this organization can access. Unchecked modules will be hidden from all users in this org.
  1194→        Click "Reset to {resetLabel}" to restore the default module set for this org's business type
  1195→        {Array.isArray(resetSavedOverride) ? ' (includes your customizations from Org Type Management)' : ''}.
  1196→      </div>
  1197→      <div className="grid grid-cols-2 sm:grid-cols-3 gap-1">
  1198→        {ALL_MODULES.map(m => {
  1199→          const checked = isEnabled(m.id)
  1200→          return (
  1201→            <label key={m.id} className="flex items-center gap-1.5 cursor-pointer p-1 rounded hover:bg-muted/50 text-xs">
  1202→              <input
  1203→                type="checkbox"
  1204→                checked={checked}
  1205→                onChange={() => toggleModule(m.id)}
  1206→                className="h-3 w-3"
  1207→              />
  1208→              <span className={checked ? 'font-medium' : 'text-muted-foreground'}>{m.label}</span>
  1209→            </label>
  1210→          )
  1211→        })}
  1212→      </div>
  1213→      <div className="flex gap-2 pt-1">
  1214→        <Button size="sm" variant="outline" className="h-6 text-[10px]" onClick={() => setAll(true)}>Enable All</Button>
  1215→        <Button size="sm" variant="outline" className="h-6 text-[10px]" onClick={() => setAll(false)}>Minimal Only</Button>
  1216→        <Button size="sm" variant="ghost" className="h-6 text-[10px] text-red-600" onClick={resetToDefault}>
  1217→          Reset to {resetLabel} ({resetModuleCount} modules)
  1218→        </Button>
  1219→      </div>
  1220→    </div>
  1221→  )
  1222→}
