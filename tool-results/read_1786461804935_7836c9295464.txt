     1→/**
     2→ * Shared registry of bulk-import definitions for all importable entity types.
     3→ *
     4→ * Used by:
     5→ *   - Settings → Backup & Restore → Bulk Imports section (the canonical UI)
     6→ *   - Anywhere else that wants to launch an import dialog (e.g. a "Quick Import"
     7→ *     button in the Residents module can simply open the same dialog using
     8→ *     the registry entry)
     9→ *
    10→ * Each definition includes:
    11→ *   - entityType: passed to /api/import-undo for the undo operation
    12→ *   - title: shown in the dialog header
    13→ *   - description: shown on the card in the Bulk Imports section
    14→ *   - icon: lucide-react icon for the card
    15→ *   - columns: CsvColumn[] definition with aliases + validators
    16→ *   - templateRows: sample rows for the downloadable template
    17→ *   - onImport: callback that creates records via the API. Receives (rows, batchId, facilityId).
    18→ *   - buildPayload: optional, builds the POST body from a mapped row. Defaults to spreading the row.
    19→ */
    20→

    28→// Business types each import applies to. `null` (or omitted) = applies to all.
    29→// Used by BulkImports.tsx to hide irrelevant imports based on the current org's business type.
    30→export type BusinessTypeKey = 'nursing_home' | 'tailor' | 'clinic' | 'generic' | string
    31→
    32→export interface BulkImportDefinition {
    33→  entityType: 'resident' | 'staff' | 'room' | 'product' | 'vendor' | 'account' | 'bankAccount' | 'expense' | 'payment' | 'inventory' | 'purchaseOrder' | 'journalEntry' | 'productVendorPrice'
    34→  title: string
    35→  description: string
    36→  icon: LucideIcon
    37→  columns: CsvColumn[]
    38→  templateRows: Record<string, any>[]
    39→  /** API endpoint to POST each row to (without query string). facilityId is appended. */
    40→  endpoint: string
    41→  /** Build the POST body from a mapped row. Defaults to spreading the row + adding importBatchId. */
    42→  buildPayload?: (row: any, batchId: string) => any
    43→  /** Business types this import is applicable to. If omitted, applies to all types. */
    44→  applicableBusinessTypes?: BusinessTypeKey[]
    45→}
