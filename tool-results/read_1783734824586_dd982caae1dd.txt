   177→// ============== CUSTOM FIELDS ==============
   178→// Two-tier custom field system:
   179→//   1. GlobalCustomField — a global library of field definitions, managed by
   180→//      the Developer. These are the "master" definitions available to all orgs.
   181→//   2. OrgCustomField — per-org selections: which global fields are enabled
   182→//      for this org, plus an optional org-specific label override.
   183→//
   184→// This allows the Developer to build a library of fields (Blood Type, Chest
   185→// Measurement, Fabric Preference, etc.) once, and each org enables the ones
   186→// relevant to their business type. Orgs can also rename fields locally
   187→// (e.g. "Chest" → "Dada" in Malay) without affecting the global definition.
   188→//
   189→// The existing CustomField model is kept for backward compatibility — the
   190→// /api/custom-fields GET endpoint merges both sources so existing UI code
   191→// continues to work.
   192→
   193→model GlobalCustomField {
   194→  id              String   @id @default(cuid())
   195→  label           String                    // Global label, e.g. "Blood Type"
   196→  key             String   @unique          // Auto-derived: "blood_type"
   197→  type            String   @default("TEXT") // TEXT, NUMBER, DATE, SELECT, TEXTAREA, REFERENCE
   198→  options         String?                   // JSON array for SELECT, e.g. ["A+","O+"]
   199→  unit            String?                   // "cm", "kg"
   200→  required        Boolean  @default(false)
   201→  targetEntity    String   @default("resident") // resident, invoice, product, staff
   202→  referenceEntity String?                   // for REFERENCE type: product, staff, resident, invoice
   203→  description     String?                   // Developer notes about this field
   204→  // Which business type(s) this field belongs to. Null = all business types.
   205→  // Stored as JSON array: ["nursing_home","clinic"] or null for all.
   206→  businessTypes   String?
   207→  sortOrder       Int      @default(0)
   208→  active          Boolean  @default(true)
   209→  createdAt       DateTime @default(now())
   210→  updatedAt       DateTime @updatedAt
   211→
   212→  orgSelections   OrgCustomField[]
   213→}
   214→
   215→model OrgCustomField {
   216→  id              String   @id @default(cuid())
   217→  orgId           String                    // links to Organization
   218→  globalFieldId   String                    // links to GlobalCustomField
   219→  globalField     GlobalCustomField @relation(fields: [globalFieldId], references: [id], onDelete: Cascade)
   220→  // Optional org-specific label override. If null, uses the global label.
   221→  labelOverride   String?
   222→  // Optional org-specific sort order (overrides global sortOrder)
   223→  sortOrderOverride Int?
   224→  // Whether the field is enabled for this org (default true — selecting a
   225→  // field enables it; deselecting sets this to false)
   226→  enabled         Boolean  @default(true)
   227→  createdAt       DateTime @default(now())
   228→  updatedAt       DateTime @updatedAt
   229→
   230→  @@unique([orgId, globalFieldId])
   231→}

   233→// ============== CUSTOM TABS ==============
   234→// Global custom tab definitions — a collection of fields (built-in + custom)
   235→// that appears as a tab in the customer detail view. Managed by the Developer
   236→// and applied globally. Each org can show/hide tabs and override labels.
   237→//
   238→// Example: A "Measurements" tab for tailors containing Chest, Waist, Hip
   239→// custom fields. A "Medical Info" tab for clinics containing Blood Type,
   240→// Allergies, etc.
   241→
   242→model GlobalCustomTab {
   243→  id          String   @id @default(cuid())
   244→  label       String                    // Tab name, e.g. "Measurements"
   245→  key         String   @unique          // Auto-derived: "measurements"
   246→  description String?                   // Developer notes
   247→  // JSON array of field IDs included in this tab. Can reference:
   248→  //   - GlobalCustomField IDs (custom fields)
   249→  //   - Built-in field keys (e.g. "firstName", "allergies", "conditions")
   250→  fields      String   @default("[]")  // JSON: ["fieldId1","fieldId2","allergies"]
   251→  // Which module this tab appears in (e.g. 'resident', 'products', 'staff')
   252→  module      String   @default("resident")
   253→  // If true, this tab supports version history — each save creates a snapshot
   254→  // of all field values, so the user can track changes over time (e.g.
   255→  // body measurements, health metrics, etc.)
   256→  enableVersioning Boolean @default(false)
   257→  // Which business type(s) this tab belongs to. Null = all business types.
   258→  businessTypes String?
   259→  sortOrder   Int      @default(0)
   260→  active      Boolean  @default(true)
   261→  createdAt   DateTime @default(now())
   262→  updatedAt   DateTime @updatedAt
   263→
   264→  orgSelections OrgCustomTab[]
   265→}
   266→
   267→model OrgCustomTab {
   268→  id              String   @id @default(cuid())
   269→  orgId           String
   270→  globalTabId     String
   271→  globalTab       GlobalCustomTab @relation(fields: [globalTabId], references: [id], onDelete: Cascade)
   272→  labelOverride   String?  // Org-specific tab name
   273→  enabled         Boolean  @default(true) // Show/hide for this org
   274→  sortOrderOverride Int?
   275→  createdAt       DateTime @default(now())
   276→  updatedAt       DateTime @updatedAt
   277→
   278→  @@unique([orgId, globalTabId])
   279→}

   281→// ============== CUSTOM FIELD VALUE VERSIONS ==============
   282→// Stores snapshots of custom field values at a point in time — like a
   283→// "measurement history". Each time a user records new measurements (e.g.
   284→// body measurements for a tailor), a new version is created instead of
   285→// overwriting the old values. The latest version's values are also kept
   286→// in CustomFieldValue (the "current" values) for quick display in forms.
   287→//
   288→// Example: A tailor takes measurements on Jan 1 (version 1), then again
   289→// on Mar 1 (version 2). The customer detail shows the latest (Mar 1),
   290→// but the user can view the history to see how measurements changed.
   291→
   292→model CustomFieldValueVersion {
   293→  id          String   @id @default(cuid())
   294→  entityId    String                    // the entity (e.g. resident ID)
   295→  entityType  String   @default("resident")
   296→  // JSON snapshot of all field values at this point: { "fieldId1": "value1", "fieldId2": "value2" }
   297→  values      String   @default("{}")  // JSON object
   298→  // Optional label for this version, e.g. "Initial", "After diet", "3-month checkup"
   299→  label       String?
   300→  recordedAt  DateTime @default(now())  // when the measurements were taken
   301→  recordedBy  String?                   // user ID who recorded this version
   302→  recordedByName String?                // user name (denormalized for display)
   303→  createdAt   DateTime @default(now())
   304→
   305→  @@index([entityId, entityType, recordedAt])
   306→}

   346→  @@unique([entityId, fieldId])

   851→// ============== JOURNAL ENTRY ==============

   859→model JournalEntry {

   871→  
   872→  // Links to source documents (optional — set when source is AUTO_*)

  1034→// A purchase order (PO) is a document sent to a vendor requesting supply of
  1035→// goods/services. Each PO has a vendor, an order date, line items, and a
  1036→// status (DRAFT, SUBMITTED, RECEIVED, CANCELLED). When a PO is received,
  1037→// stock is incremented (InventoryTransaction) and a journal entry is auto-
  1038→// posted to the GL (Dr Inventory / Cr Cash or Accounts Payable).
  1039→model PurchaseOrder {
  1040→  id              String   @id @default(cuid())
  1041→  facilityId      String?
  1042→  facility        Facility? @relation(fields: [facilityId], references: [id])
  1043→  poNumber        String   @unique  // PO-0001 format
  1044→  vendorId        String?
  1045→  vendor          Vendor?  @relation(fields: [vendorId], references: [id])
  1046→  orderDate       DateTime @default(now())
  1047→  expectedDate    DateTime?
  1048→  receivedDate    DateTime?
  1049→  status          String   @default("DRAFT")  // DRAFT, SUBMITTED, RECEIVED, CANCELLED
  1050→  paymentStatus   String   @default("UNPAID") // UNPAID, PARTIAL, PAID
  1051→  paymentMethod   String?  // CASH, BANK_TRANSFER, CHEQUE, CARD, ONLINE, CREDIT
  1052→  subtotal        Float    @default(0)
  1053→  tax             Float    @default(0)
  1054→  total           Float    @default(0)
  1055→  paidAmount      Float    @default(0)
  1056→  notes           String?
  1057→  journalEntryId  String?  // link to the JE that posted this PO (mirrors Deposit pattern)
  1058→  createdById     String?
  1059→  createdByName   String?
  1060→  createdAt       DateTime @default(now())
  1061→  updatedAt       DateTime @updatedAt
  1062→
  1063→  lines           PurchaseOrderLine[]
  1064→  transactions    InventoryTransaction[]
  1065→  journalEntries  JournalEntry[]
  1066→}

  1068→// ============== PURCHASE ORDER LINE ==============
  1069→// A single line item within a purchase order. Each line can optionally link
  1070→// to an InventoryItem (so receiving updates stock) and/or a Product (for
  1071→// catalog pricing). Free-text description is always allowed.
  1072→model PurchaseOrderLine {
  1073→  id              String   @id @default(cuid())
  1074→  purchaseOrderId String
  1075→  purchaseOrder   PurchaseOrder @relation(fields: [purchaseOrderId], references: [id], onDelete: Cascade)
  1076→  itemId          String?  // link to InventoryItem (optional — set to auto-update stock on receive)
  1077→  item            InventoryItem? @relation(fields: [itemId], references: [id])
  1078→  productId       String?  // link to Product (optional — used for catalog pricing & GL account lookup)
  1079→  product         Product?  @relation(fields: [productId], references: [id])
  1080→  description     String   // display label (auto-filled from item/product name, editable)
  1081→  quantity        Float    @default(0)
  1082→  unitPrice       Float    @default(0)
  1083→  total           Float    @default(0)  // quantity * unitPrice
  1084→  receivedQty     Float    @default(0)  // tracks partial receipts
  1085→  createdAt       DateTime @default(now())
  1086→}

  1088→// ============== PRODUCT VENDOR PRICE ==============
  1089→// Tracks vendor-specific cost prices for a product. A product can be sourced
  1090→// from multiple vendors at different prices — this lets the user pick the
  1091→// cheapest / fastest vendor when creating a Purchase Order. Also captures
  1092→// minimum order quantity, lead time, and effective date range so historical
  1093→// prices remain queryable.
  1094→model ProductVendorPrice {
  1095→  id              String   @id @default(cuid())
  1096→  facilityId      String?
  1097→  facility        Facility? @relation(fields: [facilityId], references: [id])
  1098→  productId       String
  1099→  product         Product  @relation(fields: [productId], references: [id], onDelete: Cascade)
  1100→  vendorId        String
  1101→  vendor          Vendor   @relation(fields: [vendorId], references: [id], onDelete: Cascade)
  1102→  unitCost        Float    @default(0)
  1103→  minOrderQty     Float?   // minimum order quantity (e.g. must order 10 boxes)
  1104→  leadTimeDays    Int?     // typical delivery lead time in days
  1105→  effectiveFrom   DateTime @default(now())  // when this price takes effect
  1106→  effectiveTo     DateTime?  // null = currently effective; set when superseded
  1107→  notes           String?
  1108→  createdAt       DateTime @default(now())
  1109→  updatedAt       DateTime @updatedAt
  1110→  importBatchId   String?
  1111→
  1112→  @@unique([productId, vendorId])  // one active price per product+vendor pair
  1113→  @@index([productId])
  1114→  @@index([vendorId])
  1115→}

  1117→// ============== STOCK TRANSFER ==============
  1118→// Moves stock between two facilities within the same organization. The source
  1119→// facility's InventoryItem is decremented (STOCK_OUT) and the destination
  1120→// facility's matching InventoryItem is incremented (STOCK_IN). If no matching
  1121→// item exists at the destination (by name + unit), one is auto-created so the
  1122→// receiving facility can start tracking it.
  1123→model StockTransfer {
  1124→  id              String   @id @default(cuid())
  1125→  transferNumber  String   @unique  // ST-0001 format
  1126→  fromFacilityId  String
  1127→  fromFacility    Facility  @relation("StockTransferFrom", fields: [fromFacilityId], references: [id])
  1128→  toFacilityId    String
  1129→  toFacility      Facility  @relation("StockTransferTo", fields: [toFacilityId], references: [id])
  1130→  status          String   @default("DRAFT")  // DRAFT, IN_TRANSIT, RECEIVED, CANCELLED
  1131→  transferDate    DateTime @default(now())
  1132→  receivedDate    DateTime?
  1133→  notes           String?
  1134→  createdById     String?
  1135→  createdByName   String?
  1136→  createdAt       DateTime @default(now())
  1137→  updatedAt       DateTime @updatedAt
  1138→
  1139→  lines           StockTransferLine[]
  1140→  transactions    InventoryTransaction[]
  1141→}

  1143→model StockTransferLine {
  1144→  id              String   @id @default(cuid())
  1145→  stockTransferId String

  1147→  itemId          String   // source facility's InventoryItem

  1149→  quantity        Float    @default(0)
  1150→  receivedQty     Float    @default(0)  // tracks partial receipts
  1151→  // Snapshot of source item at transfer time (so the receiving facility knows what was sent)
  1152→  itemName        String
  1153→  itemSku         String?
  1154→  itemCategory    String   @default("OTHER")

  1156→  itemUnitCost    Float    @default(0)

  1160→  createdAt        DateTime @default(now())
