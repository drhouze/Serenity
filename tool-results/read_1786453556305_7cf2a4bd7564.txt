   720→/**
   721→ * Displays a REFERENCE field value — fetches the referenced entity and shows
   722→ * its name instead of just the raw ID.
   723→ */
   724→function ReferenceFieldValue({ field, value }: { field: any; value: string }) {
   725→  const refEntity = field.referenceEntity
   726→  const fetchUrl = getReferenceFetchUrl(refEntity)
   727→  const { data: entities } = useFetch<any[]>(fetchUrl)
   728→  const entity = (entities || []).find(e => e.id === value)
   729→
   730→  if (!entity) {
   731→    // Entity not found (may have been deleted) — show the raw ID
   732→    return <span className="text-muted-foreground text-xs">{value.slice(0, 8)}…</span>
   733→  }
   734→
   735→  const name = getEntityDisplayName(entity, refEntity)
   736→  return <Badge variant="outline" className="text-blue-700 border-blue-300">{name}</Badge>
   737→}
   738→
   739→/** Returns the API URL to fetch entities of the referenced type. */
   740→function getReferenceFetchUrl(refEntity: string | null | undefined): string | null {
   741→  if (!refEntity) return null
   742→  switch (refEntity) {
   743→    case 'product':
   744→      return '/api/data?type=products'
   745→    case 'staff':
   746→      return '/api/data?type=staff'
   747→    case 'resident':
   748→      return '/api/data?type=residents'
   749→    case 'invoice':
   750→      return '/api/data?type=invoices'
   751→    default:
   752→      return null
   753→  }
   754→}
   755→
   756→/** Returns a human-readable display name for a referenced entity. */
   757→function getEntityDisplayName(e: any, refEntity: string): string {
   758→  switch (refEntity) {
   759→    case 'product':
   760→      return e.name || e.id
   761→    case 'staff':
   762→      return `${e.firstName || ''} ${e.lastName || ''}`.trim() || e.name || e.id
   763→    case 'resident':
   764→      return `${e.firstName || ''} ${e.lastName || ''}`.trim() || e.id
   765→    case 'invoice':
   766→      return e.invoiceNumber || e.id
   767→    default:
   768→      return e.name || e.id
   769→  }
   770→}

   798→  }
