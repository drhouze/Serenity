    35→function getWeekStart(date: Date): Date {
    36→  const d = new Date(date)
    37→  const day = d.getDay()
    38→  const diff = day === 0 ? -6 : 1 - day // Monday as start of week
    39→  d.setDate(d.getDate() + diff)
    40→  d.setHours(0, 0, 0, 0)
    41→  return d
    42→}
    43→
    44→// Smart default: if today is weekend (Sat/Sun), show next week (upcoming schedule)
    45→function getInitialWeekStart(): Date {
    46→  const today = new Date()
    47→  const day = today.getDay()
    48→  if (day === 6 || day === 0) {
    49→    // Saturday or Sunday — jump to next Monday
    50→    const nextMonday = new Date(today)
    51→    const diff = day === 0 ? 1 : 2 // Sunday → +1, Saturday → +2
    52→    nextMonday.setDate(nextMonday.getDate() + diff)
    53→    nextMonday.setHours(0, 0, 0, 0)
    54→    return nextMonday
    55→  }
    56→  return getWeekStart(today)
    57→}
    58→function addDays(date: Date, days: number): Date {
    59→  const d = new Date(date); d.setDate(d.getDate() + days); return d
    60→}
    61→
    62→// Timezone-safe date key: extracts YYYY-MM-DD from a Date without UTC conversion issues.
    63→// Uses the date's own year/month/day (local) to build the key, avoiding toISOString() which shifts.
    64→function dateKey(d: Date): string {
    65→  const y = d.getFullYear()
    66→  const m = String(d.getMonth() + 1).padStart(2, '0')
    67→  const day = String(d.getDate()).padStart(2, '0')
    68→  return `${y}-${m}-${day}`
    69→}
    70→function sameDay(a: Date, b: Date): boolean {
    71→  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
    72→}
    73→function isOnLeave(staffId: string, date: Date, leaves: any[]): any | null {
    74→  for (const l of leaves) {
    75→    if (l.staffId === staffId && l.status === 'APPROVED') {
    76→      const start = new Date(l.startDate); start.setHours(0, 0, 0, 0)
    77→      const end = new Date(l.endDate); end.setHours(23, 59, 59, 999)
    78→      if (date >= start && date <= end) return l
    79→    }
    80→  }
    81→  return null
    82→}

    86→  return r.json()

   351→    </div>
   352→  )
   353→}

   356→function PrintSchedule({ weekStart, shifts, staff, orgName, settings, onClose }: { weekStart: Date; shifts: any[]; staff: any[]; orgName: string; settings?: any; onClose: () => void }) {

   363→  }
   364→  for (const s of shifts) {
   365→    const key = dateKey(new Date(s.date))
   366→    if (shiftsByDay[key]) shiftsByDay[key].push(s)
   367→  }
   368→  for (const key of Object.keys(shiftsByDay)) shiftsByDay[key].sort((a, b) => a.startTime.localeCompare(b.startTime))
   369→
   370→  // Org contact info from settings
   371→  const orgPhone = settings?.organizationPhone || settings?.facilityPhone || ''
   372→  const orgEmail = settings?.organizationEmail || settings?.facilityEmail || ''
   373→  const orgAddress = settings?.organizationAddress || ''

   375→  const orgCity = settings?.organizationCity || ''
   376→  const orgState = settings?.organizationState || ''
   377→  const orgPostal = settings?.organizationPostalCode || ''
   378→  const orgCountry = settings?.organizationCountry || ''

   482→// ============ LEAVE VIEW ============
   483→function LeaveView({ staff, facilityId }: { staff: any[]; facilityId?: string }) {
   484→  const { leaveTypes } = useAppDropdowns(facilityId)
   485→  const [showRequest, setShowRequest] = useState(false)
   486→  const [filter, setFilter] = useState('PENDING')
   487→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   488→  const { data: leaves, loading, refetch } = useFetch<any[]>(`/api/data?type=leaves${filter !== 'ALL' ? `&status=${filter}` : ''}${facilityParam}`)
   489→  const { data: currentUser } = useFetch<any>('/api/auth/me')
   490→  const [refreshKey, setRefreshKey] = useState(0)
   491→  const triggerRefresh = useCallback(() => { setRefreshKey(k => k + 1); refetch() }, [refetch])
   492→
   493→  if (loading) return <Skeleton className="h-96" />
   494→
   495→  const isManager = currentUser?.user?.role === 'OWNER' || currentUser?.user?.role === 'MANAGER'
   496→
   497→  const handleApprove = async (leave: any, approved: boolean) => {
   498→    // Pre-confirm: if approving, warn about any existing shifts that will be auto-deleted
   499→    if (approved) {
   500→      try {
   501→        // Quick check via the data API for existing shifts for this staff in the leave range
   502→        const leaveStart = new Date(leave.startDate)
   503→        const leaveEnd = new Date(leave.endDate)
   504→        // We can't easily query by date range from the frontend list endpoint, so we just
   505→        // confirm with the user that any existing shifts will be removed.
   506→        const confirmed = confirm(
   507→          `Approve ${leave.type.toLowerCase()} leave for ${leave.staff?.firstName || ''} ${leave.staff?.lastName || ''} from ${leaveStart.toDateString()} to ${leaveEnd.toDateString()}?\n\n` +
   508→          `⚠ Any existing shifts for this staff during the leave period will be automatically removed to prevent conflicts.`
   509→        )
   510→        if (!confirmed) return
   511→      } catch {
   512→        // If anything fails, fall through to the actual API call
   513→      }
   514→    }
   515→    try {
   516→      const r = await fetch(`/api/data?type=leaves&id=${leave.id}`, {
   517→        method: 'PATCH',
   518→        headers: { 'Content-Type': 'application/json' },
   519→        body: JSON.stringify({
   520→          status: approved ? 'APPROVED' : 'REJECTED',
   521→          reviewedById: currentUser?.user?.id,
   522→          reviewedByName: currentUser?.user?.name,
   523→          reviewedAt: new Date().toISOString(),
   524→        }),
   525→      }).then(r => r.json())
   526→      if (!r || r.error) {
   527→        toast.error(r?.error || 'Failed to update leave')
   528→        return
   529→      }
   530→      if (approved && r.autoDeletedShifts > 0) {
   531→        toast.success(`Leave approved — ${r.autoDeletedShifts} conflicting shift(s) auto-removed`)
   532→      } else {
   533→        toast.success(`Leave ${approved ? 'approved' : 'rejected'}`)
   534→      }
   535→      triggerRefresh()
   536→    } catch (e: any) { toast.error(e.message) }
   537→  }
   538→
   539→  return (
   540→    <div className="space-y-4">
   541→      <div className="flex justify-between items-center">
   542→        <div className="flex gap-1">
   543→          {['PENDING', 'APPROVED', 'REJECTED', 'ALL'].map(s => (
   544→            <button key={s} onClick={() => setFilter(s)} className={`px-2.5 py-1 rounded-full text-xs font-medium border ${filter === s ? 'bg-primary text-primary-foreground border-primary' : 'bg-background border-border'}`}>{s}</button>
   545→          ))}
   546→        </div>
   547→        <Button size="sm" onClick={() => setShowRequest(true)}><Plane className="h-3 w-3 mr-1" /> Request Leave</Button>
   548→      </div>
   549→
   550→      <div className="grid gap-2">
   551→        {(leaves || []).length === 0 && <p className="text-sm text-muted-foreground text-center py-8">No leave requests</p>}
   552→        {(leaves || []).map(l => (
   553→          <Card key={l.id}>
   554→            <CardContent className="p-3">
   555→              <div className="flex items-start justify-between gap-2 flex-wrap">
   556→                <div className="flex-1 min-w-0">
   557→                  <div className="flex items-center gap-2 flex-wrap">
   558→                    <Badge variant="outline" className="text-xs">{l.type}</Badge>
   559→                    <StatusBadge status={l.status} />
   560→                    <span className="font-medium text-sm">{l.staff?.firstName} {l.staff?.lastName}</span>
   561→                    {l.staff?.code && <span className="text-xs font-mono text-primary">{l.staff.code}</span>}
   562→                  </div>
   563→                  <div className="text-xs text-muted-foreground mt-1">
   564→                    {fmtDate(l.startDate)} – {fmtDate(l.endDate)}
   565→                    <span className="ml-2">({Math.ceil((new Date(l.endDate).getTime() - new Date(l.startDate).getTime()) / 86400000) + 1} days)</span>
   566→                  </div>
   567→                  {l.reason && <p className="text-sm mt-1 italic">"{l.reason}"</p>}
   568→                  {l.reviewedByName && <p className="text-xs text-muted-foreground mt-1">Reviewed by {l.reviewedByName} on {l.reviewedAt ? fmtDate(l.reviewedAt) : '—'}</p>}
   569→                  {l.reviewNotes && <p className="text-xs text-muted-foreground mt-0.5">Notes: {l.reviewNotes}</p>}
   570→                </div>
   571→                {l.status === 'PENDING' && isManager && (
   572→                  <div className="flex gap-1 flex-shrink-0">
   573→                    <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => handleApprove(l, true)}><Check className="h-3 w-3 mr-1" /> Approve</Button>
   574→                    <Button size="sm" variant="outline" className="text-red-600 border-red-200" onClick={() => handleApprove(l, false)}><XCircle className="h-3 w-3 mr-1" /> Reject</Button>
   575→                  </div>
   576→                )}
   577→              </div>
   578→            </CardContent>
   579→          </Card>
   580→        ))}
   581→      </div>
   582→
   583→      {showRequest && <RequestLeaveDialog staff={staff} onClose={() => setShowRequest(false)} onSaved={() => { setShowRequest(false); triggerRefresh() }} />}
   584→    </div>
   585→  )

  1207→/**
  1208→ * StaffCustomTabView — renders a custom tab's fields for a staff member.
  1209→ * Similar to CustomTabView in Residents.tsx but for the staff entity.
  1210→ */
  1211→function StaffCustomTabView({ tabId, staff, orgId }: { tabId: string; staff: any; orgId?: string }) {
  1212→  const { data: orgCustomTabs } = useFetch<any[]>(orgId ? `/api/org-custom-tabs?orgId=${orgId}&module=staff` : null)
  1213→  const tabDef = (orgCustomTabs || []).find(t => t.globalTabId === tabId)
  1214→  const { data: customFields } = useFetch<any[]>(orgId ? `/api/custom-fields?orgId=${orgId}` : null)
  1215→  const { data: customValues } = useFetch<any[]>(staff?.id ? `/api/custom-field-values?entityId=${staff.id}&entityType=staff` : null)
  1216→
  1217→  if (!tabDef) {
  1218→    return <div className="text-center py-8 text-sm text-muted-foreground">Tab not found.</div>
  1219→  }
  1220→
  1221→  const fieldIds: string[] = JSON.parse(tabDef.fields || '[]')
  1222→  const valueByFieldId: Record<string, string> = {}
  1223→  for (const v of customValues || []) {
  1224→    valueByFieldId[v.fieldId] = v.value || ''
  1225→  }
  1226→
  1227→  const BUILTIN_LABELS: Record<string, string> = {
  1228→    firstName: 'First Name', lastName: 'Last Name', role: 'Role', phone: 'Phone',
  1229→    email: 'Email', hireDate: 'Hire Date', active: 'Status', notes: 'Notes',
  1230→  }
  1231→
  1232→  return (
  1233→    <div className="space-y-3">
  1234→      <div className="text-xs font-semibold text-muted-foreground">{tabDef.label.toUpperCase()}</div>
  1235→      {tabDef.description && <p className="text-xs text-muted-foreground -mt-2">{tabDef.description}</p>}
  1236→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1237→        {fieldIds.map(fieldId => {
  1238→          const isBuiltin = !!BUILTIN_LABELS[fieldId]
  1239→          const cf = (customFields || []).find(f => f.id === fieldId)
  1240→          const label = isBuiltin ? BUILTIN_LABELS[fieldId] : (cf?.label || fieldId)
  1241→          const val = isBuiltin ? (staff as any)[fieldId] : valueByFieldId[fieldId]
  1242→          return (
  1243→            <div key={fieldId} className="border rounded-md p-2">
  1244→              <div className="text-[10px] uppercase tracking-wide text-muted-foreground mb-0.5">{label}</div>
  1245→              <div className="font-medium text-sm break-words">
  1246→                {val !== undefined && val !== '' && val !== null
  1247→                  ? String(val)
  1248→                  : <span className="text-muted-foreground/60">—</span>}
  1249→              </div>
  1250→            </div>
  1251→          )
  1252→        })}
  1253→      </div>
  1254→    </div>
  1255→  )
  1256→}
