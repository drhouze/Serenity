   267→function ResidentsList({ facilityId, search, setSearch, onSelect }: { facilityId?: string; search: string; setSearch: (s: string) => void; onSelect: (id: string) => void }) {
   268→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   269→  const { data, loading } = useFetch<any[]>(`/api/data?type=residents${facilityParam}`)
   270→  const { data: medAdmins } = useFetch<any[]>(`/api/data?type=medAdmins&today=true${facilityParam}`)
   271→  const { data: incidents } = useFetch<any[]>(`/api/data?type=incidents${facilityParam}`)
   272→  const { residentStatuses } = useAppDropdowns(facilityId)
   273→  const [showStatusDialog, setShowStatusDialog] = useState<{ resident: any } | null>(null)
   274→
   275→  if (loading) return <div className="space-y-2">{Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-16" />)}</div>
   276→
   277→  const todayIncidents = (incidents || []).filter(i => new Date(i.occurredAt).toDateString() === new Date().toDateString())
   278→
   279→  const filtered = (data || []).filter(r => {
   280→    if (!search) return true
   281→    const s = search.toLowerCase()
   282→    return `${r.firstName} ${r.lastName}`.toLowerCase().includes(s) || r.room?.roomNumber?.includes(s) || r.code?.toLowerCase().includes(s)
   283→  })
   284→
   285→  return (
   286→    <div className="space-y-3">
   287→      <div className="relative">
   288→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   289→        <Input placeholder="Search customer or room..." value={search} onChange={e => setSearch(e.target.value)} className="pl-9 h-11" />
   290→      </div>
   291→      <div className="text-xs text-muted-foreground px-1">{filtered.length} customers</div>
   292→      <div className="space-y-2">
   293→        {filtered.map(r => {
   294→          // Count pending meds for this resident today
   295→          const pendingCount = (medAdmins || []).filter(a => a.residentId === r.id && a.status === 'PENDING').length
   296→          // Count today's incidents for this resident
   297→          const incidentCount = todayIncidents.filter(i => i.residentId === r.id).length
   298→
   299→          return (
   300→            <Card key={r.id} className="active:scale-[0.98] transition-transform">
   301→              <CardContent className="p-3">
   302→                <div className="flex items-center gap-3">
   303→                  <Avatar className="h-11 w-11 flex-shrink-0">
   304→                    <AvatarFallback className="bg-emerald-100 text-emerald-700">{initials(r.firstName, r.lastName)}</AvatarFallback>
   305→                  </Avatar>
   306→                  <button onClick={() => onSelect(r.id)} className="flex-1 min-w-0 text-left">
   307→                    <div className="font-medium truncate">
   308→                      {r.code && <span className="text-xs font-mono text-primary mr-1">{r.code}</span>}
   309→                      {r.firstName} {r.lastName}
   310→                    </div>
   311→                    <div className="text-xs text-muted-foreground flex flex-wrap items-center gap-x-2 gap-y-1">
   312→                      <span className="flex items-center gap-0.5 whitespace-nowrap"><BedDouble className="h-3 w-3" /> {r.room?.roomNumber || '—'}</span>
   313→                      {r.dateOfBirth && <span className="whitespace-nowrap">• {age(r.dateOfBirth)}y</span>}
   314→                    </div>
   315→                    {r.allergies && r.allergies !== 'None' && (
   316→                      <div className="text-xs text-red-600 truncate mt-0.5">⚠ {r.allergies}</div>
   317→                    )}
   318→                  </button>
   319→                  <div className="flex flex-col items-end gap-1 flex-shrink-0 max-w-[40%]">
   320→                    {/* Pending meds badge */}
   321→                    {pendingCount > 0 && (
   322→                      <Badge className="bg-rose-100 text-rose-700 text-[10px] flex items-center gap-0.5 whitespace-nowrap">
   323→                        <Pill className="h-2.5 w-2.5" /> {pendingCount} meds due
   324→                      </Badge>
   325→                    )}
   326→                    {/* Today's incident badge */}
   327→                    {incidentCount > 0 && (
   328→                      <Badge className="bg-amber-100 text-amber-700 text-[10px] flex items-center gap-0.5 whitespace-nowrap">
   329→                        <AlertTriangle className="h-2.5 w-2.5" /> {incidentCount} incident{incidentCount > 1 ? 's' : ''}
   330→                      </Badge>

   332→                    {/* Status badge */}
   333→                    {r.status !== 'ACTIVE' && (
   334→                      <Badge variant="outline" className="text-[10px] whitespace-nowrap">{r.status.replace(/_/g, ' ')}</Badge>
   335→                    )}
   336→                  </div>
   337→                </div>
   338→                {/* Quick status change button */}
   339→                <div className="mt-2 pt-2 border-t flex items-center justify-between gap-2">
   340→                  <button
   341→                    onClick={() => setShowStatusDialog({ resident: r })}
   342→                    className="text-[10px] text-muted-foreground hover:text-primary flex items-center gap-0.5 min-w-0 truncate"
   343→                  >
   344→                    <User className="h-2.5 w-2.5 flex-shrink-0" /> <span className="truncate">Status: {r.status?.replace(/_/g, ' ').toLowerCase()} → change</span>
   345→                  </button>
   346→                  <button
   347→                    onClick={() => onSelect(r.id)}
   348→                    className="text-[10px] text-primary flex items-center gap-0.5 whitespace-nowrap flex-shrink-0"
   349→                  >
   350→                    View rounds <ChevronRight className="h-2.5 w-2.5" />
   351→                  </button>
   352→                </div>
   353→              </CardContent>
   354→            </Card>
   355→          )
   356→        })}
   357→      </div>
   358→      {showStatusDialog && (
   359→        <QuickStatusChange
   360→          resident={showStatusDialog.resident}
   361→          residentStatuses={residentStatuses}
   362→          onClose={() => setShowStatusDialog(null)}
   363→          onSaved={() => { setShowStatusDialog(null); window.location.reload() }}
   364→        />
   365→      )}
   366→    </div>
   367→  )
   368→}

   370→// ============ QUICK STATUS CHANGE ============
   371→function QuickStatusChange({ resident, residentStatuses, onClose, onSaved }: any) {
   372→  const [status, setStatus] = useState(resident.status || 'ACTIVE')
   373→  const [reason, setReason] = useState('')
   374→  const [saving, setSaving] = useState(false)
   375→
   376→  const submit = async () => {
   377→    setSaving(true)
   378→    try {
   379→      // Update resident status
   380→      await apiPatch(`/api/data?type=residents&id=${resident.id}`, { status })
   381→      // Create a status log entry
   382→      await apiPost('/api/data?type=statusLogs', {
   383→        residentId: resident.id,
   384→        fromStatus: resident.status,
   385→        toStatus: status,
   386→        reason: reason || null,
   387→      })
   388→      toast.success(`${resident.firstName}'s status changed to ${status.replace(/_/g, ' ')}`)
   389→      onSaved()
   390→    } catch (e: any) { toast.error(e.message) }
   391→    setSaving(false)
   392→  }
   393→
   394→  return (
   395→    <div className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center p-4">
   396→      <Card className="w-full max-w-sm">
   397→        <CardContent className="p-4 space-y-3">
   398→          <div className="flex items-center justify-between">
   399→            <h3 className="font-semibold text-sm">Change Status</h3>
   400→            <Button variant="ghost" size="sm" onClick={onClose}>×</Button>
   401→          </div>
   402→          <div className="text-xs text-muted-foreground">
   403→            {resident.firstName} {resident.lastName} ({resident.code})
