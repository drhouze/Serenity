    26→    26→export function Residents({ initialId, onBack, facilityId }: { initialId?: string | null; onBack?: () => void; facilityId?: string }) {
    27→    27→  const [search, setSearch] = useState('')
    28→    28→  const [selectedId, setSelectedId] = useState<string | null>(initialId || null)
    29→    29→  const [showAdd, setShowAdd] = useState(false)
    30→    30→  const [showArchived, setShowArchived] = useState(false)
    31→    31→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
    32→    32→  const [refreshKey, setRefreshKey] = useState(0)
    33→    33→  const { data: currentUser } = useFetch<any>('/api/auth/me')
    34→    34→  const { data: settingsData } = useFetch<any>('/api/settings')
    35→    35→
    36→    36→  // Resolve the current user's org business type for field visibility

    38→    38→  const businessType = (userOrgId && settingsData?.[`businessType:${userOrgId}`]) || 'nursing_home'
    39→    39→
    40→    40→  const isFamily = currentUser?.user?.role === 'FAMILY'
    41→    41→
    42→    42→  const triggerRefresh = useCallback(() => setRefreshKey(k => k + 1), [])

    49→    49→    const action = archive ? 'Archiving' : 'Restoring'
    50→    50→    const done = archive ? 'archived' : 'restored'
    51→    51→    toast.info(`${action} ${ids.length} resident${ids.length > 1 ? 's' : ''}...`)
    52→    52→    let success = 0
    53→    53→    let failed = 0

    55→    55→      try {
    56→    56→        if (archive) {
    57→    57→          await apiPatch(`/api/data?type=residents&id=${id}`, { status: 'DISCHARGED', dischargeDate: new Date().toISOString() })
    58→    58→        } else {
    59→    59→          await apiPatch(`/api/data?type=residents&id=${id}`, { status: 'ACTIVE', dischargeDate: null })
    60→    60→        }

    62→    62→      } catch {
    63→    63→        failed++
    64→    64→      }
    65→    65→    }

    69→    69→    triggerRefresh()
    70→    70→  }

    72→    72→  const hasSelection = selectedIds.size > 0

    83→    83→          {!isFamily && (
    84→    84→            <Button variant="outline" onClick={() => setShowArchived(!showArchived)}>

    90→    90→              <Plus className="h-4 w-4 mr-1" /> Add Resident
    91→    91→            </Button>
    92→    92→          )}
    93→    93→        </div>
    94→    94→      </div>
    95→    95→
    96→    96→      {/* Bulk action bar — hidden for family */}
    97→    97→      {hasSelection && !isFamily && (
    98→    98→        <div className="sticky top-14 z-20 flex items-center justify-between gap-2 bg-primary text-primary-foreground rounded-lg p-2 px-4 shadow-md">
    99→    99→          <div className="flex items-center gap-3 text-sm">
   100→   100→            <button onClick={() => setSelectedIds(new Set())} className="hover:opacity-80">
   101→   101→              <X className="h-4 w-4" />

   108→   108→                <ArchiveRestore className="h-3 w-3 mr-1" /> Restore
   109→   109→              </Button>
   110→   110→            ) : (
   111→   111→              <Button size="sm" variant="secondary" onClick={() => {
   112→   112→                if (confirm(`Archive ${selectedIds.size} resident(s)? They will be marked as discharged and hidden from the active list.`)) {

   118→   118→            )}
   119→   119→          </div>
   120→   120→        </div>
   121→   121→      )}
   122→   122→
   123→   123→      <ResidentList
   124→   124→        key={refreshKey}
   125→   125→        search={search}
   126→   126→        onSelect={setSelectedId}
   127→   127→        showArchived={showArchived}
   128→   128→        selectedIds={selectedIds}
   129→   129→        onSelectionChange={setSelectedIds}
   130→   130→        isFamily={isFamily}
   131→   131→        facilityId={facilityId}
   132→   132→      />
   133→   133→
   134→   134→      {showAdd && <AddResidentDialog facilityId={facilityId} businessType={businessType} onClose={() => setShowAdd(false)} onCreated={(id) => { setShowAdd(false); setSelectedId(id) }} />}
   135→   135→    </div>
   136→   136→  )
   137→   137→}

   140→   140→function ResidentList({
   141→   141→  search,
   142→   142→  onSelect,
   143→   143→  showArchived,
   144→   144→  selectedIds,
   145→   145→  onSelectionChange,
   146→   146→  isFamily,
   147→   147→  facilityId,
   148→   148→}: {
   149→   149→  search: string
   150→   150→  onSelect: (id: string) => void
   151→   151→  showArchived: boolean
   152→   152→  selectedIds: Set<string>
   153→   153→  onSelectionChange: (ids: Set<string>) => void
   154→   154→  isFamily?: boolean
   155→   155→  facilityId?: string
   156→   156→}) {
   157→   157→  const facilityParam = facilityId ? `&facilityId=${facilityId}` : ''
   158→   158→  const url = showArchived
   159→   159→    ? `/api/data?type=residents&includeArchived=true${facilityParam}`
   160→   160→    : `/api/data?type=residents${facilityParam}`
   161→   161→  // Auto-refresh every 60s so multiple users see each other's changes
   162→   162→  // (e.g. new admissions by other staff, status changes, discharges)
   163→   163→  const { data, loading } = useFetch<any[]>(url, { refreshInterval: 60000 })
   164→   164→  if (loading) return <div className="grid gap-2">{Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-20" />)}</div>
   165→   165→  if (!data) return null
   166→   166→
   167→   167→  const filtered = data.filter(r => {
   168→   168→    if (!search) return true
   169→   169→    const s = search.toLowerCase()
   170→   170→    return (
   171→   171→      `${r.firstName} ${r.lastName}`.toLowerCase().includes(s) ||
   172→   172→      r.room?.roomNumber?.toLowerCase().includes(s) ||
   173→   173→      (r.conditions || '').toLowerCase().includes(s) ||
   174→   174→      (r.allergies || '').toLowerCase().includes(s)
   175→   175→    )
   176→   176→  })
   177→   177→
   178→   178→  const allSelected = filtered.length > 0 && filtered.every(r => selectedIds.has(r.id))
   179→   179→  const someSelected = filtered.some(r => selectedIds.has(r.id))
   180→   180→
   181→   181→  const toggleAll = () => {
   182→   182→    if (allSelected) {
   183→   183→      const next = new Set(selectedIds)
   184→   184→      filtered.forEach(r => next.delete(r.id))
   185→   185→      onSelectionChange(next)
   186→   186→    } else {
   187→   187→      const next = new Set(selectedIds)
   188→   188→      filtered.forEach(r => next.add(r.id))
   189→   189→      onSelectionChange(next)
   190→   190→    }
   191→   191→  }
   192→   192→
   193→   193→  const toggleOne = (id: string) => {
   194→   194→    const next = new Set(selectedIds)
   195→   195→    if (next.has(id)) next.delete(id)
   196→   196→    else next.add(id)
   197→   197→    onSelectionChange(next)
   198→   198→  }
   199→   199→
   200→   200→  return (
   201→   201→    <div className="grid gap-2">
   202→   202→      {/* Select-all header — hidden for family */}
   203→   203→      {filtered.length > 0 && !isFamily && (
   204→   204→        <div className="flex flex-wrap items-center gap-2 px-3 py-1.5 text-xs text-muted-foreground">
   205→   205→          <Checkbox
   206→   206→            checked={allSelected ? true : someSelected ? 'indeterminate' : false}
   207→   207→            onCheckedChange={toggleAll}
   208→   208→          />
   209→   209→          <span className="whitespace-nowrap">
   210→   210→            {filtered.length} resident{filtered.length !== 1 ? 's' : ''}
   211→   211→            {selectedIds.size > 0 && ` • ${selectedIds.size} selected`}
   212→   212→          </span>
   213→   213→        </div>
   214→   214→      )}
   215→   215→      {filtered.length > 0 && isFamily && (
   216→   216→        <div className="px-3 py-1.5 text-xs text-muted-foreground">
   217→   217→          {filtered.length} resident{filtered.length !== 1 ? 's' : ''} — click to view details
   218→   218→        </div>
   219→   219→      )}
   220→   220→      {filtered.map(r => {
   221→   221→        const isSelected = selectedIds.has(r.id)
   222→   222→        const isArchived = r.status !== 'ACTIVE'
   223→   223→        return (
   224→   224→          <div key={r.id} className={`flex items-center gap-2 ${isSelected ? 'opacity-90' : ''}`}>
   225→   225→            {!isFamily && (
   226→   226→            <div className="flex-shrink-0 pl-3">
   227→   227→              <Checkbox
   228→   228→                checked={isSelected}
   229→   229→                onCheckedChange={() => toggleOne(r.id)}
   230→   230→              />
   231→   231→            </div>
   232→   232→            )}
   233→   233→            <button
   234→   234→              onClick={() => onSelect(r.id)}
   235→   235→              className="text-left transition-shadow hover:shadow-md rounded-lg flex-1"
   236→   236→            >
   237→   237→              <Card className={`hover:bg-muted/30 ${isArchived ? 'opacity-60' : ''} ${isSelected ? 'ring-2 ring-primary' : ''}`}>
   238→   238→                <CardContent className="p-3 flex items-center gap-3">
   239→   239→                  <Avatar className="h-12 w-12 flex-shrink-0">
   240→   240→                    <AvatarFallback className="bg-emerald-100 text-emerald-700">
   241→   241→                      {initials(r.firstName, r.lastName)}
   242→   242→                    </AvatarFallback>
   243→   243→                  </Avatar>
   244→   244→                  <div className="flex-1 min-w-0">
   245→   245→                    <div className="flex items-center gap-2 flex-wrap">
   246→   246→                      {r.code && <Badge variant="outline" className="text-xs font-mono bg-primary/5 text-primary">{r.code}</Badge>}
   247→   247→                      <span className="font-semibold truncate">{r.firstName} {r.lastName}</span>
   248→   248→                      {r.dateOfBirth && <span className="text-xs text-muted-foreground">• {age(r.dateOfBirth)}y</span>}
   249→   249→                      <Badge variant="outline" className="text-xs">{r.gender || '—'}</Badge>
   250→   250→                    </div>
   251→   251→                    <div className="text-xs text-muted-foreground mt-0.5 flex flex-wrap items-center gap-x-2 gap-y-1">
   252→   252→                      <span className="flex items-center gap-1 whitespace-nowrap"><BedDouble className="h-3 w-3" /> Room {r.room?.roomNumber || 'Unassigned'}</span>
   253→   253→                      <span className="flex items-center gap-1 whitespace-nowrap"><Heart className="h-3 w-3" /> {r.conditions?.split(',')[0] || 'No conditions'}</span>
   254→   254→                      {r.allergies && r.allergies !== 'None' && (
   255→   255→                        <span className="flex items-center gap-1 text-red-600 whitespace-nowrap"><AlertCircle className="h-3 w-3" /> {r.allergies}</span>
   256→   256→                      )}
   257→   257→                    </div>
   258→   258→                  </div>
   259→   259→                  <div className="flex items-center gap-2">
   260→   260→                    <StatusBadge status={r.status} />
   261→   261→                    <ChevronRight className="h-4 w-4 text-muted-foreground" />
   262→   262→                  </div>
   263→   263→                </CardContent>
   264→   264→              </Card>
   265→   265→            </button>
   266→   266→          </div>
   267→   267→        )
   268→   268→      })}
   269→   269→      {filtered.length === 0 && <p className="text-center text-sm text-muted-foreground py-8">No residents found</p>}
   270→   270→    </div>
   271→   271→  )
   272→   272→}

   274→   274→function ResidentDetail({ id, onBack }: { id: string; onBack: () => void }) {

   277→   277→  // Fetch settings locally — ResidentDetail is a separate component from Residents

   280→   280→  const { data: settingsData } = useFetch<any>('/api/settings')
   281→   281→  const userOrgId = currentUser?.user?.organizationId
   282→   282→  const businessType = (userOrgId && settingsData?.[`businessType:${userOrgId}`]) || 'nursing_home'

   384→   384→          {/* Status changer — hidden for family (read-only view) */}
   385→   385→          {canEdit && (

   721→   721→ * Displays a REFERENCE field value — fetches the referenced entity and shows

   724→   724→function ReferenceFieldValue({ field, value }: { field: any; value: string }) {
   725→   725→  const refEntity = field.referenceEntity
   726→   726→  const fetchUrl = getReferenceFetchUrl(refEntity)
   727→   727→  const { data: entities } = useFetch<any[]>(fetchUrl)
   728→   728→  const entity = (entities || []).find(e => e.id === value)
   729→   729→
   730→   730→  if (!entity) {
   731→   731→    // Entity not found (may have been deleted) — show the raw ID
   732→   732→    return <span className="text-muted-foreground text-xs">{value.slice(0, 8)}…</span>
   733→   733→  }
   734→   734→
   735→   735→  const name = getEntityDisplayName(entity, refEntity)
   736→   736→  return <Badge variant="outline" className="text-blue-700 border-blue-300">{name}</Badge>
   737→   737→}

   739→   739→/** Returns the API URL to fetch entities of the referenced type. */
   740→   740→function getReferenceFetchUrl(refEntity: string | null | undefined): string | null {
   741→   741→  if (!refEntity) return null
   742→   742→  switch (refEntity) {
   743→   743→    case 'product':
   744→   744→      return '/api/data?type=products'
   745→   745→    case 'staff':
   746→   746→      return '/api/data?type=staff'
   747→   747→    case 'resident':
   748→   748→      return '/api/data?type=residents'
   749→   749→    case 'invoice':

   952→   952→/**
   953→   953→ * RecordValuesDialog — lets the user enter a new set of values for a versioned tab.
   954→   954→ * Creates a version snapshot AND updates the current values.
   955→   955→ */
   956→   956→function RecordValuesDialog({ tabDef, fieldIds, customFields, resident, currentValues, builtinLabels, onClose, onSaved }: {
   957→   957→  tabDef: any
   958→   958→  fieldIds: string[]
   959→   959→  customFields: any[]
   960→   960→  resident: any
   961→   961→  currentValues: Record<string, string>
   962→   962→  builtinLabels: Record<string, string>
   963→   963→  onClose: () => void
   964→   964→  onSaved: () => void
   965→   965→}) {
   966→   966→  const [values, setValues] = useState<Record<string, string>>(() => {
   967→   967→    // Pre-fill with current values
   968→   968→    const init: Record<string, string> = {}
   969→   969→    for (const fid of fieldIds) {
   970→   970→      const isBuiltin = !!builtinLabels[fid]
   971→   971→      init[fid] = isBuiltin ? ((resident as any)[fid] || '') : (currentValues[fid] || '')
   972→   972→    }
   973→   973→    return init
   974→   974→  })
   975→   975→  const [label, setLabel] = useState('')
   976→   976→  const [saving, setSaving] = useState(false)
   977→   977→
   978→   978→  const submit = async () => {
   979→   979→    const entries = Object.entries(values).filter(([_, v]) => v !== '' && v != null)
   980→   980→    if (entries.length === 0) { toast.error('Enter at least one value'); return }
   981→   981→    setSaving(true)
   982→   982→    try {
   983→   983→      const valuesObj: Record<string, string> = {}
   984→   984→      for (const [fieldId, value] of entries) {
   985→   985→        valuesObj[fieldId] = value
   986→   986→      }
   987→   987→      const res = await fetch('/api/custom-field-versions', {
   988→   988→        method: 'POST',
   989→   989→        headers: { 'Content-Type': 'application/json' },
   990→   990→        body: JSON.stringify({
   991→   991→          entityId: resident.id,
   992→   992→          entityType: 'resident',
   993→   993→          values: valuesObj,
   994→   994→          label: label.trim() || undefined,
   995→   995→        }),
   996→   996→      })
   997→   997→      if (!res.ok) throw new Error((await res.json().catch(() => ({}))).error || `HTTP ${res.status}`)
   998→   998→      toast.success(label.trim() ? `Recorded "${label.trim()}"` : 'Values recorded')
   999→   999→      onSaved()
  1000→  1000→    } catch (e: any) { toast.error(e.message) }
  1001→  1001→    setSaving(false)
  1002→  1002→  }
  1003→  1003→
  1004→  1004→  return (
  1005→  1005→    <Modal title={`Record New — ${tabDef.label}`} onClose={onClose}>
  1006→  1006→      <div className="space-y-3">
  1007→  1007→        <div>
  1008→  1008→          <label className="text-xs font-medium text-muted-foreground mb-1 block">Label (optional)</label>
  1009→  1009→          <Input value={label} onChange={e => setLabel(e.target.value)} placeholder="e.g. Initial, 3-month checkup" className="text-sm" />
  1010→  1010→        </div>
  1011→  1011→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1012→  1012→          {fieldIds.map(fieldId => {
  1013→  1013→            const isBuiltin = !!builtinLabels[fieldId]
  1014→  1014→            const cf = customFields.find(f => f.id === fieldId)
  1015→  1015→            const labelName = isBuiltin ? builtinLabels[fieldId] : (cf?.label || fieldId)
  1016→  1016→            const val = values[fieldId] || ''
  1017→  1017→            const unitSuffix = cf?.unit ? ` (${cf.unit})` : ''
  1018→  1018→            // Built-in fields are read-only in versioned tabs (they don't change often)
  1019→  1019→            if (isBuiltin) {
  1020→  1020→              return (
  1021→  1021→                <div key={fieldId}>
  1022→  1022→                  <label className="text-xs font-medium text-muted-foreground mb-1 block">{labelName}{unitSuffix}</label>
  1023→  1023→                  <Input value={val} disabled className="text-sm bg-muted/50" />
  1024→  1024→                  <p className="text-[9px] text-muted-foreground">Built-in field (read-only)</p>
  1025→  1025→                </div>
  1026→  1026→              )
  1027→  1027→            }
  1028→  1028→            return (
  1029→  1029→              <div key={fieldId}>
  1030→  1030→                <label className="text-xs font-medium text-muted-foreground mb-1 block">{labelName}{unitSuffix}</label>
  1031→  1031→                {cf?.type === 'SELECT' ? (
  1032→  1032→                  <select className="w-full border rounded px-2 py-1.5 text-sm" value={val} onChange={e => setValues({ ...values, [fieldId]: e.target.value })}>
  1033→  1033→                    <option value="">—</option>
  1034→  1034→                    {(cf.options ? JSON.parse(cf.options) : []).map((opt: string) => <option key={opt} value={opt}>{opt}</option>)}
  1035→  1035→                  </select>

  1037→  1037→                  <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={val} onChange={e => setValues({ ...values, [fieldId]: e.target.value })} placeholder={labelName} />

  1039→  1039→                  <Input type={cf?.type === 'NUMBER' ? 'number' : cf?.type === 'DATE' ? 'date' : 'text'} value={val} onChange={e => setValues({ ...values, [fieldId]: e.target.value })} placeholder={labelName} className="text-sm" />
  1040→  1040→                )}
  1041→  1041→              </div>

  1046→  1046→          <Button variant="outline" onClick={onClose}>Cancel</Button>

  1048→  1048→            {saving ? 'Saving...' : 'Record Values'}
  1049→  1049→          </Button>
  1050→  1050→        </div>
  1051→  1051→      </div>
  1052→  1052→    </Modal>
  1053→  1053→  )
  1054→  1054→}

  1316→  1316→/**
  1317→  1317→ * Parses vital signs from free-text input.
  1318→  1318→ * Supports common formats:
  1319→  1319→ *   "BP 140/90, HR 72, Temp 37.2, RR 18, SpO2 98, BS 5.5, Wt 65"
  1320→  1320→ *   "Blood pressure 140/90, heart rate 72, temperature 37.2"
  1321→  1321→ *   "140/90, 72, 37.2, 18, 98"
  1322→  1322→ *
  1323→  1323→ * Returns an object with any parsed vitals, or null if nothing was parsed.
  1324→  1324→ */
  1325→  1325→function parseVitalsFromText(text: string): any | null {
  1326→  1326→  const result: any = {}
  1327→  1327→  const lower = text.toLowerCase()
  1328→  1328→
  1329→  1329→  // Blood pressure: "BP 140/90", "Blood pressure 140/90", "140/90"
  1330→  1330→  const bpMatch = lower.match(/(?:bp|blood\s*pressure)?\s*(\d{2,3})\s*\/\s*(\d{2,3})/)
  1331→  1331→  if (bpMatch) {
  1332→  1332→    result.bloodPressureSystolic = parseInt(bpMatch[1])
  1333→  1333→    result.bloodPressureDiastolic = parseInt(bpMatch[2])
  1334→  1334→  }
  1335→  1335→
  1336→  1336→  // Heart rate: "HR 72", "Heart rate 72", "pulse 72"
  1337→  1337→  const hrMatch = lower.match(/(?:hr|heart\s*rate|pulse)\s*:?\s*(\d{2,3})/)
  1338→  1338→  if (hrMatch) {
  1339→  1339→    result.heartRate = parseInt(hrMatch[1])
  1340→  1340→  }
  1341→  1341→
  1342→  1342→  // Temperature: "Temp 37.2", "Temperature 37.2", "37.2°C", "37.2C"
  1343→  1343→  const tempMatch = lower.match(/(?:temp|temperature)?\s*:?\s*(\d{2,3}(?:\.\d)?)\s*(?:°?c|°?f)?/i)
  1344→  1344→  if (tempMatch) {
  1345→  1345→    const temp = parseFloat(tempMatch[1])
  1346→  1346→    // Only accept if it looks like a temperature (30-45°C or 90-115°F)
  1347→  1347→    if (temp >= 30 && temp <= 45) {
  1348→  1348→      result.temperature = temp
  1349→  1349→    } else if (temp >= 90 && temp <= 115) {
  1350→  1350→      // Convert Fahrenheit to Celsius
  1351→  1351→      result.temperature = parseFloat(((temp - 32) * 5/9).toFixed(1))
  1352→  1352→    }
  1353→  1353→  }
  1354→  1354→
  1355→  1355→  // Respiratory rate: "RR 18", "Respiratory rate 18", "Resp 18"
  1356→  1356→  const rrMatch = lower.match(/(?:rr|resp(?:iratory)?\s*rate)\s*:?\s*(\d{2,3})/)
  1357→  1357→  if (rrMatch) {
  1358→  1358→    const rr = parseInt(rrMatch[1])
  1359→  1359→    if (rr >= 8 && rr <= 60) result.respiratoryRate = rr
  1360→  1360→  }
  1361→  1361→
  1362→  1362→  // Oxygen saturation: "SpO2 98", "O2 98", "Oxygen 98%", "Sat 98"
  1363→  1363→  const o2Match = lower.match(/(?:spo2|o2|oxygen|sat)\s*:?\s*(\d{2,3})\s*%?/)
  1364→  1364→  if (o2Match) {
  1365→  1365→    const o2 = parseInt(o2Match[1])
  1366→  1366→    if (o2 >= 50 && o2 <= 100) result.oxygenSaturation = o2
  1367→  1367→  }
  1368→  1368→
  1369→  1369→  // Blood sugar: "BS 5.5", "Blood sugar 5.5", "Glucose 5.5", "BGL 5.5"
  1370→  1370→  const bsMatch = lower.match(/(?:bs|blood\s*sugar|glucose|bgl)\s*:?\s*(\d{1,3}(?:\.\d)?)\s*(?:mmol\/l|mg\/dl)?/)
  1371→  1371→  if (bsMatch) {
  1372→  1372→    const bs = parseFloat(bsMatch[1])
  1373→  1373→    // Accept both mmol/L (3-30) and mg/dL (50-500)
  1374→  1374→    if (bs >= 3 && bs <= 30) {
  1375→  1375→      result.bloodSugar = bs
  1376→  1376→    } else if (bs >= 50 && bs <= 500) {
  1377→  1377→      // Convert mg/dL to mmol/L
  1378→  1378→      result.bloodSugar = parseFloat((bs / 18.0182).toFixed(1))
  1379→  1379→    }
  1380→  1380→  }
  1381→  1381→
  1382→  1382→  // Weight: "Wt 65", "Weight 65", "65kg", "65 kg"
  1383→  1383→  const wtMatch = lower.match(/(?:wt|weight)\s*:?\s*(\d{2,3}(?:\.\d)?)\s*kg?/)
  1384→  1384→  if (wtMatch) {
  1385→  1385→    result.weight = parseFloat(wtMatch[1])
  1386→  1386→  }
  1387→  1387→
  1388→  1388→  // Return null if nothing was parsed
  1389→  1389→  const hasAny = Object.keys(result).length > 0
  1390→  1390→  return hasAny ? result : null
  1391→  1391→}
