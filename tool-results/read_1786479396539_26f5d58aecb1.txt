   186→// ============== CUSTOM FIELDS ==============
   187→// Two-tier custom field system:
   188→//   1. GlobalCustomField — a global library of field definitions, managed by
   189→//      the Developer. These are the "master" definitions available to all orgs.
   190→//   2. OrgCustomField — per-org selections: which global fields are enabled
   191→//      for this org, plus an optional org-specific label override.
   192→//
   193→// This allows the Developer to build a library of fields (Blood Type, Chest
   194→// Measurement, Fabric Preference, etc.) once, and each org enables the ones
   195→// relevant to their business type. Orgs can also rename fields locally
   196→// (e.g. "Chest" → "Dada" in Malay) without affecting the global definition.
   197→//
   198→// The existing CustomField model is kept for backward compatibility — the
   199→// /api/custom-fields GET endpoint merges both sources so existing UI code
   200→// continues to work.
   201→
   202→model GlobalCustomField {
   203→  id              String   @id @default(cuid())
   204→  label           String                    // Global label, e.g. "Blood Type"
   205→  key             String   @unique          // Auto-derived: "blood_type"
   206→  type            String   @default("TEXT") // TEXT, NUMBER, DATE, SELECT, TEXTAREA, REFERENCE
   207→  options         String?                   // JSON array for SELECT, e.g. ["A+","O+"]
   208→  unit            String?                   // "cm", "kg"
   209→  required        Boolean  @default(false)
   210→  targetEntity    String   @default("resident") // resident, invoice, product, staff
   211→  referenceEntity String?                   // for REFERENCE type: product, staff, resident, invoice
   212→  description     String?                   // Developer notes about this field
   213→  // Which business type(s) this field belongs to. Null = all business types.
   214→  // Stored as JSON array: ["nursing_home","clinic"] or null for all.
   215→  businessTypes   String?
   216→  sortOrder       Int      @default(0)
   217→  active          Boolean  @default(true)
   218→  createdAt       DateTime @default(now())
   219→  updatedAt       DateTime @updatedAt
   220→
   221→  orgSelections   OrgCustomField[]
   222→}
   223→
   224→model OrgCustomField {
   225→  id              String   @id @default(cuid())
   226→  orgId           String                    // links to Organization
   227→  globalFieldId   String                    // links to GlobalCustomField
   228→  globalField     GlobalCustomField @relation(fields: [globalFieldId], references: [id], onDelete: Cascade)
   229→  // Optional org-specific label override. If null, uses the global label.
   230→  labelOverride   String?
   231→  // Optional org-specific sort order (overrides global sortOrder)
   232→  sortOrderOverride Int?
   233→  // Whether the field is enabled for this org (default true — selecting a
   234→  // field enables it; deselecting sets this to false)
   235→  enabled         Boolean  @default(true)
   236→  createdAt       DateTime @default(now())
   237→  updatedAt       DateTime @updatedAt
   238→
   239→  @@unique([orgId, globalFieldId])
   240→}

   242→// ============== CUSTOM TABS ==============
   243→// Global custom tab definitions — a collection of fields (built-in + custom)
   244→// that appears as a tab in the customer detail view. Managed by the Developer
   245→// and applied globally. Each org can show/hide tabs and override labels.
   246→//
   247→// Example: A "Measurements" tab for tailors containing Chest, Waist, Hip
   248→// custom fields. A "Medical Info" tab for clinics containing Blood Type,
   249→// Allergies, etc.
   250→
   251→model GlobalCustomTab {
   252→  id          String   @id @default(cuid())
   253→  label       String                    // Tab name, e.g. "Measurements"
   254→  key         String   @unique          // Auto-derived: "measurements"
   255→  description String?                   // Developer notes
   256→  // JSON array of field IDs included in this tab. Can reference:
   257→  //   - GlobalCustomField IDs (custom fields)
   258→  //   - Built-in field keys (e.g. "firstName", "allergies", "conditions")
   259→  fields      String   @default("[]")  // JSON: ["fieldId1","fieldId2","allergies"]
   260→  // Which module this tab appears in (e.g. 'resident', 'products', 'staff')
   261→  module      String   @default("resident")
   262→  // If true, this tab supports version history — each save creates a snapshot
   263→  // of all field values, so the user can track changes over time (e.g.
   264→  // body measurements, health metrics, etc.)
   265→  enableVersioning Boolean @default(false)
   266→  // Which business type(s) this tab belongs to. Null = all business types.
   267→  businessTypes String?
   268→  sortOrder   Int      @default(0)
   269→  active      Boolean  @default(true)
   270→  createdAt   DateTime @default(now())
   271→  updatedAt   DateTime @updatedAt
   272→
   273→  orgSelections OrgCustomTab[]
   274→}

   290→// ============== CUSTOM FIELD VALUE VERSIONS ==============
   291→// Stores snapshots of custom field values at a point in time — like a
   292→// "measurement history". Each time a user records new measurements (e.g.
   293→// body measurements for a tailor), a new version is created instead of
   294→// overwriting the old values. The latest version's values are also kept
   295→// in CustomFieldValue (the "current" values) for quick display in forms.
   296→//
   297→// Example: A tailor takes measurements on Jan 1 (version 1), then again
   298→// on Mar 1 (version 2). The customer detail shows the latest (Mar 1),
   299→// but the user can view the history to see how measurements changed.
   300→
   301→model CustomFieldValueVersion {
   302→  id          String   @id @default(cuid())
   303→  entityId    String                    // the entity (e.g. resident ID)
   304→  entityType  String   @default("resident")
   305→  // JSON snapshot of all field values at this point: { "fieldId1": "value1", "fieldId2": "value2" }
   306→  values      String   @default("{}")  // JSON object
   307→  // Optional label for this version, e.g. "Initial", "After diet", "3-month checkup"
   308→  label       String?
   309→  recordedAt  DateTime @default(now())  // when the measurements were taken
   310→  recordedBy  String?                   // user ID who recorded this version
   311→  recordedByName String?                // user name (denormalized for display)
   312→  createdAt   DateTime @default(now())
   313→
   314→  @@index([entityId, entityType, recordedAt])
   315→}

   451→model StaffAttendance {
   452→  id            String   @id @default(cuid())

   455→  date          DateTime // the work date

   457→  checkOut      DateTime? // actual check-out time
   458→  shiftStart    String?  // scheduled start "07:00"
   459→  shiftEnd      String?  // scheduled end "15:00"
   460→  status        String   @default("PRESENT") // PRESENT, ABSENT, LATE, HALF_DAY, ON_LEAVE, HOLIDAY
   461→  // Computed fields (filled by system on checkout)
   462→  workedHours   Float?   // actual hours worked (checkOut - checkIn, minus break)
   463→  overtimeHours Float?   // hours beyond 8h
   464→  lateMinutes   Int?     // minutes late (checkIn - shiftStart)
   465→  earlyLeaveMins Int?    // minutes early departure
   466→  // Location (optional GPS)
   467→  checkInLat    Float?
   468→  checkInLng    Float?
   469→  checkOutLat   Float?
   470→  checkOutLng   Float?
   471→  notes         String?
   472→  recordedById  String?  // who recorded (if manual entry)

   474→  createdAt     DateTime @default(now())
   475→  updatedAt     DateTime @updatedAt

   477→  @@unique([staffId, date]) // one attendance record per staff per day
   478→  @@index([staffId, date])

   481→// ============== PAYROLL ==============
   482→// Monthly payroll with Malaysian statutory deductions:
   483→//   EPF (KWSP): employee 11%, employer 12-13% (based on wage ceiling)
   484→//   SOCSO (PERKESO): employee 0.5%, employer 1.75% (Employment Injury + Invalidity)
   485→//   EIS (SIP): employee 0.2%, employer 0.2% ( Employment Insurance System)
   486→//   PCB (MTD): Monthly Tax Deduction — LHDN schedule
   487→//   Overtime: 1.5x on normal days, 2.0x rest days, 3.0x public holidays
   488→model Payroll {
   489→  id              String   @id @default(cuid())
   490→  staffId         String
   491→  staff           Staff    @relation(fields: [staffId], references: [id])
   492→  facilityId      String?
   493→  facility        Facility? @relation(fields: [facilityId], references: [id])
   494→  // Period
   495→  payrollMonth    String   // "2026-07" (YYYY-MM)
   496→  periodStart     DateTime
   497→  periodEnd       DateTime
   498→  status          String   @default("DRAFT") // DRAFT, APPROVED, PAID, CANCELLED
   499→  // Earnings
   500→  basicSalary     Float    @default(0)  // monthly basic
   501→  overtimePay     Float    @default(0)  // computed from attendance
   502→  allowances      Float    @default(0)  // housing, transport, meal, etc.
   503→  bonus           Float    @default(0)
   504→  commission      Float    @default(0)
   505→  grossPay        Float    @default(0)  // basic + OT + allowances + bonus + commission
   506→  // Deductions (Malaysian statutory)
   507→  epfEmployee     Float    @default(0)  // 11% of gross (capped at RM4,000 wage = max RM858.50)
   508→  epfEmployer     Float    @default(0)  // 12% or 13% of gross (capped)
   509→  socsoEmployee   Float    @default(0)  // 0.5% (capped at RM4,000 wage = max RM19.75)

   983→// ============== JOURNAL ENTRY ==============
   984→// A journal entry is a collection of journal lines that must balance
   985→// (total debits = total credits). Each entry has a source:
   986→//   - MANUAL: created by the user via the Journal Entries UI
   987→//   - AUTO_INVOICE: auto-posted when an invoice is created
   988→//   - AUTO_EXPENSE: auto-posted when an expense is recorded
   989→//   - AUTO_PAYMENT: auto-posted when a payment is received
   990→//   - AUTO_RECURRING: posted by a recurring JE template
   991→model JournalEntry {
