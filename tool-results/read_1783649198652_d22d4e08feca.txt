    76→    76→export default function Home() {
    77→    77→  const [user, setUser] = useState<CurrentUser | null>(null)

    81→    81→  const [userMenuOpen, setUserMenuOpen] = useState(false)

    84→    84→  const [facilityMenuOpen, setFacilityMenuOpen] = useState(false)
    85→    85→  // Module reordering
    86→    86→  const [customOrder, setCustomOrder] = useState<string[] | null>(null)
    87→    87→  const [editMode, setEditMode] = useState(false)
    88→    88→  const [draggedId, setDraggedId] = useState<string | null>(null)
    89→    89→  const [savingOrder, setSavingOrder] = useState(false)
    90→    90→  // Organization branding (loaded from /api/settings)

    92→    92→  const [orgLogoUrl, setOrgLogoUrl] = useState<string>('')
    93→    93→  const [appName, setAppName] = useState<string>('Serenity Care Home')
    94→    94→  const [appTagline, setAppTagline] = useState<string>('Resident & Operations Management')
    95→    95→  const [settingsData, setSettingsData] = useState<any>(null)
    96→    96→  const [organizations, setOrganizations] = useState<any[]>([])
    97→    97→
    98→    98→  // Initial session check is handled by <Login />, but we also do a top-level check
    99→    99→  // to skip the login screen if a session cookie already exists.
   100→   100→  useEffect(() => {
   101→   101→    fetch('/api/auth/me')
   102→   102→      .then(r => r.json())
   103→   103→      .then(data => {
   104→   104→        if (data.user) setUser(data.user)
   105→   105→      })
   106→   106→      .catch(() => {})
   107→   107→      .finally(() => setCheckingAuth(false))
   108→   108→  }, [])
   109→   109→
   110→   110→  // Handle Google Drive OAuth result query params at the top level.
   111→   111→  // The OAuth callback redirects to /?google_connected=1 or /?google_error=...,
   112→   112→  // but the user may land on any module (Dashboard by default). The previous
   113→   113→  // handler lived inside BackupRestoreSettings, which is only mounted when the
   114→   114→  // user is on Settings → Backup & Restore — so the result was silently lost
   115→   115→  // whenever the user landed elsewhere. Moving it here ensures the toast always
   116→   116→  // shows, regardless of the active module.
   117→   117→  useEffect(() => {
   118→   118→    if (typeof window === 'undefined') return
   119→   119→    const params = new URLSearchParams(window.location.search)
   120→   120→    let changed = false
   121→   121→    if (params.get('google_connected')) {
   122→   122→      toast.success('Google Drive connected successfully')
   123→   123→      params.delete('google_connected')
   124→   124→      changed = true
   125→   125→    }
   126→   126→    const err = params.get('google_error')
   127→   127→    if (err) {
   128→   128→      toast.error(`Google connection failed: ${err}`)
   129→   129→      params.delete('google_error')
   130→   130→      changed = true
   131→   131→    }
   132→   132→    if (changed) {
   133→   133→      const remaining = params.toString()
   134→   134→      const newUrl = remaining
   135→   135→        ? `${window.location.pathname}?${remaining}`
   136→   136→        : window.location.pathname
   137→   137→      window.history.replaceState({}, '', newUrl)
   138→   138→    }
   139→   139→  }, [])
   140→   140→
   141→   141→  // Load accessible facilities when user changes
   142→   142→  useEffect(() => {
   143→   143→    if (!user) {
   144→   144→      setFacilities([])
   145→   145→      setSelectedFacilityId('')
   146→   146→      setCustomOrder(null)
   147→   147→      return
   148→   148→    }
   149→   149→    fetch('/api/facilities/accessible')
   150→   150→      .then(r => r.json())
   151→   151→      .then(data => {
   152→   152→        setFacilities(data.facilities || [])
   153→   153→        // Auto-select first facility for non-Developer/Owner users
   154→   154→        // Developer and Owner start with "All Facilities" so they see everything
   155→   155→        if (data.facilities?.length > 0 && !selectedFacilityId && user.level > 1) {
   156→   156→          setSelectedFacilityId(data.facilities[0].id)
   157→   157→        }
   158→   158→      })
   159→   159→      .catch(() => {})
   160→   160→    // Load custom module order
   161→   161→    fetch('/api/module-order')
   162→   162→      .then(r => r.json())
   163→   163→      .then(data => {
   164→   164→        if (data.order && Array.isArray(data.order)) {
   165→   165→          setCustomOrder(data.order)
   166→   166→        }
   167→   167→      })
   168→   168→      .catch(() => {})
   169→   169→    // Load organization branding (name + logo) from settings
   170→   170→    fetch('/api/settings')
   171→   171→      .then(r => r.json())
   172→   172→      .then(data => {
   173→   173→        setSettingsData(data)
   174→   174→        if (data.organizationName) setOrgName(data.organizationName)
   175→   175→        const logo = data.appLogoUrl || data.organizationLogoUrl || ''
   176→   176→        if (logo) setOrgLogoUrl(logo)
   177→   177→        if (data.appName) setAppName(data.appName)
   178→   178→        if (data.appTagline) setAppTagline(data.appTagline)
   179→   179→      })
   180→   180→      .catch(() => {})
   181→   181→    // Load organizations (for businessType lookup — the Organization table's
   182→   182→    // businessType field is the source of truth for module visibility + labels)
   183→   183→    fetch('/api/organizations')
   184→   184→      .then(r => r.json())
   185→   185→      .then(data => {
   186→   186→        if (Array.isArray(data)) setOrganizations(data)
   187→   187→      })
   188→   188→      .catch(() => {})
   189→   189→  }, [user])
   190→   190→
   191→   191→  const handleLogout = useCallback(async () => {
   192→   192→    try {
   193→   193→      await fetch('/api/auth/logout', { method: 'POST' })
   194→   194→    } catch {}
   195→   195→    setUser(null)
   196→   196→    setFacilities([])
   197→   197→    setSelectedFacilityId('')
   198→   198→    setActiveModule('dashboard')
   199→   199→    setUserMenuOpen(false)
   200→   200→    toast.success('Signed out')
   201→   201→  }, [])
   202→   202→
   203→   203→  // While checking auth, show a small loading state
   204→   204→  if (checkingAuth) {
   205→   205→    return (
   206→   206→      <div className="min-h-screen flex items-center justify-center bg-muted/20">
   207→   207→        <div className="flex items-center gap-3 text-muted-foreground">
   208→   208→          <HeartPulse className="h-6 w-6 animate-pulse text-rose-500" />
   209→   209→          <span className="text-sm">Loading Serenity Care Home...</span>
   210→   210→        </div>
   211→   211→      </div>
   212→   212→    )
   213→   213→  }
   214→   214→
   215→   215→  // Not logged in → show login screen
   216→   216→  if (!user) {
   217→   217→    return <Login onLogin={(u) => setUser(u as any)} />
   218→   218→  }
   219→   219→
   220→   220→  const role = user.role as Role
   221→   221→  // Use custom moduleAccess if set (even if empty string), otherwise fall back to level-based settings, then role defaults
   222→   222→  // Level-based settings: org-scoped (levelModules:<orgId>:<level>) takes priority over global (levelModules:<level>)
   223→   223→  // Org-level module access (orgModules:<orgId>) acts as a gate — modules not in the org list are removed entirely
   224→   224→  let allowedModules: string[]
   225→   225→  if (user.moduleAccess !== null && user.moduleAccess !== undefined) {
   226→   226→    // Individual user override
   227→   227→    allowedModules = user.moduleAccess.split(',').map(s => s.trim()).filter(Boolean)
   228→   228→  } else {
   229→   229→    // Check org-scoped level modules first, then global, then role defaults
   230→   230→    const userOrgId = (user as any).organizationId
   231→   231→    const orgLevelKey = `levelModules:${userOrgId}:${user.level}`
   232→   232→    const globalLevelKey = `levelModules:${user.level}`
   233→   233→    const orgLevelModules = settingsData?.[orgLevelKey]
   234→   234→    const globalLevelModules = settingsData?.[globalLevelKey]
   235→   235→    if (Array.isArray(orgLevelModules)) {
   236→   236→      allowedModules = orgLevelModules
   237→   237→    } else if (Array.isArray(globalLevelModules)) {
   238→   238→      allowedModules = globalLevelModules
   239→   239→    } else {
   240→   240→      allowedModules = ROLE_MODULES[role]
   241→   241→    }
   242→   242→  }
   243→   243→
   244→   244→  // Org-level module gate: if orgModules:<orgId> is set, filter out modules not in the org list
   245→   245→  // Developer (level 0) bypasses this — they always see all modules
   246→   246→  if (role !== 'APP_DEVELOPER') {
   247→   247→    // Resolve the user's org: check organizationId first, then derive from facilityIds
   248→   248→    let userOrgId = (user as any).organizationId
   249→   249→    if (!userOrgId && user.facilityIds) {
   250→   250→      // Look up which org contains the user's first facility
   251→   251→      const userFids = user.facilityIds.split(',').map((s: string) => s.trim()).filter(Boolean)
   252→   252→      if (userFids.length > 0) {
   253→   253→        const orgFacility = facilities.find(f => f.id === userFids[0])
   254→   254→        userOrgId = (orgFacility as any)?.organizationId
   255→   255→      }
   256→   256→    }
   257→   257→    if (userOrgId) {
   258→   258→      const orgModulesKey = `orgModules:${userOrgId}`
   259→   259→      const orgModules = settingsData?.[orgModulesKey]
   260→   260→      if (Array.isArray(orgModules)) {
   261→   261→        allowedModules = allowedModules.filter(m => orgModules.includes(m))
   262→   262→      }
   263→   263→    }
   264→   264→
   265→   265→    // Business-type-based module filtering: hide modules not in the business type preset
   266→   266→    // e.g. a tailor org hides Medications, Vitals, Care Rounds, Rooms, Incidents, Messages
   267→   267→    const userOrgIdForBusiness = (user as any).organizationId || (() => {
   268→   268→      if (user.facilityIds) {
   269→   269→        const userFids = user.facilityIds.split(',').map((s: string) => s.trim()).filter(Boolean)
   270→   270→        if (userFids.length > 0) {
   271→   271→          return (facilities.find(f => f.id === userFids[0]) as any)?.organizationId
   272→   272→        }
   273→   273→      }
   274→   274→      return null
   275→   275→    })()
   276→   276→    if (userOrgIdForBusiness) {
   277→   277→      // Look up the org's business type from settings first (stored as setting key: businessType:<orgId>),
   278→   278→      // then fall back to the Organization table's businessType field (fetched via /api/organizations).
   279→   279→      const businessTypeKey = `businessType:${userOrgIdForBusiness}`
   280→   280→      let businessType = settingsData?.[businessTypeKey]
   281→   281→      if (!businessType) {
   282→   282→        // Fall back to the Organization table's businessType field
   283→   283→        const org = organizations.find(o => o.id === userOrgIdForBusiness)
   284→   284→        businessType = org?.businessType || 'nursing_home'
   285→   285→      }
   286→   286→      // Save for module label overrides
   287→   287→      ;(user as any)._businessType = businessType
   288→   288→      // Check for Developer-customized module list for this business type
   289→   289→      const typeModulesKey = `businessTypeModules:${businessType}`
   290→   290→      const customModules = settingsData?.[typeModulesKey]
   291→   291→      if (Array.isArray(customModules)) {
   292→   292→        // Developer has customized the module list for this business type — use it
   293→   293→        allowedModules = allowedModules.filter(m => customModules.includes(m))
   294→   294→      } else {
   295→   295→        // Use the preset defaults
   296→   296→        allowedModules = allowedModules.filter(m => isModuleVisible(businessType, m))
   297→   297→      }
   298→   298→    }
   299→   299→  }
   300→   300→
   301→   301→  const defaultModules = ALL_MODULES.filter(m => allowedModules.includes(m.id))
   302→   302→
   303→   303→  // Apply custom order if set — modules in the saved order first, then any
   304→   304→  // new modules the user has access to but weren't in the saved order
   305→   305→  const modules = customOrder
   306→   306→    ? [
   307→   307→        ...customOrder
   308→   308→          .filter(id => defaultModules.find(m => m.id === id))
   309→   309→          .map(id => defaultModules.find(m => m.id === id)!),
   310→   310→        ...defaultModules.filter(m => !customOrder.includes(m.id)),
   311→   311→      ]
   312→   312→    : defaultModules
   313→   313→
   314→   314→  // Apply custom module labels.
   315→   315→  // Resolution order (highest priority first):
   316→   316→  //   1. Per-org override: orgModuleLabels:<orgId> → { moduleId: "custom label" }
   317→   317→  //      Set by the org owner in Settings → Customization.
   318→   318→  //   2. Per-business-type: businessTypeModuleLabels:<type> → { moduleId: "custom label" }
   319→   319→  //      Set by the Developer in Developer → Customization → Org Type Management.
   320→   320→  //   3. Built-in default label from ALL_MODULES.
   321→   321→  const businessTypeLabelsKey = `businessTypeModuleLabels:${(user as any)._businessType || 'nursing_home'}`
   322→   322→  const businessTypeLabels = settingsData?.[businessTypeLabelsKey]
   323→   323→  const orgLabelsKey = `orgModuleLabels:${userOrgIdForBusiness}`
   324→   324→  const orgLabels = settingsData?.[orgLabelsKey]
   325→   325→
   326→   326→  const modulesWithLabels = modules.map(m => {
   327→   327→    // Org-level override takes priority over business-type-level
   328→   328→    let label = m.label
   329→   329→    if (businessTypeLabels && typeof businessTypeLabels === 'object' && businessTypeLabels[m.id]) {
   330→   330→      label = businessTypeLabels[m.id]
   331→   331→    }
   332→   332→    if (orgLabels && typeof orgLabels === 'object' && orgLabels[m.id]) {
   333→   333→      label = orgLabels[m.id]
   334→   334→    }
   335→   335→    return { ...m, label }
   336→   336→  })
   337→   337→
   338→   338→  const active = modulesWithLabels.find(m => m.id === activeModule) || modulesWithLabels[0]
   339→   339→
   340→   340→  // If role changes and current module not allowed, reset
   341→   341→  if (!allowedModules.includes(activeModule) && activeModule !== 'dashboard') {
   342→   342→    setActiveModule('dashboard')
   343→   343→  }
   344→   344→
   345→   345→  // ============ Module reorder handlers ============
   346→   346→  const moveModule = (id: string, direction: 'up' | 'down') => {
   347→   347→    const ids = modules.map(m => m.id)
   348→   348→    const idx = ids.indexOf(id)
   349→   349→    if (idx === -1) return
   350→   350→    const newIdx = direction === 'up' ? idx - 1 : idx + 1
   351→   351→    if (newIdx < 0 || newIdx >= ids.length) return
   352→   352→    ;[ids[idx], ids[newIdx]] = [ids[newIdx], ids[idx]]
   353→   353→    setCustomOrder(ids)
   354→   354→  }
   355→   355→
   356→   356→  const handleDragStart = (id: string) => setDraggedId(id)
   357→   357→  const handleDragOver = (e: React.DragEvent, overId: string) => {
   358→   358→    e.preventDefault()
   359→   359→    if (!draggedId || draggedId === overId) return
   360→   360→    const ids = modules.map(m => m.id)
   361→   361→    const fromIdx = ids.indexOf(draggedId)
   362→   362→    const toIdx = ids.indexOf(overId)
   363→   363→    if (fromIdx === -1 || toIdx === -1) return
   364→   364→    ids.splice(fromIdx, 1)
   365→   365→    ids.splice(toIdx, 0, draggedId)
   366→   366→    setCustomOrder(ids)
   367→   367→  }
   368→   368→  const handleDragEnd = () => setDraggedId(null)
   369→   369→
   370→   370→  const saveOrder = async () => {
   371→   371→    setSavingOrder(true)
   372→   372→    try {
   373→   373→      const order = modules.map(m => m.id)
   374→   374→      const res = await fetch('/api/module-order', {
   375→   375→        method: 'POST',
   376→   376→        headers: { 'Content-Type': 'application/json' },
   377→   377→        body: JSON.stringify({ order }),
   378→   378→      })
   379→   379→      if (!res.ok) throw new Error('Failed to save')
   380→   380→      setCustomOrder(order)
   381→   381→      setEditMode(false)
   382→   382→      toast.success('Module layout saved')
   383→   383→    } catch (e: any) {
   384→   384→      toast.error(e.message || 'Failed to save layout')
   385→   385→    }
   386→   386→    setSavingOrder(false)
   387→   387→  }
   388→   388→
   389→   389→  const resetOrder = async () => {
   390→   390→    setSavingOrder(true)
   391→   391→    try {
   392→   392→      await fetch('/api/module-order', { method: 'DELETE' })
   393→   393→      setCustomOrder(null)
   394→   394→      setEditMode(false)
   395→   395→      toast.success('Reset to default layout')
