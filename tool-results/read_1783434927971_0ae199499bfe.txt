    55→  }, [activityLog])

    80→      const blob = await response.blob()

    82→      if (blob.size < 10000) {
    83→        throw new Error(`File too small (${blob.size} bytes)`)
    84→      }

   131→  const handleDatabaseRestore = async () => {

   214→          setTimeout(() => {
   215→            window.location.href = '/?restored=' + Date.now()

   217→        } else {
   218→          addLog('Restore DB', 'error', '❌ SOME TABLES MISMATCHED — restore may be incomplete')
   219→          addLog('Restore DB', 'info', 'The data was uploaded but some tables have different counts. Try refreshing the page.')
   220→          toast.error('Restore uploaded but some tables mismatched — check log')
   221→        }
   222→      } else if (result.verified === true) {
   223→        addLog('Restore DB', 'success', '✅ RESTORE CONFIRMED SUCCESSFUL — server verified the data')
   224→        toast.success('Database restored and verified! Reloading...')
   225→        setSelectedFile(null)
   226→        const fileInput = document.getElementById('db-upload') as HTMLInputElement
   227→        if (fileInput) fileInput.value = ''
   228→        bumpDataVersion()
   229→        addLog('Restore DB', 'info', 'Performing full page reload in 2 seconds...')
   230→        setTimeout(() => {
   231→          window.location.href = '/?restored=' + Date.now()
   232→        }, 2000)
   233→      } else {
   234→        addLog('Restore DB', 'error', '❌ RESTORE VERIFICATION FAILED')
   235→        addLog('Restore DB', 'info', 'Try refreshing the page — the data may still have been restored.')
   236→        toast.error('Restore verification failed — check log')
   237→      }
   238→
   239→    } catch (e: any) {
   240→      addLog('Restore DB', 'error', `Failed: ${e.message || 'Unknown error'}`)
   241→      toast.error(e.message || 'Failed to restore database')
   242→    }

   247→    <div className="space-y-4">

   266→          <div className="text-sm text-amber-900">
   267→            <div className="font-semibold mb-1">Developer Tools — Owner Access Only</div>

   269→          </div>

   302→        <CardContent className="space-y-4">

   306→              <div className="font-medium">Download Backup</div>

   369→            <div className="text-xs text-amber-700 bg-amber-50 border border-amber-200 rounded-md p-2.5 flex items-start gap-2">

   462→// ============ APP CUSTOMERS TAB ============
   463→function AppCustomersTab() {
   464→  const { data: allUsers, loading, refetch } = useFetch<any[]>('/api/users?demoOnly=true')
   465→  const { data: realUsers, loading: realLoading, refetch: refetchReal } = useFetch<any[]>('/api/users?allExceptDemo=true')
   466→  const { data: facilities } = useFetch<any[]>('/api/facilities')
   467→  const { data: settings } = useFetch<any>('/api/settings')
   468→  const [demoMode, setDemoMode] = useState(false)
   469→  const [demoFacilityIds, setDemoFacilityIds] = useState<string[]>([])
   470→
   471→  useEffect(() => {
   472→    if (settings) {
   473→      setDemoMode(settings.demoMode || false)
   474→      setDemoFacilityIds(settings.demoFacilityIds || [])
   475→    }
   476→  }, [settings])
   477→
   478→  const allFacilities = facilities || []
   479→  const demoUsers = allUsers || []
   480→  const facilityUsers = realUsers || []
   481→
   482→  const saveSetting = async (key: string, value: any) => {
   483→    try {
   484→      await fetch('/api/settings', {
