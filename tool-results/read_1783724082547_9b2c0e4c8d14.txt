   173→// ============== CUSTOM FIELDS ==============
   174→// Two-tier custom field system:
   175→//   1. GlobalCustomField — a global library of field definitions, managed by
   176→//      the Developer. These are the "master" definitions available to all orgs.
   177→//   2. OrgCustomField — per-org selections: which global fields are enabled
   178→//      for this org, plus an optional org-specific label override.
   179→//
   180→// This allows the Developer to build a library of fields (Blood Type, Chest
   181→// Measurement, Fabric Preference, etc.) once, and each org enables the ones
   182→// relevant to their business type. Orgs can also rename fields locally
   183→// (e.g. "Chest" → "Dada" in Malay) without affecting the global definition.
   184→//
   185→// The existing CustomField model is kept for backward compatibility — the
   186→// /api/custom-fields GET endpoint merges both sources so existing UI code
   187→// continues to work.
   188→
   189→model GlobalCustomField {
   190→  id              String   @id @default(cuid())
   191→  label           String                    // Global label, e.g. "Blood Type"
   192→  key             String   @unique          // Auto-derived: "blood_type"
   193→  type            String   @default("TEXT") // TEXT, NUMBER, DATE, SELECT, TEXTAREA, REFERENCE
   194→  options         String?                   // JSON array for SELECT, e.g. ["A+","O+"]
   195→  unit            String?                   // "cm", "kg"
   196→  required        Boolean  @default(false)
   197→  targetEntity    String   @default("resident") // resident, invoice, product, staff
   198→  referenceEntity String?                   // for REFERENCE type: product, staff, resident, invoice
   199→  description     String?                   // Developer notes about this field
   200→  // Which business type(s) this field belongs to. Null = all business types.
   201→  // Stored as JSON array: ["nursing_home","clinic"] or null for all.
   202→  businessTypes   String?
   203→  sortOrder       Int      @default(0)
   204→  active          Boolean  @default(true)
   205→  createdAt       DateTime @default(now())
   206→  updatedAt       DateTime @updatedAt
   207→
   208→  orgSelections   OrgCustomField[]
   209→}
   210→
   211→model OrgCustomField {
   212→  id              String   @id @default(cuid())
   213→  orgId           String                    // links to Organization
   214→  globalFieldId   String                    // links to GlobalCustomField
   215→  globalField     GlobalCustomField @relation(fields: [globalFieldId], references: [id], onDelete: Cascade)
   216→  // Optional org-specific label override. If null, uses the global label.
   217→  labelOverride   String?
   218→  // Optional org-specific sort order (overrides global sortOrder)
   219→  sortOrderOverride Int?
   220→  // Whether the field is enabled for this org (default true — selecting a
   221→  // field enables it; deselecting sets this to false)
   222→  enabled         Boolean  @default(true)
   223→  createdAt       DateTime @default(now())
   224→  updatedAt       DateTime @updatedAt
   225→
   226→  @@unique([orgId, globalFieldId])
   227→}

   229→// ============== CUSTOM TABS ==============
   230→// Global custom tab definitions — a collection of fields (built-in + custom)
   231→// that appears as a tab in the customer detail view. Managed by the Developer
   232→// and applied globally. Each org can show/hide tabs and override labels.
   233→//
   234→// Example: A "Measurements" tab for tailors containing Chest, Waist, Hip
   235→// custom fields. A "Medical Info" tab for clinics containing Blood Type,
   236→// Allergies, etc.
   237→
   238→model GlobalCustomTab {
   239→  id          String   @id @default(cuid())
   240→  label       String                    // Tab name, e.g. "Measurements"
   241→  key         String   @unique          // Auto-derived: "measurements"
   242→  description String?                   // Developer notes
   243→  // JSON array of field IDs included in this tab. Can reference:
   244→  //   - GlobalCustomField IDs (custom fields)
   245→  //   - Built-in field keys (e.g. "firstName", "allergies", "conditions")
   246→  fields      String   @default("[]")  // JSON: ["fieldId1","fieldId2","allergies"]
   247→  // Which module this tab appears in (e.g. 'resident', 'products', 'staff')
   248→  module      String   @default("resident")
   249→  // If true, this tab supports version history — each save creates a snapshot
   250→  // of all field values, so the user can track changes over time (e.g.
   251→  // body measurements, health metrics, etc.)
   252→  enableVersioning Boolean @default(false)
   253→  // Which business type(s) this tab belongs to. Null = all business types.
   254→  businessTypes String?
   255→  sortOrder   Int      @default(0)
   256→  active      Boolean  @default(true)
   257→  createdAt   DateTime @default(now())
   258→  updatedAt   DateTime @updatedAt
   259→
   260→  orgSelections OrgCustomTab[]
   261→}
   262→
   263→model OrgCustomTab {
   264→  id              String   @id @default(cuid())
   265→  orgId           String
   266→  globalTabId     String
   267→  globalTab       GlobalCustomTab @relation(fields: [globalTabId], references: [id], onDelete: Cascade)
   268→  labelOverride   String?  // Org-specific tab name
   269→  enabled         Boolean  @default(true) // Show/hide for this org
   270→  sortOrderOverride Int?
   271→  createdAt       DateTime @default(now())
   272→  updatedAt       DateTime @updatedAt
   273→
   274→  @@unique([orgId, globalTabId])

   277→// ============== CUSTOM FIELD VALUE VERSIONS ==============
   278→// Stores snapshots of custom field values at a point in time — like a
   279→// "measurement history". Each time a user records new measurements (e.g.
   280→// body measurements for a tailor), a new version is created instead of
   281→// overwriting the old values. The latest version's values are also kept
   282→// in CustomFieldValue (the "current" values) for quick display in forms.
   283→//
   284→// Example: A tailor takes measurements on Jan 1 (version 1), then again
   285→// on Mar 1 (version 2). The customer detail shows the latest (Mar 1),
   286→// but the user can view the history to see how measurements changed.
   287→
   288→model CustomFieldValueVersion {
   289→  id          String   @id @default(cuid())
   290→  entityId    String                    // the entity (e.g. resident ID)
   291→  entityType  String   @default("resident")
   292→  // JSON snapshot of all field values at this point: { "fieldId1": "value1", "fieldId2": "value2" }
   293→  values      String   @default("{}")  // JSON object
   294→  // Optional label for this version, e.g. "Initial", "After diet", "3-month checkup"
   295→  label       String?
   296→  recordedAt  DateTime @default(now())  // when the measurements were taken
   297→  recordedBy  String?                   // user ID who recorded this version
   298→  recordedByName String?                // user name (denormalized for display)
   299→  createdAt   DateTime @default(now())
   300→
   301→  @@index([entityId, entityType, recordedAt])
   302→}
