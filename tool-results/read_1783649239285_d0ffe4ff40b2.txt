   562→// ============ REUSABLE STRING-LIST EDITOR (for simple dropdown option lists) ============
   563→// Used by DropdownSettings to edit arrays like roomTypes, expenseCategories, visitTypes, etc.
   564→function StringListEditor({
   565→  title,
   566→  description,
   567→  items,
   568→  onSave,
   569→  placeholder,
   570→}: {
   571→  title: string
   572→  description?: string
   573→  items: string[]
   574→  onSave: (items: string[]) => Promise<void>
   575→  placeholder?: string
   576→}) {
   577→  const [list, setList] = useState<string[]>(items)
   578→  const [newItem, setNewItem] = useState('')
   579→  const [saving, setSaving] = useState(false)
   580→
   581→  // Sync when parent items change (e.g. after facility switch)
   582→  useEffect(() => { setList(items) }, [items])
   583→
   584→  const add = () => {
   585→    const v = newItem.trim()
   586→    if (v && !list.includes(v)) {
   587→      setList([...list, v])
   588→      setNewItem('')
   589→    }
   590→  }
   591→  const remove = (i: number) => setList(list.filter((_, idx) => idx !== i))
   592→
   593→  return (
   594→    <Card>
   595→      <CardHeader>
   596→        <CardTitle className="text-sm">{title}</CardTitle>
   597→        {description && <CardDescription>{description}</CardDescription>}
   598→      </CardHeader>
   599→      <CardContent className="space-y-3">
   600→        <div className="flex flex-wrap gap-1.5">
   601→          {list.map((item, i) => (
   602→            <Badge key={i} variant="outline" className="text-xs pr-1 group">
   603→              {item}
   604→              <button
   605→                onClick={() => remove(i)}
   606→                className="ml-1 text-red-400 hover:text-red-600"
   607→                title="Delete"
   608→              >
   609→                <Trash2 className="h-3 w-3" />
   610→              </button>
   611→            </Badge>
   612→          ))}
   613→          {list.length === 0 && <p className="text-sm text-muted-foreground italic">No options configured</p>}
   614→        </div>
   615→        <div className="flex gap-2">
   616→          <Input
   617→            placeholder={placeholder || 'Add new option...'}
   618→            value={newItem}
   619→            onChange={e => setNewItem(e.target.value)}
   620→            onKeyDown={e => { if (e.key === 'Enter') { e.preventDefault(); add() } }}
   621→            className="flex-1"
   622→          />
   623→          <Button size="sm" onClick={add}><Plus className="h-3 w-3 mr-1" /> Add</Button>
   624→        </div>
   625→        <Button size="sm" variant="outline" onClick={async () => { setSaving(true); await onSave(list); setSaving(false) }} disabled={saving}>
   626→          {saving ? <Check className="h-3 w-3 mr-1" /> : <Save className="h-3 w-3 mr-1" />}
   627→          {saving ? 'Saved!' : 'Save'}
   628→        </Button>
   629→      </CardContent>
   630→    </Card>
   631→  )
   632→}
