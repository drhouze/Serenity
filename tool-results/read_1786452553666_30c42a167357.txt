   183→// ============== CUSTOM FIELDS ==============
   184→// Two-tier custom field system:
   185→//   1. GlobalCustomField — a global library of field definitions, managed by
   186→//      the Developer. These are the "master" definitions available to all orgs.
   187→//   2. OrgCustomField — per-org selections: which global fields are enabled
   188→//      for this org, plus an optional org-specific label override.
   189→//
   190→// This allows the Developer to build a library of fields (Blood Type, Chest
   191→// Measurement, Fabric Preference, etc.) once, and each org enables the ones
   192→// relevant to their business type. Orgs can also rename fields locally
   193→// (e.g. "Chest" → "Dada" in Malay) without affecting the global definition.
   194→//
   195→// The existing CustomField model is kept for backward compatibility — the
   196→// /api/custom-fields GET endpoint merges both sources so existing UI code
   197→// continues to work.
   198→
   199→model GlobalCustomField {
   200→  id              String   @id @default(cuid())
   201→  label           String                    // Global label, e.g. "Blood Type"
   202→  key             String   @unique          // Auto-derived: "blood_type"
   203→  type            String   @default("TEXT") // TEXT, NUMBER, DATE, SELECT, TEXTAREA, REFERENCE
   204→  options         String?                   // JSON array for SELECT, e.g. ["A+","O+"]
   205→  unit            String?                   // "cm", "kg"
   206→  required        Boolean  @default(false)
   207→  targetEntity    String   @default("resident") // resident, invoice, product, staff
   208→  referenceEntity String?                   // for REFERENCE type: product, staff, resident, invoice
   209→  description     String?                   // Developer notes about this field
   210→  // Which business type(s) this field belongs to. Null = all business types.
   211→  // Stored as JSON array: ["nursing_home","clinic"] or null for all.
   212→  businessTypes   String?
   213→  sortOrder       Int      @default(0)
   214→  active          Boolean  @default(true)
   215→  createdAt       DateTime @default(now())
   216→  updatedAt       DateTime @updatedAt
   217→
   218→  orgSelections   OrgCustomField[]
   219→}
   220→
   221→model OrgCustomField {
   222→  id              String   @id @default(cuid())
   223→  orgId           String                    // links to Organization
   224→  globalFieldId   String                    // links to GlobalCustomField
   225→  globalField     GlobalCustomField @relation(fields: [globalFieldId], references: [id], onDelete: Cascade)
   226→  // Optional org-specific label override. If null, uses the global label.
   227→  labelOverride   String?
   228→  // Optional org-specific sort order (overrides global sortOrder)
   229→  sortOrderOverride Int?
   230→  // Whether the field is enabled for this org (default true — selecting a
   231→  // field enables it; deselecting sets this to false)
   232→  enabled         Boolean  @default(true)
   233→  createdAt       DateTime @default(now())
   234→  updatedAt       DateTime @updatedAt
   235→
   236→  @@unique([orgId, globalFieldId])
   237→}

   239→// ============== CUSTOM TABS ==============
   240→// Global custom tab definitions — a collection of fields (built-in + custom)
   241→// that appears as a tab in the customer detail view. Managed by the Developer
   242→// and applied globally. Each org can show/hide tabs and override labels.
   243→//
   244→// Example: A "Measurements" tab for tailors containing Chest, Waist, Hip
   245→// custom fields. A "Medical Info" tab for clinics containing Blood Type,
   246→// Allergies, etc.
   247→
   248→model GlobalCustomTab {
   249→  id          String   @id @default(cuid())
   250→  label       String                    // Tab name, e.g. "Measurements"
   251→  key         String   @unique          // Auto-derived: "measurements"
   252→  description String?                   // Developer notes
   253→  // JSON array of field IDs included in this tab. Can reference:
   254→  //   - GlobalCustomField IDs (custom fields)
   255→  //   - Built-in field keys (e.g. "firstName", "allergies", "conditions")
   256→  fields      String   @default("[]")  // JSON: ["fieldId1","fieldId2","allergies"]
   257→  // Which module this tab appears in (e.g. 'resident', 'products', 'staff')
   258→  module      String   @default("resident")
   259→  // If true, this tab supports version history — each save creates a snapshot
   260→  // of all field values, so the user can track changes over time (e.g.
   261→  // body measurements, health metrics, etc.)
   262→  enableVersioning Boolean @default(false)
   263→  // Which business type(s) this tab belongs to. Null = all business types.
   264→  businessTypes String?
   265→  sortOrder   Int      @default(0)
   266→  active      Boolean  @default(true)
   267→  createdAt   DateTime @default(now())
   268→  updatedAt   DateTime @updatedAt
   269→
   270→  orgSelections OrgCustomTab[]
   271→}

   287→// ============== CUSTOM FIELD VALUE VERSIONS ==============
   288→// Stores snapshots of custom field values at a point in time — like a
   289→// "measurement history". Each time a user records new measurements (e.g.
   290→// body measurements for a tailor), a new version is created instead of
   291→// overwriting the old values. The latest version's values are also kept
   292→// in CustomFieldValue (the "current" values) for quick display in forms.
   293→//
   294→// Example: A tailor takes measurements on Jan 1 (version 1), then again
   295→// on Mar 1 (version 2). The customer detail shows the latest (Mar 1),
   296→// but the user can view the history to see how measurements changed.
   297→
   298→model CustomFieldValueVersion {
   299→  id          String   @id @default(cuid())
   300→  entityId    String                    // the entity (e.g. resident ID)
   301→  entityType  String   @default("resident")
   302→  // JSON snapshot of all field values at this point: { "fieldId1": "value1", "fieldId2": "value2" }
   303→  values      String   @default("{}")  // JSON object
   304→  // Optional label for this version, e.g. "Initial", "After diet", "3-month checkup"
   305→  label       String?
   306→  recordedAt  DateTime @default(now())  // when the measurements were taken
   307→  recordedBy  String?                   // user ID who recorded this version
   308→  recordedByName String?                // user name (denormalized for display)
   309→  createdAt   DateTime @default(now())
   310→
   311→  @@index([entityId, entityType, recordedAt])
   312→}

   448→model StaffAttendance {

   458→  // Computed fields (filled by system on checkout)
   459→  workedHours   Float?   // actual hours worked (checkOut - checkIn, minus break)
   460→  overtimeHours Float?   // hours beyond 8h
   461→  lateMinutes   Int?     // minutes late (checkIn - shiftStart)
   462→  earlyLeaveMins Int?    // minutes early departure
   463→  // Location (optional GPS)
   464→  checkInLat    Float?
   465→  checkInLng    Float?
   466→  checkOutLat   Float?

   478→// ============== PAYROLL ==============
   479→// Monthly payroll with Malaysian statutory deductions:
   480→//   EPF (KWSP): employee 11%, employer 12-13% (based on wage ceiling)
   481→//   SOCSO (PERKESO): employee 0.5%, employer 1.75% (Employment Injury + Invalidity)
   482→//   EIS (SIP): employee 0.2%, employer 0.2% ( Employment Insurance System)
   483→//   PCB (MTD): Monthly Tax Deduction — LHDN schedule
   484→//   Overtime: 1.5x on normal days, 2.0x rest days, 3.0x public holidays
   485→model Payroll {
   486→  id              String   @id @default(cuid())
   487→  staffId         String
   488→  staff           Staff    @relation(fields: [staffId], references: [id])
   489→  facilityId      String?
   490→  facility        Facility? @relation(fields: [facilityId], references: [id])
   491→  // Period
   492→  payrollMonth    String   // "2026-07" (YYYY-MM)
   493→  periodStart     DateTime
   494→  periodEnd       DateTime
   495→  status          String   @default("DRAFT") // DRAFT, APPROVED, PAID, CANCELLED
   496→  // Earnings
   497→  basicSalary     Float    @default(0)  // monthly basic
   498→  overtimePay     Float    @default(0)  // computed from attendance
   499→  allowances      Float    @default(0)  // housing, transport, meal, etc.
   500→  bonus           Float    @default(0)
   501→  commission      Float    @default(0)
   502→  grossPay        Float    @default(0)  // basic + OT + allowances + bonus + commission
   503→  // Deductions (Malaysian statutory)
   504→  epfEmployee     Float    @default(0)  // 11% of gross (capped at RM4,000 wage = max RM858.50)
   505→  epfEmployer     Float    @default(0)  // 12% or 13% of gross (capped)
   506→  socsoEmployee   Float    @default(0)  // 0.5% (capped at RM4,000 wage = max RM19.75)
   507→  socsoEmployer   Float    @default(0)  // 1.75% (capped at RM4,000 wage = max RM69.05)
   508→  eisEmployee     Float    @default(0)  // 0.2% (capped at RM4,000 wage = max RM7.90)
   509→  eisEmployer     Float    @default(0)  // 0.2% (capped at RM4,000 wage = max RM7.90)

   975→// ============== JOURNAL ENTRY ==============
   976→// A journal entry is a collection of journal lines that must balance
   977→// (total debits = total credits). Each entry has a source:
   978→//   - MANUAL: created by the user via the Journal Entries UI
   979→//   - AUTO_INVOICE: auto-posted when an invoice is created
   980→//   - AUTO_EXPENSE: auto-posted when an expense is recorded
   981→//   - AUTO_PAYMENT: auto-posted when a payment is received
   982→//   - AUTO_RECURRING: posted by a recurring JE template
   983→model JournalEntry {
   984→  id            String   @id @default(cuid())
   985→  facilityId    String?
   986→  facility      Facility? @relation(fields: [facilityId], references: [id])
   987→  entryNumber   String   @unique          // JE-000001 format
   988→  entryDate     DateTime @default(now())
   989→  memo          String                    // description / narration
   990→  source        String   @default("MANUAL") // MANUAL, AUTO_INVOICE, AUTO_EXPENSE, AUTO_PAYMENT, AUTO_RECURRING
   991→  reference     String?                   // e.g. invoice number, expense description, payment code
   992→  posted        Boolean  @default(true)   // false = draft, true = posted to GL
   993→  reversedBy    String?                   // ID of reversing JE (if this entry was reversed)
   994→  reversedJEId  String?                   // ID of the JE that reversed this one
   995→  
   996→  // Links to source documents (optional — set when source is AUTO_*)
   997→  invoiceId     String?
