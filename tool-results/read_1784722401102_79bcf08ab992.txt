     8→// Helper: marks a stock transfer as RECEIVED. For each line:
     9→//   1. Find (or auto-create) a matching destination InventoryItem by name + unit
    10→//      within the destination facility.
    11→//   2. Create a TRANSFER_IN InventoryTransaction on the destination item.
    12→//   3. Increment destination item's currentStock + update unitCost (to source's cost).
    13→//   4. Stamp the line's receivedQty + destinationItemId.
    14→//   5. Update transfer.status = 'RECEIVED', receivedDate = now.
    15→//
    16→// Source stock was already decremented when the transfer moved to IN_TRANSIT
    17→// (or when RECEIVED was set at creation — see POST handler). This helper only
    18→// handles the "receive" half of the move.
    19→async function receiveStockTransfer(transfer: any, currentUser: any) {
    20→  const toFacility = await db.facility.findUnique({ where: { id: transfer.toFacilityId }, select: { name: true } })
    21→  const fromFacility = await db.facility.findUnique({ where: { id: transfer.fromFacilityId }, select: { name: true } })
    22→  for (const line of transfer.lines) {
    23→    // Find matching destination item by name + unit (case-insensitive) within the dest facility
    24→    let destItem = await db.inventoryItem.findFirst({
    25→      where: {
    26→        facilityId: transfer.toFacilityId,
    27→        name: { equals: line.itemName },
    28→        // match by unit too — same name but different unit is treated as different item
    29→      },
    30→    })
    31→    // If not found, auto-create it
    32→    if (!destItem) {
    33→      const destCode = await generateInventoryCode(transfer.toFacilityId)
    34→      destItem = await db.inventoryItem.create({
    35→        data: {
    36→          code: destCode,
    37→          facilityId: transfer.toFacilityId,
    38→          name: line.itemName,
    39→          sku: line.itemSku || null,
    40→          category: line.itemCategory || 'OTHER',
    41→          unit: line.itemUnit || 'each',
    42→          currentStock: 0,  // will be incremented below
    43→          reorderLevel: 10,
    44→          reorderQty: 50,
    45→          unitCost: line.itemUnitCost || 0,
    46→          supplier: `Transferred from ${fromFacility?.name || 'another facility'}`,
    47→          active: true,
    48→          lastCountDate: new Date(),
    49→        },
    50→      })
    51→    }
    52→    // TRANSFER_IN transaction on destination item
    53→    await db.inventoryTransaction.create({
    54→      data: {
    55→        itemId: destItem.id,
    56→        type: 'TRANSFER_IN',
    57→        quantity: line.quantity,  // positive for in
    58→        reason: `Transfer ${transfer.transferNumber} ← ${fromFacility?.name || 'source facility'}`,
    59→        date: new Date(),
    60→        recordedBy: currentUser.name,
    61→        stockTransferId: transfer.id,
    62→      },
    63→    })
    64→    // Increment destination stock + update unitCost
    65→    await db.inventoryItem.update({
    66→      where: { id: destItem.id },
    67→      data: {
    68→        currentStock: { increment: line.quantity },
    69→        lastCountDate: new Date(),
    70→        unitCost: line.itemUnitCost > 0 ? line.itemUnitCost : undefined,
    71→      },
    72→    })
    73→    // Stamp the line
    74→    await db.stockTransferLine.update({
    75→      where: { id: line.id },
    76→      data: { receivedQty: line.quantity, destinationItemId: destItem.id },
    77→    })
    78→  }
    79→  await db.stockTransfer.update({
    80→    where: { id: transfer.id },
    81→    data: { status: 'RECEIVED', receivedDate: new Date() },
    82→  })
    83→}

    98→export async function GET(req: NextRequest) {
    99→  const { searchParams } = new URL(req.url)
   100→  const type = searchParams.get('type') || ''
   101→  const id = searchParams.get('id')
   102→  const residentId = searchParams.get('residentId')
   103→  const upcoming = searchParams.get('upcoming') === 'true'
   104→  const unbilled = searchParams.get('unbilled') === 'true'
   105→  const date = searchParams.get('date')
   106→  const facilityId = searchParams.get('facilityId') // selected facility from frontend
   107→
   108→  // Auth check — all data access requires login
   109→  const currentUser = await getSessionUser(req)
   110→  if (!currentUser) {
   111→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   112→  }
   113→
   114→  // Determine which facility IDs the user can access.
   115→  // Hierarchy:
   116→  //   - APP_DEVELOPER (L0): sees ALL facilities across ALL organizations (no scope)
   117→  //   - OWNER (L1): sees only facilities in their OWN organization
   118→  //   - MANAGER (L2) and below: sees only their assigned facilities
   119→  let accessibleFacilityIds: string[] = []
   120→  if (currentUser.role === 'APP_DEVELOPER') {
   121→    // Developer: no scope — sees everything
   122→    if (facilityId) accessibleFacilityIds = [facilityId]
   123→  } else if (currentUser.level === 1) {
   124→    // Owner: scoped to their organization's facilities
   125→    if (!currentUser.organizationId) {
   126→      // Owner without org → sees nothing (data isolation safeguard)
   127→      accessibleFacilityIds = ['__NO_ORG__'] // impossible ID → returns no rows
   128→    } else {
   129→      const orgFacilities = await db.facility.findMany({
   130→        where: { organizationId: currentUser.organizationId },
   131→        select: { id: true },
   132→      })
   133→      const orgFidSet = orgFacilities.map(f => f.id)
   134→      if (facilityId && orgFidSet.includes(facilityId)) {
   135→        accessibleFacilityIds = [facilityId]
   136→      } else {
   137→        accessibleFacilityIds = orgFidSet
   138→      }
   139→    }
   140→  } else {
   141→    // Non-owner: only their assigned facilities
   142→    const userFacilityIds = (currentUser.facilityIds || '').split(',').map(s => s.trim()).filter(Boolean)
   143→    if (facilityId && userFacilityIds.includes(facilityId)) {
   144→      accessibleFacilityIds = [facilityId]
   145→    } else {
   146→      accessibleFacilityIds = userFacilityIds
   147→    }
   148→  }
   149→
   150→  // Helper: build facility filters for queries
   151→  // Different models need different filter strategies based on how they relate to Facility:
   152→  //   - Direct facilityId          → residents, rooms, staff, expenses, products, inventory, invoices
   153→  //   - Via resident.facilityId    → medications, medAdmins, vitals, visits, incidents, careLogs, messages, invoiceItems
   154→  //   - Via staff.facilityId       → shifts
   155→  //   - Via item.facilityId        → inventoryTransactions
   156→  //   - No facility link           → auditLogs (system-wide), users (filter via facilityIds string match)
   157→  //
   158→  const hasFacilityScope = accessibleFacilityIds.length > 0
   159→  const facilityFilter = hasFacilityScope ? { facilityId: { in: accessibleFacilityIds } } : {}
   160→  const residentFacilityFilter = hasFacilityScope ? { resident: { facilityId: { in: accessibleFacilityIds } } } : {}
   161→  const staffFacilityFilter = hasFacilityScope ? { staff: { facilityId: { in: accessibleFacilityIds } } } : {}
   162→  const itemFacilityFilter = hasFacilityScope ? { item: { facilityId: { in: accessibleFacilityIds } } } : {}
   163→  // For users: facilityIds is a comma-separated string. Filter where the comma-separated list contains any of the accessible facility IDs.
   164→  // SQLite doesn't have native array operations, so we use contains for each ID. This is approximate but works for Owner's single-facility view.
   165→  const userFacilityFilter = hasFacilityScope
   166→    ? { OR: accessibleFacilityIds.flatMap(fid => [
   167→        { facilityIds: { contains: fid } },
   168→        { level: 1, role: 'OWNER' as any }, // Owner always visible (Owner sees all facilities)
   169→      ]) }
   170→    : {}
   171→
   172→  // For FAMILY role, restrict to linked residents only
   173→  const linkedResidentIds: string[] = currentUser.role === 'FAMILY'
   174→    ? (currentUser.linkedResidentIds || '').split(',').map(s => s.trim()).filter(Boolean)
   175→    : []
   176→
   177→  try {
   178→    // FAMILY role: block access to sensitive data types that they should never see directly.
   179→    // NOTE: incidents, careLogs, and visits are ALLOWED for family users — they are scoped
   180→    // to their linked residents via the `linkedResidentIds` filter in each case handler.
   181→    if (currentUser.role === 'FAMILY') {
   182→      const blockedTypes = ['rooms', 'staff', 'medications', 'medAdmins', 'invoices', 'invoiceItems', 'expenses', 'payments', 'paymentApplications', 'deposits', 'accounts', 'journalEntries', 'vendors', 'bankAccounts', 'shifts', 'users', 'products', 'inventory', 'inventoryTransactions', 'finance']
   183→      if (blockedTypes.includes(type)) {
   184→        return NextResponse.json({ error: 'Access denied for your role' }, { status: 403 })
   185→      }
   186→      // incidents, careLogs, visits, messages are allowed — each handler restricts to linkedResidentIds
   187→    }
   188→
   189→    switch (type) {
   190→      case 'residents': {
   191→        // FAMILY users can only see their linked residents
   192→        if (currentUser.role === 'FAMILY') {
   193→          if (id) {
   194→            // Verify this resident is linked to this family user
   195→            if (!linkedResidentIds.includes(id)) {
   196→              return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   197→            }
   198→          }
   199→          // Include the same relations as the staff query so the family user sees
   200→          // incidents, care logs, visits, vitals, and med administrations in the resident detail view.
   201→          // (Billing/financial relations like invoiceItems are intentionally excluded for family.)
   202→          const list = await db.resident.findMany({
   203→            where: { id: { in: linkedResidentIds }, status: 'ACTIVE' },
   204→            include: {
   205→              room: true,
   206→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   207→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   208→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   209→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   210→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   211→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   212→            },
   213→            orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   214→          })
   215→          if (id) {
   216→            const r = list.find(x => x.id === id)
   217→            if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   218→            return NextResponse.json(r)
   219→          }
   220→          return NextResponse.json(list)
   221→        }
   222→
   223→        if (id) {
   224→          const r = await db.resident.findUnique({
   225→            where: { id },
   226→            include: {
   227→              room: true,
   228→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   229→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   230→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   231→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   232→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   233→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   234→              invoiceItems: { where: { billed: false }, orderBy: { createdAt: 'desc' } },
   235→            },
   236→          })
   237→          if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   238→          return NextResponse.json(r)
   239→        }
   240→        const includeArchived = searchParams.get('includeArchived') === 'true'
   241→        const statusFilter = searchParams.get('status') // ACTIVE, DISCHARGED, etc.
   242→        const where: any = { ...facilityFilter }
   243→        if (statusFilter) {
   244→          where.status = statusFilter
   245→        } else if (!includeArchived) {
   246→          where.status = 'ACTIVE'
   247→        }
   248→        const list = await db.resident.findMany({
   249→          where,
   250→          include: { room: true },
   251→          orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   252→        })
   253→        return NextResponse.json(list)
   254→      }
   255→
   256→      case 'rooms': {
   257→        const rooms = await db.room.findMany({
   258→          include: { residents: { where: { status: 'ACTIVE' } } },
   259→          orderBy: { roomNumber: 'asc' },
   260→        })
   261→        return NextResponse.json(rooms)
   262→      }
   263→
   264→      case 'staff': {
   265→        const includeArchived = searchParams.get('includeArchived') === 'true'
   266→        // Multi-facility match: staff may be assigned via facilityId (primary) OR via the
   267→        // comma-separated facilityIds field (multi-facility assignment).
   268→        const where: any = hasFacilityScope
   269→          ? { OR: [
   270→              { facilityId: { in: accessibleFacilityIds } },
   271→              ...accessibleFacilityIds.flatMap(fid => [{ facilityIds: { contains: fid } }]),
   272→            ] }
   273→          : {}
   274→        if (!includeArchived) where.active = true
   275→        const staff = await db.staff.findMany({
   276→          where,
   277→          include: { _count: { select: { shifts: true, visits: true } } },
   278→          orderBy: [{ role: 'asc' }, { lastName: 'asc' }],
   279→        })
   280→        return NextResponse.json(staff)
   281→      }
   282→
   283→      case 'medications': {
   284→        const where: any = { ...residentFacilityFilter }
   285→        if (residentId) where.residentId = residentId
   286→        if (searchParams.get('active') !== 'false') where.active = true
   287→        const meds = await db.medication.findMany({
   288→          where,
   289→          include: { resident: { include: { room: true } } },
   290→          orderBy: { name: 'asc' },
   291→        })
   292→        return NextResponse.json(meds)
   293→      }
   294→
   295→      case 'medAdmins': {
   296→        const where: any = { ...residentFacilityFilter }
   297→        if (residentId) where.residentId = residentId
   298→        if (searchParams.get('today') === 'true') {
   299→          const start = new Date(); start.setHours(0, 0, 0, 0)
   300→          const end = new Date(); end.setHours(23, 59, 59, 999)
   301→          where.scheduledAt = { gte: start, lte: end }
   302→        }
   303→        const status = searchParams.get('status')
   304→        if (status) where.status = status
   305→        const admins = await db.medAdministration.findMany({
   306→          where,
   307→          include: { medication: true, resident: { include: { room: true } }, staff: true },
   308→          orderBy: { scheduledAt: 'desc' },
   309→          take: 200,
   310→        })
   311→        return NextResponse.json(admins)
   312→      }
   313→
   314→      case 'vitals': {
   315→        const where: any = { ...residentFacilityFilter }
   316→        if (residentId) where.residentId = residentId
   317→        // FAMILY users: restrict to their linked residents only
   318→        if (currentUser.role === 'FAMILY') {
   319→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   320→          where.residentId = { in: linkedResidentIds }

   322→        const vitals = await db.vitalSign.findMany({
   323→          where,
   324→          include: { resident: { include: { room: true } } },
   325→          orderBy: { recordedAt: 'desc' },

   328→        return NextResponse.json(vitals)

   330→
   331→      case 'visits': {
   332→        const where: any = { ...residentFacilityFilter }
   333→        if (residentId) where.residentId = residentId
   334→        // FAMILY users can only see visits for their linked residents
   335→        if (currentUser.role === 'FAMILY') {
   336→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   337→          where.residentId = { in: linkedResidentIds }
   338→        }
   339→        if (upcoming) {
   340→          where.scheduledAt = { gte: new Date() }
   341→          where.status = 'SCHEDULED'
   342→        } else {
   343→          // Past visits: exclude SCHEDULED (only show COMPLETED, CANCELLED, NO_SHOW)
   344→          where.status = { not: 'SCHEDULED' }
   345→        }
   346→        const visits = await db.visit.findMany({
   347→          where,
   348→          include: { resident: { include: { room: true } }, staff: true },

   354→

   356→        const where: any = { ...residentFacilityFilter }
   357→        if (residentId) where.residentId = residentId
   358→        // FAMILY users: restrict to their linked residents only
   359→        if (currentUser.role === 'FAMILY') {
   360→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   361→          where.residentId = { in: linkedResidentIds }
   362→        }
   363→        const incidents = await db.incidentReport.findMany({
   364→          where,
   365→          include: { resident: { include: { room: true } }, reportedBy: true },
   366→          orderBy: { occurredAt: 'desc' },
   367→          take: 100,
   368→        })
   369→        return NextResponse.json(incidents)
   370→      }
   371→
   372→      case 'careLogs': {
   373→        const where: any = { ...residentFacilityFilter }
   374→        if (residentId) where.residentId = residentId
   375→        // FAMILY users: restrict to their linked residents only
   376→        if (currentUser.role === 'FAMILY') {
   377→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   378→          where.residentId = { in: linkedResidentIds }
   379→        }
   380→        const logs = await db.careLog.findMany({
   381→          where,
   382→          include: { resident: { include: { room: true } }, staff: true },
   383→          orderBy: { recordedAt: 'desc' },
   384→          take: 100,
   385→        })
   386→        return NextResponse.json(logs)
   387→      }
   388→
   389→      case 'messages': {
   390→        const where: any = { ...residentFacilityFilter }
   391→        if (residentId) where.residentId = residentId
   392→        // FAMILY users can only see messages for their linked residents
   393→        if (currentUser.role === 'FAMILY') {
   394→          if (linkedResidentIds.length === 0) {
   395→            return NextResponse.json([])
   396→          }
   397→          where.residentId = { in: linkedResidentIds }

   399→        const msgs = await db.familyMessage.findMany({

   401→          include: { resident: { include: { room: true } }, sender: true },

   407→
   408→      case 'invoices': {

   412→            where: { id },
   413→            include: {
   414→              resident: { include: { room: true } },
   415→              items: true,
   416→              payments: { select: { id: true, paymentCode: true, amount: true, paymentDate: true, method: true, status: true } },
   417→            },
   418→          })
   419→          if (!invoice) return NextResponse.json({ error: 'Invoice not found' }, { status: 404 })
   420→          return NextResponse.json(invoice)
   421→        }
   422→        const where: any = { ...facilityFilter }
   423→        if (residentId) where.residentId = residentId
   424→        const status = searchParams.get('status')
   425→        if (status) where.status = status
   426→        const invoices = await db.invoice.findMany({

   428→          include: { resident: { include: { room: true } }, items: true },

   437→        const where: any = { ...residentFacilityFilter }

   440→        const items = await db.invoiceItem.findMany({

   442→          include: { resident: { include: { room: true } }, invoice: true },

   448→
   449→      case 'expenses': {
   450→        const where: any = { ...facilityFilter }
   451→        const category = searchParams.get('category')
   452→        if (category) where.category = category
   453→        const paidByStaffId = searchParams.get('paidByStaffId')
   454→        if (paidByStaffId) where.paidByStaffId = paidByStaffId
   455→        const vendorId = searchParams.get('vendorId')
   456→        if (vendorId) where.vendorId = vendorId
   457→        const reimbursementStatus = searchParams.get('reimbursementStatus')
   458→        if (reimbursementStatus) where.reimbursementStatus = reimbursementStatus
   459→        const expenses = await db.expense.findMany({
   460→          where,
   461→          orderBy: { date: 'desc' },
   462→          take: 200,
   463→          include: {
   464→            vendor: { select: { id: true, code: true, name: true } },
   465→            paidByStaff: { select: { id: true, code: true, firstName: true, lastName: true, role: true } },
   466→          },
   467→        })
   468→        return NextResponse.json(expenses)
   469→      }
   470→
   471→      case 'shifts': {
   472→        // Shifts are tied to staff (which has facilityId) — filter via staff relation
   473→        const where: any = { ...staffFacilityFilter }
   474→        if (date) {
   475→          // When a date is passed, return shifts for the entire WEEK containing that date
   476→          const d = new Date(date)
   477→          const dayOfWeek = d.getDay()
   478→          const diff = dayOfWeek === 0 ? -6 : 1 - dayOfWeek // Monday as start of week
   479→          const weekStart = new Date(d)
   480→          weekStart.setDate(d.getDate() + diff)
   481→          weekStart.setHours(0, 0, 0, 0)
   482→          const weekEnd = new Date(weekStart)
   483→          weekEnd.setDate(weekEnd.getDate() + 7)
   484→          where.date = { gte: weekStart, lt: weekEnd }
   485→        } else {
   486→          // Default: today onwards (for next 14 days)
   487→          const start = new Date(); start.setHours(0, 0, 0, 0)
   488→          const end = new Date(); end.setDate(end.getDate() + 14)
   489→          where.date = { gte: start, lt: end }

   829→                },
   830→              },
   831→            },
   832→          },
   833→          orderBy: { code: 'asc' },

   838→          const totalCredit = b.account?.journalLines?.reduce((s: number, l: any) => s + l.credit, 0) || 0
   839→          const computedBalance = (b.openingBalance || 0) + totalDebit - totalCredit
   840→          return {
   841→            ...b,
   842→            account: b.account ? { id: b.account.id, code: b.account.code, name: b.account.name } : null,
   843→            totalDebit,
   844→            totalCredit,
   845→            currentBalance: Math.round(computedBalance * 100) / 100,
   846→            transactionCount: b.account?.journalLines?.length || 0,
   847→          }
   848→        })
   849→        return NextResponse.json(banksWithBalance)
   850→      }
   851→
   852→      case 'deposits': {
   853→        const where: any = { ...facilityFilter }
   854→        const depResidentId = searchParams.get('residentId')
   855→        if (depResidentId) where.residentId = depResidentId
   856→        const depStatus = searchParams.get('status')
   857→        if (depStatus) where.status = depStatus

   860→          include: { resident: { select: { id: true, code: true, firstName: true, lastName: true } } },
   861→          orderBy: { paymentDate: 'desc' },
   862→          take: 500,
   863→        })
   864→        return NextResponse.json(deposits)
   865→      }
   866→
   867→      default:
   868→        return NextResponse.json({ error: 'Unknown type' }, { status: 400 })
   869→    }
   870→  } catch (e: any) {
   871→    console.error('API /data error:', e)
   872→    return NextResponse.json({ error: e.message }, { status: 500 })
   873→  }
   874→}

   876→// POST /api/data?type=... — create record
   877→export async function POST(req: NextRequest) {
   878→  const { searchParams } = new URL(req.url)
   879→  const type = searchParams.get('type') || ''
   880→
   881→  // Auth check
   882→  const currentUser = await getSessionUser(req)
   883→  if (!currentUser) {
   884→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   885→  }
   886→
   887→  const body = await req.json()
   888→
   889→  // Parse facilityId from query param — used to auto-assign facilityId to new records
   890→  const requestFacilityId = searchParams.get('facilityId') || ''
   891→
   892→  // === Facility ownership check ===
   893→  // Resolve the user's accessible facility IDs ONCE for this request. Individual
   894→  // handlers below use `canWriteFacility(facilityId)` to verify that the user is
   895→  // allowed to write to the facility they're stamping on a new record.
   896→  const { accessibleFacilityIds: postAccessibleFacilityIds, isScoped: postIsScoped } = await resolveAccessibleFacilityIds(currentUser, requestFacilityId || null)
   897→  const canWriteFacility = (fid: string | null | undefined): boolean => {
   898→    if (!fid) return false
   899→    if (!postIsScoped) return true  // Developer with no scope = all facilities
   900→    return postAccessibleFacilityIds.includes(fid)
   901→  }
   902→  // For handlers that need the full list (e.g. to verify body.facilityId):
   903→  // use `postAccessibleFacilityIds` directly.
   904→
   905→  try {
   906→    switch (type) {
   907→      case 'residents': {
   908→        if (!body.firstName || !body.lastName) return NextResponse.json({ error: 'First name and last name are required' }, { status: 400 })
   909→        // Auto-assign facilityId from the request if not already in body
   910→        const facilityId = body.facilityId || requestFacilityId || null
   911→        // Facility-ownership check
   912→        if (facilityId && !canWriteFacility(facilityId)) {
   913→          return NextResponse.json({ error: 'You do not have access to this facility' }, { status: 403 })
   914→        }
   915→        const residentCode = await generateResidentCode(facilityId)
   916→        const resident = await db.resident.create({ data: { ...body, code: residentCode, facilityId } })
   917→        const resFacilityName = await getFacilityName(facilityId)
   918→        await logAudit({
   919→          userId: currentUser.id, userName: currentUser.name, userCode: currentUser.code, userRole: currentUser.role,
   920→          action: AUDIT_ACTIONS.RESIDENT_CREATED, entityType: 'RESIDENT', entityId: resident.id,
   921→          description: `${currentUser.name} created resident ${residentCode} ${resident.firstName} ${resident.lastName}`,
   922→          metadata: { residentCode, firstName: resident.firstName, lastName: resident.lastName },
   923→          facilityId, facilityName: resFacilityName,
   924→        })
   925→        return NextResponse.json(resident)
   926→      }
   927→
   928→      case 'rooms': {
   929→        if (!body.roomNumber) return NextResponse.json({ error: 'Room number is required' }, { status: 400 })
   930→        const facilityId = body.facilityId || requestFacilityId || null
   931→        // Check for duplicate room number (scoped to facility)
   932→        const existing = await db.room.findFirst({
   933→          where: { roomNumber: body.roomNumber, facilityId },
   934→        })
   935→        if (existing) return NextResponse.json({ error: `Room number "${body.roomNumber}" already exists` }, { status: 400 })
   936→        const capacity = Math.max(1, parseInt(String(body.capacity ?? 1)) || 1)
   937→        if (capacity < 1) return NextResponse.json({ error: 'Capacity must be at least 1' }, { status: 400 })
   938→        const roomCode = await generateRoomCode(facilityId)
   939→        const room = await db.room.create({
   940→          data: {
   941→            code: roomCode, roomNumber: body.roomNumber, floor: parseInt(String(body.floor ?? 1)) || 1,
   942→            capacity, type: body.type || 'PRIVATE', status: body.status || 'AVAILABLE',
   943→            notes: body.notes || null, facilityId,
   944→            // Bulk import tracking (null for manually-created rooms)
   945→            importBatchId: body.importBatchId || null,
   946→          },
   947→        })
   948→        const roomFacilityName = await getFacilityName(facilityId)
   949→        await logAudit({
   950→          userId: currentUser.id, userName: currentUser.name, userCode: currentUser.code, userRole: currentUser.role,
   951→          action: 'ROOM_CREATED', entityType: 'ROOM', entityId: room.id,
   952→          description: `${currentUser.name} created room ${roomCode} (${body.roomNumber}, ${body.type || 'PRIVATE'})`,
   953→          metadata: { roomCode, roomNumber: body.roomNumber, type: body.type, capacity },
   954→          facilityId, facilityName: roomFacilityName,
   955→        })
   956→        return NextResponse.json(room)
   957→      }
   958→
   959→      case 'staff': {
   960→        const facilityId = body.facilityId || requestFacilityId || null
   961→        const staffCode = await generateStaffCode(facilityId)
   962→        const staffMember = await db.staff.create({ data: { ...body, code: staffCode, facilityId } })
   963→        const staffFacilityName = await getFacilityName(facilityId)
   964→        await logAudit({
   965→          userId: currentUser.id, userName: currentUser.name, userCode: currentUser.code, userRole: currentUser.role,
   966→          action: 'STAFF_CREATED', entityType: 'STAFF', entityId: staffMember.id,
   967→          description: `${currentUser.name} created staff ${staffCode} ${body.firstName} ${body.lastName} (${body.role || ''})`,
   968→          metadata: { staffCode, firstName: body.firstName, lastName: body.lastName, role: body.role },

   972→      }

   977→        const medResident = body.residentId ? await db.resident.findUnique({ where: { id: body.residentId }, select: { firstName: true, lastName: true, code: true, facilityId: true } }) : null
   978→        const medFacilityName = await getFacilityName(medResident?.facilityId || body.facilityId || requestFacilityId || null)
   979→        await logAudit({
   980→          userId: currentUser.id, userName: currentUser.name, userCode: currentUser.code, userRole: currentUser.role,
   981→          action: 'MEDICATION_CREATED', entityType: 'MEDICATION', entityId: med.id,

   984→          facilityId: medResident?.facilityId || body.facilityId || requestFacilityId || null,

   989→
