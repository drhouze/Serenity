    62→    62→        filteredCount={filtered.length}
    63→    63→      />
    64→    64→
    65→    65→      {Object.keys(grouped).length === 0 && <p className="text-sm text-muted-foreground text-center py-8">No {tab} visits</p>}
    66→    66→
    67→    67→      {Object.entries(grouped).map(([date, visits]) => (
    68→    68→        <div key={date}>
    69→    69→          <h3 className="text-sm font-semibold text-muted-foreground mb-2">{date}</h3>
    70→    70→          <div className="grid gap-2">
    71→    71→            {visits.map(v => (
    72→    72→              <Card key={v.id}>
    73→    73→                <CardContent className="p-3">
    74→    74→                  <div className="flex flex-col sm:flex-row sm:items-start justify-between gap-2">

    86→    86→
    87→    87→                      {/* Clinical note fields — shown if filled */}

    97→    97→                    </div>

   100→   100→                        <>

   109→   109→                            <X className="h-3 w-3" />

   113→   113→                      {v.status === 'COMPLETED' && (

   127→   127→      {/* Visit note dialog */}

   134→   134→
   135→   135→/**
   136→   136→ * VisitFormDialog — clinical notes form for completing a visit.
   137→   137→ * Includes chief complaint, vitals (auto-saved to Vital Signs), findings,
   138→   138→ * diagnosis, treatment plan, prescription, and follow-up.
   139→   139→ */
   140→   140→function VisitFormDialog({ visit, onClose, onSaved }: { visit: any; onClose: () => void; onSaved: () => void }) {
   141→   141→  const { data: currentUser } = useFetch<any>('/api/auth/me')
   142→   142→
   143→   143→  // No access denial — if the user can see the button, they can fill/edit.
   144→   144→  // Buttons are only shown to users with module access (controlled by role/level).
   145→   145→  const [editMode, setEditMode] = useState(false)
   146→   146→  const isReadOnly = visit.status === 'COMPLETED' && !editMode
   147→   147→
   148→   148→  // Prefill start/end with current date/time (formatted for datetime-local input)
   149→   149→  const nowLocal = () => {
   150→   150→    const d = new Date()
   151→   151→    const off = d.getTimezoneOffset()
   152→   152→    return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16)
   153→   153→  }
   154→   154→  const addMinutes = (dtStr: string, mins: number) => {
   155→   155→    const d = new Date(dtStr)
   156→   156→    d.setMinutes(d.getMinutes() + mins)
   157→   157→    const off = d.getTimezoneOffset()
   158→   158→    return new Date(d.getTime() - off * 60000).toISOString().slice(0, 16)
   159→   159→  }
   160→   160→
   161→   161→  const [form, setForm] = useState<any>({
   162→   162→    visitStart: visit.completedAt ? new Date(visit.completedAt).toISOString().slice(0, 16) : nowLocal(),
   163→   163→    visitEnd: visit.completedAt && visit.duration ? addMinutes(new Date(visit.completedAt).toISOString().slice(0, 16), visit.duration) : addMinutes(nowLocal(), 30),
   164→   164→    chiefComplaint: visit.chiefComplaint || '',
   165→   165→    // Structured vitals fields
   166→   166→    bpSys: visit.vitalsNote ? (visit.vitalsNote.match(/BP\s*(\d+)/i)?.[1] || '') : '',
   167→   167→    bpDia: visit.vitalsNote ? (visit.vitalsNote.match(/BP\s*\d+\/(\d+)/i)?.[1] || '') : '',
   168→   168→    hr: visit.vitalsNote ? (visit.vitalsNote.match(/HR\s*(\d+)/i)?.[1] || '') : '',
   169→   169→    temp: visit.vitalsNote ? (visit.vitalsNote.match(/Temp\s*([\d.]+)/i)?.[1] || '') : '',
   170→   170→    rr: visit.vitalsNote ? (visit.vitalsNote.match(/RR\s*(\d+)/i)?.[1] || '') : '',
   171→   171→    o2: visit.vitalsNote ? (visit.vitalsNote.match(/SpO2?\s*(\d+)/i)?.[1] || '') : '',
   172→   172→    bs: visit.vitalsNote ? (visit.vitalsNote.match(/BS\s*([\d.]+)/i)?.[1] || '') : '',
   173→   173→    wt: visit.vitalsNote ? (visit.vitalsNote.match(/Wt\s*([\d.]+)/i)?.[1] || '') : '',
   174→   174→    vitalsNote: visit.vitalsNote || '',
   175→   175→    findings: visit.findings || '',
   176→   176→    diagnosis: visit.diagnosis || '',
   177→   177→    treatmentPlan: visit.treatmentPlan || '',
   178→   178→    prescription: visit.prescription || '',
   179→   179→    followUpNote: visit.followUpNote || '',
   180→   180→    recommendations: visit.recommendations || '',
   181→   181→    duration: visit.duration || '',
   182→   182→  })
   183→   183→  const [saving, setSaving] = useState(false)
   184→   184→
   185→   185→  // Auto-calculate duration when start/end changes
   186→   186→  const calcDuration = () => {
   187→   187→    if (form.visitStart && form.visitEnd) {
   188→   188→      const start = new Date(form.visitStart)
   189→   189→      const end = new Date(form.visitEnd)
   190→   190→      const diffMin = Math.round((end.getTime() - start.getTime()) / 60000)
   191→   191→      if (diffMin > 0) return diffMin
   192→   192→    }
   193→   193→    return form.duration ? parseInt(form.duration) : null
   194→   194→  }
   195→   195→
   196→   196→  const submit = async () => {
   197→   197→    setSaving(true)
   198→   198→    try {
   199→   199→      const userName = currentUser?.user?.name || 'Unknown'
   200→   200→      const duration = calcDuration()
   201→   201→
   202→   202→      // Build vitalsNote string from structured fields (for display on the visit record)
   203→   203→      const vitalsParts: string[] = []
   204→   204→      if (form.bpSys && form.bpDia) vitalsParts.push(`BP ${form.bpSys}/${form.bpDia}`)
   205→   205→      if (form.hr) vitalsParts.push(`HR ${form.hr}`)
   206→   206→      if (form.temp) vitalsParts.push(`Temp ${form.temp}`)
   207→   207→      if (form.rr) vitalsParts.push(`RR ${form.rr}`)
   208→   208→      if (form.o2) vitalsParts.push(`SpO2 ${form.o2}`)
   209→   209→      if (form.bs) vitalsParts.push(`BS ${form.bs}`)
   210→   210→      if (form.wt) vitalsParts.push(`Wt ${form.wt}`)
   211→   211→      const vitalsNoteStr = vitalsParts.join(', ')
   212→   212→
   213→   213→      const payload: any = {
   214→   214→        chiefComplaint: form.chiefComplaint || null,
   215→   215→        vitalsNote: vitalsNoteStr || null,
   216→   216→        findings: form.findings || null,
   217→   217→        diagnosis: form.diagnosis || null,
   218→   218→        treatmentPlan: form.treatmentPlan || null,
   219→   219→        prescription: form.prescription || null,
   220→   220→        followUpNote: form.followUpNote || null,
   221→   221→        recommendations: form.recommendations || null,
   222→   222→        duration: duration,
   223→   223→        scheduledAt: form.visitStart ? new Date(form.visitStart).toISOString() : undefined,
   224→   224→      }
   225→   225→      if (visit.status === 'SCHEDULED') {
   226→   226→        payload.status = 'COMPLETED'
   227→   227→        payload.completedAt = form.visitEnd ? new Date(form.visitEnd).toISOString() : new Date().toISOString()
   228→   228→        payload.completedById = currentUser?.user?.id || null
   229→   229→        payload.completedByName = userName
   230→   230→      } else {
   231→   231→        payload.completedAt = form.visitEnd ? new Date(form.visitEnd).toISOString() : undefined
   232→   232→      }
   233→   233→      await apiPatch(`/api/data?type=visits&id=${visit.id}`, payload)
   234→   234→
   235→   235→      // Auto-create a VitalSign record from the structured vitals fields
   236→   236→      const vitalsData: any = {}
   237→   237→      if (form.bpSys) vitalsData.bloodPressureSystolic = parseInt(form.bpSys)
   238→   238→      if (form.bpDia) vitalsData.bloodPressureDiastolic = parseInt(form.bpDia)
   239→   239→      if (form.hr) vitalsData.heartRate = parseInt(form.hr)
   240→   240→      if (form.temp) vitalsData.temperature = parseFloat(form.temp)
   241→   241→      if (form.rr) vitalsData.respiratoryRate = parseInt(form.rr)
   242→   242→      if (form.o2) vitalsData.oxygenSaturation = parseInt(form.o2)
   243→   243→      if (form.bs) vitalsData.bloodSugar = parseFloat(form.bs)
   244→   244→      if (form.wt) vitalsData.weight = parseFloat(form.wt)
   245→   245→
   246→   246→      if (Object.keys(vitalsData).length > 0) {
   247→   247→        try {
   248→   248→          const { apiPost } = await import('./api')
   249→   249→          await apiPost('/api/data?type=vitals', {
   250→   250→            residentId: visit.residentId,
   251→   251→            ...vitalsData,
   252→   252→            notes: `Recorded during ${visit.visitType.replace(/_/g, ' ')} visit by ${userName}`,
   253→   253→            recordedById: currentUser?.user?.id || null,
   254→   254→          })
   255→   255→          toast.success('Vitals auto-saved to Vital Signs module')
   256→   256→        } catch (e: any) {
   257→   257→          console.log('Vitals auto-save failed (non-critical):', e.message)
   258→   258→        }
   259→   259→      }
   260→   260→
   261→   261→      toast.success(visit.status === 'SCHEDULED' ? 'Visit form submitted — moved to Past Visits' : 'Visit notes updated')
   262→   262→      onSaved()
   263→   263→    } catch (e: any) { toast.error(e.message) }
   264→   264→    setSaving(false)
   265→   265→  }
   266→   266→
   267→   267→  return (
   268→   268→    <div className="fixed inset-0 bg-black/50 z-50 flex items-start sm:items-center justify-center p-4 overflow-y-auto">
   269→   269→      <div className="bg-background rounded-lg shadow-xl w-full max-w-lg my-8 max-h-[calc(100vh-4rem)] flex flex-col">
   270→   270→        <div className="flex justify-between items-center border-b p-4 flex-shrink-0">
   271→   271→          <h3 className="font-semibold">
   272→   272→            {isReadOnly ? 'Visit Notes (Read-Only)' : visit.status === 'SCHEDULED' ? 'Visit Form — Clinical Notes' : 'Edit Visit Notes'}
   273→   273→          </h3>
   274→   274→          <Button variant="ghost" size="sm" onClick={onClose}>×</Button>
   275→   275→        </div>
   276→   276→        <div className="p-4 overflow-y-auto flex-1 space-y-3">
   277→   277→          {/* Visit info */}
   278→   278→          <div className="text-xs text-muted-foreground bg-muted/30 rounded p-2 flex flex-wrap items-center gap-x-2 gap-y-1">
   279→   279→            <Badge variant="outline">{visit.visitType.replace(/_/g, ' ')}</Badge>
   280→   280→            <span className="whitespace-nowrap">Scheduled: {fmtDateTime(visit.scheduledAt)}</span>
   281→   281→            {visit.staff && <span className="whitespace-nowrap">• Assigned: {visit.staff.firstName} {visit.staff.lastName}</span>}
   282→   282→            {visit.resident && <span className="whitespace-nowrap">• {visit.resident.firstName} {visit.resident.lastName}</span>}
   283→   283→            {visit.completedByName && <span className="whitespace-nowrap">• Filled by: {visit.completedByName}</span>}
   284→   284→            {isReadOnly && <Badge variant="outline" className="text-[9px] text-amber-700 border-amber-300">Read-Only</Badge>}
   285→   285→          </div>
   286→   286→
   287→   287→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 text-sm">
   288→   288→            {/* Visit start & end date/time — prefilled with current time */}
   289→   289→            <div>
   290→   290→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Visit Start *</label>
   291→   291→              <Input type="datetime-local" value={form.visitStart} onChange={e => setForm({ ...form, visitStart: e.target.value })} disabled={isReadOnly} className="text-sm" />
   292→   292→            </div>
   293→   293→            <div>
   294→   294→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Visit End</label>
   295→   295→              <Input type="datetime-local" value={form.visitEnd} onChange={e => setForm({ ...form, visitEnd: e.target.value })} disabled={isReadOnly} className="text-sm" />
   296→   296→              {calcDuration() && <p className="text-[9px] text-muted-foreground mt-0.5">Duration: {calcDuration()} min</p>}
   297→   297→            </div>
   298→   298→            <div className="sm:col-span-2">
   299→   299→              <label className="text-xs font-medium text-muted-foreground mb-1 block">Chief Complaint</label>
   300→   300→              <textarea className="w-full border rounded px-2 py-1.5 text-sm" rows={2} value={form.chiefComplaint} onChange={e => setForm({ ...form, chiefComplaint: e.target.value })} disabled={isReadOnly} placeholder="Patient's main complaint, e.g. lower back pain for 2 weeks" />
   301→   301→            </div>
   302→   302→            <div className="sm:col-span-2 border rounded-md p-2 bg-emerald-50/30">
   303→   303→              <div className="text-xs font-semibold text-emerald-700 mb-2 flex items-center gap-1">
   304→   304→                <Activity className="h-3.5 w-3.5" /> Vital Signs <span className="text-[9px] text-muted-foreground font-normal">(auto-saves to Vital Signs module)</span>
   305→   305→              </div>
   306→   306→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   307→   307→                <div>
   308→   308→                  <label className="text-[10px] text-muted-foreground block mb-0.5">BP Systolic (mmHg)</label>
   309→   309→                  <Input type="number" value={form.bpSys || ''} onChange={e => setForm({ ...form, bpSys: e.target.value })} disabled={isReadOnly} placeholder="140" className="text-sm h-8" />
   310→   310→                </div>
   311→   311→                <div>
   312→   312→                  <label className="text-[10px] text-muted-foreground block mb-0.5">BP Diastolic (mmHg)</label>
   313→   313→                  <Input type="number" value={form.bpDia || ''} onChange={e => setForm({ ...form, bpDia: e.target.value })} disabled={isReadOnly} placeholder="90" className="text-sm h-8" />
   314→   314→                </div>
   315→   315→                <div>
   316→   316→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Heart Rate (bpm)</label>
   317→   317→                  <Input type="number" value={form.hr || ''} onChange={e => setForm({ ...form, hr: e.target.value })} disabled={isReadOnly} placeholder="72" className="text-sm h-8" />
   318→   318→                </div>
   319→   319→                <div>
   320→   320→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Temperature (°C)</label>
   321→   321→                  <Input type="number" step="0.1" value={form.temp || ''} onChange={e => setForm({ ...form, temp: e.target.value })} disabled={isReadOnly} placeholder="37.2" className="text-sm h-8" />
   322→   322→                </div>
   323→   323→                <div>
   324→   324→                  <label className="text-[10px] text-muted-foreground block mb-0.5">Resp Rate (/min)</label>
   325→   325→                  <Input type="number" value={form.rr || ''} onChange={e => setForm({ ...form, rr: e.target.value })} disabled={isReadOnly} placeholder="18" className="text-sm h-8" />
   326→   326→                </div>
   327→   327→                <div>
   328→   328→                  <label className="text-[10px] text-muted-foreground block mb-0.5">SpO₂ (%)</label>
   329→   329→                  <Input type="number" value={form.o2 || ''} onChange={e => setForm({ ...form, o2: e.target.value })} disabled={isReadOnly} placeholder="98" className="text-sm h-8" />
   330→   330→                </div>

   393→   393→/**
   394→   394→ * Parses vital signs from free-text input.
   395→   395→ * Supports: "BP 140/90, HR 72, Temp 37.2, RR 18, SpO2 98, BS 5.5, Wt 65"
   396→   396→ */
   397→   397→function parseVitalsFromText(text: string): any | null {
   398→   398→  const result: any = {}
   399→   399→  const lower = text.toLowerCase()
   400→   400→
   401→   401→  const bpMatch = lower.match(/(?:bp|blood\s*pressure)?\s*(\d{2,3})\s*\/\s*(\d{2,3})/)
   402→   402→  if (bpMatch) { result.bloodPressureSystolic = parseInt(bpMatch[1]); result.bloodPressureDiastolic = parseInt(bpMatch[2]) }
   403→   403→
   404→   404→  const hrMatch = lower.match(/(?:hr|heart\s*rate|pulse)\s*:?\s*(\d{2,3})/)
   405→   405→  if (hrMatch) result.heartRate = parseInt(hrMatch[1])
   406→   406→
   407→   407→  const tempMatch = lower.match(/(?:temp|temperature)?\s*:?\s*(\d{2,3}(?:\.\d)?)\s*(?:°?c|°?f)?/i)
   408→   408→  if (tempMatch) {
   409→   409→    const temp = parseFloat(tempMatch[1])
   410→   410→    if (temp >= 30 && temp <= 45) result.temperature = temp
   411→   411→    else if (temp >= 90 && temp <= 115) result.temperature = parseFloat(((temp - 32) * 5/9).toFixed(1))
   412→   412→  }
   413→   413→
   414→   414→  const rrMatch = lower.match(/(?:rr|resp(?:iratory)?\s*rate)\s*:?\s*(\d{2,3})/)
   415→   415→  if (rrMatch) { const rr = parseInt(rrMatch[1]); if (rr >= 8 && rr <= 60) result.respiratoryRate = rr }
   416→   416→
   417→   417→  const o2Match = lower.match(/(?:spo2|o2|oxygen|sat)\s*:?\s*(\d{2,3})\s*%?/)
   418→   418→  if (o2Match) { const o2 = parseInt(o2Match[1]); if (o2 >= 50 && o2 <= 100) result.oxygenSaturation = o2 }
   419→   419→
   420→   420→  const bsMatch = lower.match(/(?:bs|blood\s*sugar|glucose|bgl)\s*:?\s*(\d{1,3}(?:\.\d)?)\s*(?:mmol\/l|mg\/dl)?/)
   421→   421→  if (bsMatch) {
   422→   422→    const bs = parseFloat(bsMatch[1])
   423→   423→    if (bs >= 3 && bs <= 30) result.bloodSugar = bs
   424→   424→    else if (bs >= 50 && bs <= 500) result.bloodSugar = parseFloat((bs / 18.0182).toFixed(1))
   425→   425→  }
   426→   426→
   427→   427→  const wtMatch = lower.match(/(?:wt|weight)\s*:?\s*(\d{2,3}(?:\.\d)?)\s*kg?/)
   428→   428→  if (wtMatch) result.weight = parseFloat(wtMatch[1])
   429→   429→
   430→   430→  return Object.keys(result).length > 0 ? result : null
   431→   431→}
