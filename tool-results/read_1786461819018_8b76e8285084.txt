     8→// Helper: marks a stock transfer as RECEIVED. For each line:
     9→//   1. Find (or auto-create) a matching destination InventoryItem by name + unit
    10→//      within the destination facility.
    11→//   2. Create a TRANSFER_IN InventoryTransaction on the destination item.
    12→//   3. Increment destination item's currentStock + update unitCost (to source's cost).
    13→//   4. Stamp the line's receivedQty + destinationItemId.
    14→//   5. Update transfer.status = 'RECEIVED', receivedDate = now.
    15→//
    16→// Source stock was already decremented when the transfer moved to IN_TRANSIT
    17→// (or when RECEIVED was set at creation — see POST handler). This helper only
    18→// handles the "receive" half of the move.
    19→async function receiveStockTransfer(transfer: any, currentUser: any) {
    20→  const toFacility = await db.facility.findUnique({ where: { id: transfer.toFacilityId }, select: { name: true } })
    21→  const fromFacility = await db.facility.findUnique({ where: { id: transfer.fromFacilityId }, select: { name: true } })
    22→  for (const line of transfer.lines) {
    23→    // Match destination InventoryItem by (in priority order):
    24→    //   1. line.itemId — if the source item exists in the destination facility (cross-facility item ID link)
    25→    //   2. line.itemSku — same SKU within the destination facility (most reliable for shared catalogues)
    26→    //   3. line.itemName + unit — last-resort name match (case-insensitive)
    27→    // Previously this matched ONLY by name, which created duplicates when names differed by case or when
    28→    // multiple items shared a name. SKU is the canonical product identifier.
    29→    let destItem: any = null
    30→    // 1. Try by source itemId (only useful if the destination facility happens to share the same item ID — rare)
    31→    if (line.itemId) {
    32→      destItem = await db.inventoryItem.findFirst({
    33→        where: { id: line.itemId, facilityId: transfer.toFacilityId },
    34→      })
    35→    }
    36→    // 2. Try by SKU within the destination facility
    37→    if (!destItem && line.itemSku) {
    38→      destItem = await db.inventoryItem.findFirst({
    39→        where: { sku: line.itemSku, facilityId: transfer.toFacilityId },
    40→      })
    41→    }
    42→    // 3. Last resort: match by name (case-insensitive) within the destination facility
    43→    if (!destItem && line.itemName) {
    44→      destItem = await db.inventoryItem.findFirst({
    45→        where: {
    46→          facilityId: transfer.toFacilityId,
    47→          name: { equals: line.itemName, mode: 'insensitive' },
    48→        },
    49→      })
    50→    }
    51→    // If still not found, auto-create it using the snapshot data from the transfer line
    52→    if (!destItem) {
    53→      const destCode = await generateInventoryCode(transfer.toFacilityId)
    54→      destItem = await db.inventoryItem.create({
    55→        data: {
    56→          code: destCode,
    57→          facilityId: transfer.toFacilityId,
    58→          name: line.itemName,
    59→          sku: line.itemSku || null,
    60→          category: line.itemCategory || 'OTHER',
    61→          unit: line.itemUnit || 'each',
    62→          currentStock: 0,  // will be incremented below
    63→          reorderLevel: 10,
    64→          reorderQty: 50,
    65→          unitCost: line.itemUnitCost || 0,
    66→          supplier: `Transferred from ${fromFacility?.name || 'another facility'}`,
    67→          active: true,
    68→          lastCountDate: new Date(),
    69→        },
    70→      })
    71→    }
    72→    // TRANSFER_IN transaction on destination item
    73→    await db.inventoryTransaction.create({
    74→      data: {
    75→        itemId: destItem.id,
    76→        type: 'TRANSFER_IN',
    77→        quantity: line.quantity,  // positive for in
    78→        reason: `Transfer ${transfer.transferNumber} ← ${fromFacility?.name || 'source facility'}`,
    79→        date: new Date(),
    80→        recordedBy: currentUser.name,
    81→        recordedById: currentUser.id,
    82→        stockTransferId: transfer.id,
    83→      },
    84→    })
    85→    // Increment destination stock + update unitCost
    86→    await db.inventoryItem.update({
    87→      where: { id: destItem.id },
    88→      data: {
    89→        currentStock: { increment: line.quantity },
    90→        lastCountDate: new Date(),
    91→        unitCost: line.itemUnitCost > 0 ? line.itemUnitCost : undefined,
    92→      },
    93→    })
    94→    // Stamp the line
    95→    await db.stockTransferLine.update({
    96→      where: { id: line.id },
    97→      data: { receivedQty: line.quantity, destinationItemId: destItem.id },
    98→    })
    99→  }
   100→  await db.stockTransfer.update({
   101→    where: { id: transfer.id },
   102→    data: { status: 'RECEIVED', receivedDate: new Date() },
   103→  })
   104→}

   175→      default:
   176→        return undefined
   177→    }
   178→  } catch {
   179→    return undefined
   180→  }

   196→export async function GET(req: NextRequest) {
   197→  const { searchParams } = new URL(req.url)
   198→  const type = searchParams.get('type') || ''
   199→  const id = searchParams.get('id')
   200→  const residentId = searchParams.get('residentId')
   201→  const upcoming = searchParams.get('upcoming') === 'true'
   202→  const unbilled = searchParams.get('unbilled') === 'true'
   203→  const date = searchParams.get('date')
   204→  const facilityId = searchParams.get('facilityId') // selected facility from frontend
   205→
   206→  // Auth check — all data access requires login
   207→  const currentUser = await getSessionUser(req)
   208→  if (!currentUser) {
   209→    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
   210→  }
   211→
   212→  // Determine which facility IDs the user can access.
   213→  // Hierarchy:
   214→  //   - APP_DEVELOPER (L0): sees ALL facilities across ALL organizations (no scope)
   215→  //   - OWNER (L1): sees only facilities in their OWN organization
   216→  //   - MANAGER (L2) and below: sees only their assigned facilities
   217→  let accessibleFacilityIds: string[] = []
   218→  if (currentUser.role === 'APP_DEVELOPER') {
   219→    // Developer: no scope — sees everything
   220→    if (facilityId) accessibleFacilityIds = [facilityId]
   221→  } else if (currentUser.level === 1) {
   222→    // Owner: scoped to their organization's facilities
   223→    if (!currentUser.organizationId) {
   224→      // Owner without org → sees nothing (data isolation safeguard)
   225→      accessibleFacilityIds = ['__NO_ORG__'] // impossible ID → returns no rows
   226→    } else {
   227→      const orgFacilities = await db.facility.findMany({
   228→        where: { organizationId: currentUser.organizationId },
   229→        select: { id: true },
   230→      })
   231→      const orgFidSet = orgFacilities.map(f => f.id)
   232→      if (facilityId && orgFidSet.includes(facilityId)) {
   233→        accessibleFacilityIds = [facilityId]
   234→      } else {
   235→        accessibleFacilityIds = orgFidSet
   236→      }
   237→    }
   238→  } else {
   239→    // Non-owner: only their assigned facilities
   240→    const userFacilityIds = (currentUser.facilityIds || '').split(',').map(s => s.trim()).filter(Boolean)
   241→    if (facilityId && userFacilityIds.includes(facilityId)) {
   242→      accessibleFacilityIds = [facilityId]
   243→    } else {
   244→      accessibleFacilityIds = userFacilityIds
   245→    }
   246→  }
   247→
   248→  // Helper: build facility filters for queries
   249→  // Different models need different filter strategies based on how they relate to Facility:
   250→  //   - Direct facilityId          → residents, rooms, staff, expenses, products, inventory, invoices
   251→  //   - Via resident.facilityId    → medications, medAdmins, vitals, visits, incidents, careLogs, messages, invoiceItems
   252→  //   - Via staff.facilityId       → shifts
   253→  //   - Via item.facilityId        → inventoryTransactions
   254→  //   - No facility link           → auditLogs (system-wide), users (filter via facilityIds string match)
   255→  //
   256→  const hasFacilityScope = accessibleFacilityIds.length > 0
   257→  const facilityFilter = hasFacilityScope ? { facilityId: { in: accessibleFacilityIds } } : {}
   258→  const residentFacilityFilter = hasFacilityScope ? { resident: { facilityId: { in: accessibleFacilityIds } } } : {}
   259→  const staffFacilityFilter = hasFacilityScope ? { staff: { facilityId: { in: accessibleFacilityIds } } } : {}
   260→  const itemFacilityFilter = hasFacilityScope ? { item: { facilityId: { in: accessibleFacilityIds } } } : {}
   261→  // For users: facilityIds is a comma-separated string. Filter where the comma-separated list contains any of the accessible facility IDs.
   262→  // SQLite doesn't have native array operations, so we use contains for each ID. This is approximate but works for Owner's single-facility view.
   263→  const userFacilityFilter = hasFacilityScope
   264→    ? { OR: accessibleFacilityIds.flatMap(fid => [
   265→        { facilityIds: { contains: fid } },
   266→        { level: 1, role: 'OWNER' as any }, // Owner always visible (Owner sees all facilities)
   267→      ]) }
   268→    : {}
   269→
   270→  // For FAMILY role, restrict to linked residents only
   271→  const linkedResidentIds: string[] = currentUser.role === 'FAMILY'
   272→    ? (currentUser.linkedResidentIds || '').split(',').map(s => s.trim()).filter(Boolean)
   273→    : []
   274→
   275→  try {
   276→    // FAMILY role: block access to sensitive data types that they should never see directly.
   277→    // NOTE: incidents, careLogs, and visits are ALLOWED for family users — they are scoped
   278→    // to their linked residents via the `linkedResidentIds` filter in each case handler.
   279→    if (currentUser.role === 'FAMILY') {
   280→      const blockedTypes = ['rooms', 'staff', 'medications', 'medAdmins', 'invoices', 'invoiceItems', 'expenses', 'payments', 'paymentApplications', 'deposits', 'accounts', 'journalEntries', 'vendors', 'bankAccounts', 'shifts', 'users', 'products', 'inventory', 'inventoryTransactions', 'finance']
   281→      if (blockedTypes.includes(type)) {
   282→        return NextResponse.json({ error: 'Access denied for your role' }, { status: 403 })
   283→      }
   284→      // incidents, careLogs, visits, messages are allowed — each handler restricts to linkedResidentIds
   285→    }
   286→
   287→    switch (type) {
   288→      case 'residents': {
   289→        // FAMILY users can only see their linked residents
   290→        if (currentUser.role === 'FAMILY') {
   291→          if (id) {
   292→            // Verify this resident is linked to this family user
   293→            if (!linkedResidentIds.includes(id)) {
   294→              return NextResponse.json({ error: 'Access denied' }, { status: 403 })
   295→            }
   296→          }
   297→          // Include the same relations as the staff query so the family user sees
   298→          // incidents, care logs, visits, vitals, and med administrations in the resident detail view.
   299→          // (Billing/financial relations like invoiceItems are intentionally excluded for family.)
   300→          const list = await db.resident.findMany({
   301→            where: { id: { in: linkedResidentIds }, status: 'ACTIVE' },
   302→            include: {
   303→              room: true,
   304→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   305→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   306→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   307→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   308→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   309→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   310→            },
   311→            orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   312→          })
   313→          if (id) {
   314→            const r = list.find(x => x.id === id)
   315→            if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   316→            return NextResponse.json(r)
   317→          }
   318→          return NextResponse.json(list)
   319→        }
   320→
   321→        if (id) {
   322→          const r = await db.resident.findUnique({
   323→            where: { id },
   324→            include: {
   325→              room: true,
   326→              medications: { where: { active: true }, orderBy: { name: 'asc' } },
   327→              vitals: { orderBy: { recordedAt: 'desc' }, take: 30 },
   328→              careLogs: { orderBy: { recordedAt: 'desc' }, take: 30, include: { staff: true } },
   329→              visits: { orderBy: { scheduledAt: 'desc' }, take: 20, include: { staff: true } },
   330→              incidents: { orderBy: { occurredAt: 'desc' }, take: 10, include: { reportedBy: true } },
   331→              medAdmins: { orderBy: { scheduledAt: 'desc' }, take: 30, include: { medication: true, staff: true } },
   332→              invoiceItems: { where: { billed: false }, orderBy: { createdAt: 'desc' } },
   333→            },
   334→          })
   335→          if (!r) return NextResponse.json({ error: 'Not found' }, { status: 404 })
   336→          return NextResponse.json(r)
   337→        }
   338→        const includeArchived = searchParams.get('includeArchived') === 'true'
   339→        const statusFilter = searchParams.get('status') // ACTIVE, DISCHARGED, etc.
   340→        const where: any = { ...facilityFilter }
   341→        if (statusFilter) {
   342→          where.status = statusFilter
   343→        } else if (!includeArchived) {
   344→          where.status = 'ACTIVE'
   345→        }
   346→        const list = await db.resident.findMany({
   347→          where,
   348→          include: { room: true },
   349→          orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
   350→        })
   351→        return NextResponse.json(list)
   352→      }
   353→
   354→      case 'rooms': {
   355→        const rooms = await db.room.findMany({
   356→          include: { residents: { where: { status: 'ACTIVE' } } },
   357→          orderBy: { roomNumber: 'asc' },
   358→        })
   359→        return NextResponse.json(rooms)
   360→      }
   361→
   362→      case 'staff': {
   363→        const includeArchived = searchParams.get('includeArchived') === 'true'
   364→        // Multi-facility match: staff may be assigned via facilityId (primary) OR via the
   365→        // comma-separated facilityIds field (multi-facility assignment).
   366→        const where: any = hasFacilityScope
   367→          ? { OR: [
   368→              { facilityId: { in: accessibleFacilityIds } },
   369→              ...accessibleFacilityIds.flatMap(fid => [{ facilityIds: { contains: fid } }]),
   370→            ] }
   371→          : {}
   372→        if (!includeArchived) where.active = true
   373→        const staff = await db.staff.findMany({
   374→          where,
   375→          include: { _count: { select: { shifts: true, visits: true } } },
   376→          orderBy: [{ role: 'asc' }, { lastName: 'asc' }],
   377→        })
   378→        return NextResponse.json(staff)
   379→      }
   380→
   381→      case 'medications': {
   382→        const where: any = { ...residentFacilityFilter }
   383→        if (residentId) where.residentId = residentId
   384→        if (searchParams.get('active') !== 'false') where.active = true
   385→        const meds = await db.medication.findMany({
   386→          where,
   387→          include: { resident: { include: { room: true } } },
   388→          orderBy: { name: 'asc' },
   389→        })
   390→        return NextResponse.json(meds)
   391→      }
   392→
   393→      case 'medAdmins': {
   394→        const where: any = { ...residentFacilityFilter }
   395→        if (residentId) where.residentId = residentId
   396→        if (searchParams.get('today') === 'true') {
   397→          const start = new Date(); start.setHours(0, 0, 0, 0)
   398→          const end = new Date(); end.setHours(23, 59, 59, 999)
   399→          where.scheduledAt = { gte: start, lte: end }
   400→        } else {
   401→          // Generic date-range filter (yyyy-MM-dd, inclusive on both ends).
   402→          // Server-side; reduces payload when users pick a preset like Last 7 / This month.
   403→          const startDate = searchParams.get('startDate')
   404→          const endDate = searchParams.get('endDate')
   405→          if (startDate || endDate) {
   406→            where.scheduledAt = {}
   407→            if (startDate) {
   408→              const s = new Date(startDate); s.setHours(0, 0, 0, 0)
   409→              where.scheduledAt.gte = s
   410→            }
   411→            if (endDate) {
   412→              const e = new Date(endDate); e.setHours(23, 59, 59, 999)
   413→              where.scheduledAt.lte = e
   414→            }
   415→          }
   416→        }
   417→        const status = searchParams.get('status')
   418→        if (status) where.status = status
   419→        const admins = await db.medAdministration.findMany({
   420→          where,
   421→          include: { medication: true, resident: { include: { room: true } }, staff: true },
   422→          orderBy: { scheduledAt: 'desc' },
   423→          take: 500,
   424→        })
   425→        return NextResponse.json(admins)
   426→      }
   427→
   428→      case 'vitals': {
   429→        const where: any = { ...residentFacilityFilter }
   430→        if (residentId) where.residentId = residentId
   431→        // FAMILY users: restrict to their linked residents only
   432→        if (currentUser.role === 'FAMILY') {
   433→          if (linkedResidentIds.length === 0) return NextResponse.json([])
   434→          where.residentId = { in: linkedResidentIds }
   435→        }
   436→        // Date-range filter (yyyy-MM-dd, inclusive on both ends)
   437→        const startDate = searchParams.get('startDate')
   438→        const endDate = searchParams.get('endDate')
   439→        if (startDate || endDate) {
   440→          where.recordedAt = {}
   441→          if (startDate) {
   442→            const s = new Date(startDate); s.setHours(0, 0, 0, 0)
   443→            where.recordedAt.gte = s
   444→          }
   445→          if (endDate) {
   446→            const e = new Date(endDate); e.setHours(23, 59, 59, 999)
   447→            where.recordedAt.lte = e
   448→          }

   985→            fromFacility: { select: { id: true, name: true } },
   986→            toFacility: { select: { id: true, name: true } },
   987→            lines: {
   988→              include: {
   989→                item: { select: { id: true, name: true, code: true, unit: true, currentStock: true } },
   990→                destinationItem: { select: { id: true, name: true, code: true, currentStock: true } },
   991→              },
   992→            },
   993→          },
   994→          orderBy: { transferDate: 'desc' },
   995→          take: 500,
   996→        })
   997→        return NextResponse.json(transfers)
   998→      }
   999→
  1000→      case 'bankAccounts': {
  1001→        const where: any = { ...facilityFilter }
  1002→        const includeInactive = searchParams.get('includeInactive') === 'true'
  1003→        if (!includeInactive) where.active = true
  1004→        const banks = await db.bankAccount.findMany({
  1005→          where,
  1006→          include: {
  1007→            account: {
  1008→              include: {
  1009→                journalLines: {
  1010→                  where: { journalEntry: { posted: true } },
  1011→                  select: { debit: true, credit: true },
  1012→                },
  1013→              },
  1014→            },
  1015→          },
  1016→          orderBy: { code: 'asc' },
  1017→        })
  1018→        // Compute real current balance = openingBalance + sum(debits) - sum(credits)
  1019→        const banksWithBalance = banks.map(b => {
  1020→          const totalDebit = b.account?.journalLines?.reduce((s: number, l: any) => s + l.debit, 0) || 0
  1021→          const totalCredit = b.account?.journalLines?.reduce((s: number, l: any) => s + l.credit, 0) || 0
  1022→          const computedBalance = (b.openingBalance || 0) + totalDebit - totalCredit
  1023→          return {
  1024→            ...b,
  1025→            account: b.account ? { id: b.account.id, code: b.account.code, name: b.account.name } : null,
  1026→            totalDebit,
  1027→            totalCredit,
  1028→            currentBalance: Math.round(computedBalance * 100) / 100,
  1029→            transactionCount: b.account?.journalLines?.length || 0,
  1030→          }
  1031→        })
  1032→        return NextResponse.json(banksWithBalance)
  1033→      }
  1034→
  1035→      case 'deposits': {
  1036→        const where: any = { ...facilityFilter }
  1037→        const depResidentId = searchParams.get('residentId')
  1038→        if (depResidentId) where.residentId = depResidentId
  1039→        const depStatus = searchParams.get('status')
  1040→        if (depStatus) where.status = depStatus
  1041→        const deposits = await db.deposit.findMany({
  1042→          where,
  1043→          include: { resident: { select: { id: true, code: true, firstName: true, lastName: true } } },
  1044→          orderBy: { paymentDate: 'desc' },
  1045→          take: 500,
  1046→        })
  1047→        return NextResponse.json(deposits)
  1048→      }
  1049→
  1050→      default:
  1051→        return NextResponse.json({ error: 'Unknown type' }, { status: 400 })
  1052→    }
  1053→  } catch (e: any) {
  1054→    console.error('API /data error:', e)
  1055→    return NextResponse.json({ error: e.message }, { status: 500 })
  1056→  }
  1057→}
